# 06 - 窗口和应用程序管理

本章介绍如何使用 FlaUI 管理应用程序和窗口，包括启动、附加、关闭、窗口操作等。

---

## 应用程序启动

### 1. 启动应用程序

```csharp
using FlaUI.Core;

// 方式1：通过可执行文件名启动
var app = Application.Launch("notepad.exe");

// 方式2：通过完整路径启动
var app = Application.Launch(@"C:\Windows\System32\calc.exe");

// 方式3：启动并传递参数
var app = Application.Launch("notepad.exe", @"C:\test.txt");

// 方式4：使用ProcessStartInfo（更多控制）
var startInfo = new System.Diagnostics.ProcessStartInfo
{
    FileName = "notepad.exe",
    Arguments = @"C:\test.txt",
    WorkingDirectory = @"C:\",
    WindowStyle = System.Diagnostics.ProcessWindowStyle.Maximized
};
var app = Application.Launch(startInfo);
```

### 2. 附加到已运行的应用

```csharp
// 方式1：通过进程名附加
var app = Application.Attach("notepad.exe");

// 方式2：通过进程ID附加
var app = Application.Attach(12345);

// 方式3：通过进程对象附加
var process = System.Diagnostics.Process.GetProcessesByName("notepad")[0];
var app = Application.Attach(process);

// 方式4：查找并附加到特定窗口标题的进程
var processes = System.Diagnostics.Process.GetProcessesByName("notepad");
foreach (var proc in processes)
{
    var tempApp = Application.Attach(proc);
    var window = tempApp.GetMainWindow(automation);
    if (window.Title.Contains("特定标题"))
    {
        app = tempApp;
        break;
    }
}
```

### 完整示例

```csharp
using System;
using System.Linq;
using FlaUI.Core;
using FlaUI.UIA3;

class ApplicationExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        
        // 检查应用是否已运行
        var existingProcesses = System.Diagnostics.Process.GetProcessesByName("notepad");
        
        Application app;
        if (existingProcesses.Any())
        {
            Console.WriteLine("记事本已运行，附加到现有进程");
            app = Application.Attach(existingProcesses[0]);
        }
        else
        {
            Console.WriteLine("启动新的记事本进程");
            app = Application.Launch("notepad.exe");
        }
        
        // 获取主窗口
        var window = app.GetMainWindow(automation);
        Console.WriteLine($"窗口标题: {window.Title}");
        
        System.Threading.Thread.Sleep(3000);
        
        // 关闭应用
        app.Close();
        Console.WriteLine("应用已关闭");
    }
}
```

## 获取窗口

### 1. 获取主窗口

```csharp
using FlaUI.Core;
using FlaUI.UIA3;

using var automation = new UIA3Automation();
var app = Application.Launch("calc.exe");

// 获取主窗口（自动等待窗口出现）
var mainWindow = app.GetMainWindow(automation);

// 带超时的获取
var mainWindow = app.GetMainWindow(automation, timeout: TimeSpan.FromSeconds(10));
```

### 2. 获取所有窗口

```csharp
// 获取应用的所有窗口
var allWindows = app.GetAllTopLevelWindows(automation);

Console.WriteLine($"应用有 {allWindows.Length} 个窗口:");
foreach (var win in allWindows)
{
    Console.WriteLine($"  - {win.Title} (ID: {win.AutomationId})");
}
```

### 3. 通过条件查找窗口

```csharp
// 通过标题查找窗口
var window = automation.GetDesktop()
    .FindFirstChild(cf => cf.ByName("计算器").And(cf.ByControlType(ControlType.Window)))
    ?.AsWindow();

// 查找所有计算器窗口
var calcWindows = automation.GetDesktop()
    .FindAllChildren(cf => cf.ByClassName("CalcFrame"));
```

## 窗口操作

### 1. 窗口状态

```csharp
using FlaUI.Core.Definitions;

var window = app.GetMainWindow(automation);

// 最小化
window.SetWindowVisualState(WindowVisualState.Minimized);

// 最大化
window.SetWindowVisualState(WindowVisualState.Maximized);

// 正常大小
window.SetWindowVisualState(WindowVisualState.Normal);

// 获取当前状态
var state = window.WindowVisualState;
Console.WriteLine($"窗口状态: {state}");
```

### 2. 窗口大小和位置

```csharp
// 获取窗口边界
var bounds = window.BoundingRectangle;
Console.WriteLine($"位置: ({bounds.X}, {bounds.Y})");
Console.WriteLine($"大小: {bounds.Width} × {bounds.Height}");

// 移动窗口
window.Move(100, 100);

// 调整大小（某些窗口可能不支持）
// 注意：并非所有窗口都支持直接调整大小
```

### 3. 窗口焦点

```csharp
// 设置焦点
window.SetForeground();

// 检查是否有焦点
bool hasFocus = window.IsFocused;
Console.WriteLine($"窗口是否有焦点: {hasFocus}");
```

### 4. 窗口属性

```csharp
// 窗口标题
string title = window.Title;

// 窗口类名
string className = window.ClassName;

// 窗口句柄
IntPtr handle = window.Properties.NativeWindowHandle.Value;
Console.WriteLine($"窗口句柄: 0x{handle:X}");

// 检查窗口状态
bool isModal = window.IsModal;
bool isTopmost = window.IsTopmost;
bool isEnabled = window.IsEnabled;

Console.WriteLine($"模态窗口: {isModal}");
Console.WriteLine($"置顶窗口: {isTopmost}");
Console.WriteLine($"窗口可用: {isEnabled}");
```

### 完整示例

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class WindowOperationsExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        
        // 打印窗口信息
        PrintWindowInfo(window);
        
        // 最大化窗口
        Console.WriteLine("\n最大化窗口...");
        window.SetWindowVisualState(WindowVisualState.Maximized);
        System.Threading.Thread.Sleep(1000);
        
        // 恢复正常
        Console.WriteLine("恢复正常大小...");
        window.SetWindowVisualState(WindowVisualState.Normal);
        System.Threading.Thread.Sleep(1000);
        
        // 移动窗口
        Console.WriteLine("移动窗口到 (200, 200)...");
        window.Move(200, 200);
        System.Threading.Thread.Sleep(1000);
        
        // 最小化
        Console.WriteLine("最小化窗口...");
        window.SetWindowVisualState(WindowVisualState.Minimized);
        System.Threading.Thread.Sleep(1000);
        
        // 恢复并设置焦点
        Console.WriteLine("恢复窗口...");
        window.SetWindowVisualState(WindowVisualState.Normal);
        window.SetForeground();
        
        System.Threading.Thread.Sleep(2000);
        app.Close();
    }
    
    static void PrintWindowInfo(Window window)
    {
        Console.WriteLine("=== 窗口信息 ===");
        Console.WriteLine($"标题: {window.Title}");
        Console.WriteLine($"类名: {window.ClassName}");
        Console.WriteLine($"AutomationId: {window.AutomationId}");
        
        var bounds = window.BoundingRectangle;
        Console.WriteLine($"位置: ({bounds.X}, {bounds.Y})");
        Console.WriteLine($"大小: {bounds.Width} × {bounds.Height}");
        Console.WriteLine($"状态: {window.WindowVisualState}");
        Console.WriteLine($"可用: {window.IsEnabled}");
        Console.WriteLine($"有焦点: {window.IsFocused}");
    }
}
```

## 关闭应用程序

### 1. 正常关闭

```csharp
// 关闭应用（发送关闭消息）
app.Close();

// 关闭主窗口
var window = app.GetMainWindow(automation);
window.Close();
```

### 2. 强制终止

```csharp
// 强制终止进程
app.Kill();

// 终止进程树（包括子进程）
app.KillAll();
```

### 3. 等待应用退出

```csharp
// 关闭应用
app.Close();

// 等待应用退出
bool exited = app.WaitWhileBusy(TimeSpan.FromSeconds(5));
if (!exited)
{
    Console.WriteLine("应用未正常退出，强制终止");
    app.Kill();
}
```

### 4. 处理保存对话框

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class CloseWithDialogExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        
        // 输入一些文本
        var editBox = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.Edit))?.AsTextBox();
        editBox.Text = "测试文本";
        
        System.Threading.Thread.Sleep(500);
        
        // 尝试关闭
        app.Close();
        System.Threading.Thread.Sleep(500);
        
        // 查找保存对话框
        var desktop = automation.GetDesktop();
        var saveDialog = desktop.FindFirstChild(cf => 
            cf.ByName("记事本").And(cf.ByControlType(ControlType.Window)));
        
        if (saveDialog != null)
        {
            Console.WriteLine("检测到保存对话框");
            
            // 点击"不保存"按钮
            var dontSaveButton = saveDialog.FindFirstDescendant(cf => 
                cf.ByName("不保存").Or(cf.ByName("Don't Save")))?.AsButton();
            
            if (dontSaveButton != null)
            {
                dontSaveButton.Invoke();
                Console.WriteLine("已点击'不保存'");
            }
        }
        
        Console.WriteLine("应用已关闭");
    }
}
```

## 多窗口管理

### 1. 处理对话框

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;

class DialogExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        var window = app.GetMainWindow(automation);
        
        // 打开"关于"对话框
        var helpMenu = window.FindFirstDescendant(cf => 
            cf.ByName("帮助").And(cf.ByControlType(ControlType.MenuItem)))?.AsMenuItem();
        helpMenu?.Invoke();
        
        System.Threading.Thread.Sleep(300);
        
        var aboutMenu = window.FindFirstDescendant(cf => 
            cf.ByName("关于记事本").And(cf.ByControlType(ControlType.MenuItem)))?.AsMenuItem();
        aboutMenu?.Invoke();
        
        System.Threading.Thread.Sleep(500);
        
        // 查找"关于"对话框
        var desktop = automation.GetDesktop();
        var aboutDialog = desktop.FindFirstChild(cf => 
            cf.ByName("关于记事本").And(cf.ByControlType(ControlType.Window)));
        
        if (aboutDialog != null)
        {
            Console.WriteLine($"找到对话框: {aboutDialog.AsWindow().Title}");
            
            // 关闭对话框
            var okButton = aboutDialog.FindFirstDescendant(cf => 
                cf.ByName("确定").And(cf.ByControlType(ControlType.Button)))?.AsButton();
            okButton?.Invoke();
            
            Console.WriteLine("对话框已关闭");
        }
        
        System.Threading.Thread.Sleep(1000);
        app.Close();
    }
}
```

### 2. 切换窗口

```csharp
using System;
using System.Linq;
using FlaUI.Core;
using FlaUI.Core.Definitions;

class SwitchWindowExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        
        // 启动多个记事本实例
        var app1 = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var app2 = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        // 获取所有记事本窗口
        var desktop = automation.GetDesktop();
        var notepadWindows = desktop.FindAllChildren(cf => 
            cf.ByClassName("Notepad"));
        
        Console.WriteLine($"找到 {notepadWindows.Length} 个记事本窗口");
        
        // 操作每个窗口
        for (int i = 0; i < notepadWindows.Length; i++)
        {
            var win = notepadWindows[i].AsWindow();
            
            // 设置焦点
            win.SetForeground();
            Console.WriteLine($"\n当前窗口 {i + 1}: {win.Title}");
            
            // 输入文本
            var editBox = win.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit))?.AsTextBox();
            editBox.Text = $"这是窗口 {i + 1}";
            
            System.Threading.Thread.Sleep(1000);
        }
        
        System.Threading.Thread.Sleep(2000);
        
        // 关闭所有窗口
        app1.Close();
        app2.Close();
        
        Console.WriteLine("\n所有窗口已关闭");
    }
}
```

## 窗口等待

### 1. 等待窗口出现

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Definitions;

public static class WindowWaiter
{
    /// <summary>
    /// 等待窗口出现
    /// </summary>
    public static Window WaitForWindow(AutomationBase automation, string windowName, 
        int timeoutMs = 5000)
    {
        var startTime = DateTime.Now;
        var desktop = automation.GetDesktop();
        
        while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
        {
            var window = desktop.FindFirstChild(cf => 
                cf.ByName(windowName).And(cf.ByControlType(ControlType.Window)))?.AsWindow();
            
            if (window != null)
            {
                return window;
            }
            
            System.Threading.Thread.Sleep(500);
        }
        
        return null;
    }
    
    /// <summary>
    /// 等待窗口消失
    /// </summary>
    public static bool WaitForWindowClose(Window window, int timeoutMs = 5000)
    {
        var startTime = DateTime.Now;
        
        while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
        {
            try
            {
                // 尝试访问窗口属性
                var _ = window.Name;
                System.Threading.Thread.Sleep(500);
            }
            catch
            {
                // 窗口已关闭
                return true;
            }
        }
        
        return false;
    }
}

// 使用示例
class WaitExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        
        // 启动应用
        var app = Application.Launch("calc.exe");
        
        // 等待窗口出现
        var window = WindowWaiter.WaitForWindow(automation, "计算器", 10000);
        
        if (window != null)
        {
            Console.WriteLine($"窗口已出现: {window.Title}");
            
            // 进行操作...
            System.Threading.Thread.Sleep(2000);
            
            // 关闭窗口
            app.Close();
            
            // 等待窗口关闭
            bool closed = WindowWaiter.WaitForWindowClose(window, 5000);
            Console.WriteLine(closed ? "窗口已关闭" : "窗口关闭超时");
        }
        else
        {
            Console.WriteLine("窗口未出现");
            app.Kill();
        }
    }
}
```

## 应用程序信息

### 1. 获取进程信息

```csharp
var app = Application.Launch("notepad.exe");

// 进程ID
int processId = app.ProcessId;
Console.WriteLine($"进程ID: {processId}");

// 进程名称
string processName = app.Name;
Console.WriteLine($"进程名称: {processName}");

// 是否正在运行
bool isRunning = !app.HasExited;
Console.WriteLine($"是否运行: {isRunning}");

// 主窗口句柄
IntPtr mainWindowHandle = app.MainWindowHandle;
Console.WriteLine($"主窗口句柄: 0x{mainWindowHandle:X}");
```

### 2. 监控应用程序状态

```csharp
using System;
using System.Diagnostics;
using FlaUI.Core;

class ApplicationMonitor
{
    static void Main()
    {
        var app = Application.Launch("notepad.exe");
        
        Console.WriteLine("监控应用程序状态（按任意键停止）...\n");
        
        while (!Console.KeyAvailable)
        {
            try
            {
                if (!app.HasExited)
                {
                    var process = Process.GetProcessById(app.ProcessId);
                    
                    Console.WriteLine($"[{DateTime.Now:HH:mm:ss}] " +
                        $"CPU: {process.TotalProcessorTime.TotalSeconds:F2}s, " +
                        $"内存: {process.WorkingSet64 / 1024 / 1024}MB");
                }
                else
                {
                    Console.WriteLine("应用程序已退出");
                    break;
                }
            }
            catch (Exception ex)
            {
                Console.WriteLine($"错误: {ex.Message}");
                break;
            }
            
            System.Threading.Thread.Sleep(1000);
        }
        
        if (!app.HasExited)
        {
            app.Close();
        }
    }
}
```
