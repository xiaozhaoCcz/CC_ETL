# 09 - 截图和日志记录

在自动化测试中，截图和日志记录对于问题定位和测试报告至关重要。本章介绍如何使用 FlaUI 进行截图和日志记录。

---

## 截图功能

### 1. 截取整个屏幕

```csharp
using FlaUI.Core.Capturing;
using System.Drawing;
using System.Drawing.Imaging;

// 截取整个屏幕
var capture = Capture.Screen();

// 保存截图
capture.ToFile("screenshot.png");

// 指定图片格式
capture.ToFile("screenshot.jpg", ImageFormat.Jpeg);

// 获取 Bitmap 对象
Bitmap bitmap = capture.Bitmap;
```

### 2. 截取特定区域

```csharp
using FlaUI.Core.Capturing;
using System.Drawing;

// 定义截取区域
var rectangle = new Rectangle(100, 100, 800, 600);

// 截取指定区域
var capture = Capture.Rectangle(rectangle);
capture.ToFile("region.png");
```

### 3. 截取元素

```csharp
using FlaUI.Core;
using FlaUI.Core.Capturing;
using FlaUI.UIA3;

using var automation = new UIA3Automation();
var app = Application.Launch("notepad.exe");
var window = app.GetMainWindow(automation);

// 截取窗口
var windowCapture = Capture.Element(window);
windowCapture.ToFile("window.png");

// 截取特定元素
var editBox = window.FindFirstDescendant(cf => cf.ByControlType(ControlType.Edit));
var elementCapture = Capture.Element(editBox);
elementCapture.ToFile("editbox.png");

app.Close();
```

### 4. 完整截图示例

```csharp
using System;
using System.IO;
using FlaUI.Core;
using FlaUI.Core.Capturing;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class ScreenshotExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("calc.exe");
        System.Threading.Thread.Sleep(1500);
        
        var window = app.GetMainWindow(automation);
        
        // 创建截图目录
        string screenshotDir = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "Screenshots");
        Directory.CreateDirectory(screenshotDir);
        
        // 截取整个屏幕
        Console.WriteLine("截取整个屏幕...");
        var screenCapture = Capture.Screen();
        screenCapture.ToFile(Path.Combine(screenshotDir, "01_fullscreen.png"));
        
        // 截取计算器窗口
        Console.WriteLine("截取计算器窗口...");
        var windowCapture = Capture.Element(window);
        windowCapture.ToFile(Path.Combine(screenshotDir, "02_calculator.png"));
        
        // 点击一些按钮
        var button1 = window.FindFirstDescendant(cf => 
            cf.ByName("一").And(cf.ByControlType(ControlType.Button)))?.AsButton();
        button1?.Invoke();
        System.Threading.Thread.Sleep(300);
        
        // 截取操作后的状态
        Console.WriteLine("截取操作后的状态...");
        var afterClickCapture = Capture.Element(window);
        afterClickCapture.ToFile(Path.Combine(screenshotDir, "03_after_click.png"));
        
        Console.WriteLine($"截图已保存到: {screenshotDir}");
        
        System.Threading.Thread.Sleep(1000);
        app.Close();
    }
}
```

## 截图工具类封装

### 带时间戳的截图类

```csharp
using System;
using System.IO;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Capturing;

public class Screenshot
{
    private readonly string _baseDirectory;
    
    public Screenshot(string baseDirectory = null)
    {
        _baseDirectory = baseDirectory ?? Path.Combine(
            AppDomain.CurrentDomain.BaseDirectory, 
            "Screenshots"
        );
        
        Directory.CreateDirectory(_baseDirectory);
    }
    
    /// <summary>
    /// 截取整个屏幕
    /// </summary>
    public string CaptureScreen(string filename = null)
    {
        filename = filename ?? $"Screen_{GetTimestamp()}.png";
        var fullPath = Path.Combine(_baseDirectory, filename);
        
        var capture = Capture.Screen();
        capture.ToFile(fullPath);
        
        Console.WriteLine($"截图已保存: {fullPath}");
        return fullPath;
    }
    
    /// <summary>
    /// 截取元素
    /// </summary>
    public string CaptureElement(AutomationElement element, string filename = null)
    {
        filename = filename ?? $"Element_{GetTimestamp()}.png";
        var fullPath = Path.Combine(_baseDirectory, filename);
        
        var capture = Capture.Element(element);
        capture.ToFile(fullPath);
        
        Console.WriteLine($"元素截图已保存: {fullPath}");
        return fullPath;
    }
    
    /// <summary>
    /// 截取窗口
    /// </summary>
    public string CaptureWindow(Window window, string filename = null)
    {
        filename = filename ?? $"Window_{window.Title}_{GetTimestamp()}.png";
        // 清理文件名中的非法字符
        filename = string.Join("_", filename.Split(Path.GetInvalidFileNameChars()));
        
        var fullPath = Path.Combine(_baseDirectory, filename);
        
        var capture = Capture.Element(window);
        capture.ToFile(fullPath);
        
        Console.WriteLine($"窗口截图已保存: {fullPath}");
        return fullPath;
    }
    
    /// <summary>
    /// 截取指定区域
    /// </summary>
    public string CaptureRegion(System.Drawing.Rectangle region, string filename = null)
    {
        filename = filename ?? $"Region_{GetTimestamp()}.png";
        var fullPath = Path.Combine(_baseDirectory, filename);
        
        var capture = Capture.Rectangle(region);
        capture.ToFile(fullPath);
        
        Console.WriteLine($"区域截图已保存: {fullPath}");
        return fullPath;
    }
    
    private string GetTimestamp()
    {
        return DateTime.Now.ToString("yyyyMMdd_HHmmss_fff");
    }
}

// 使用示例
class ScreenshotUsageExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        var screenshot = new Screenshot();
        
        // 截取屏幕
        screenshot.CaptureScreen("notepad_screen.png");
        
        // 截取窗口
        screenshot.CaptureWindow(window);
        
        // 输入一些文本
        var editBox = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.Edit))?.AsTextBox();
        editBox.Text = "测试文本";
        
        // 截取编辑框
        screenshot.CaptureElement(editBox, "editbox.png");
        
        System.Threading.Thread.Sleep(1000);
        app.Close();
        System.Threading.Thread.Sleep(500);
        
        // 点击"不保存"
        var desktop = automation.GetDesktop();
        var dialog = desktop.FindFirstChild(cf => cf.ByControlType(ControlType.Window));
        if (dialog != null)
        {
            screenshot.CaptureWindow(dialog.AsWindow(), "save_dialog.png");
            var noButton = dialog.FindFirstDescendant(cf => cf.ByName("不保存"))?.AsButton();
            noButton?.Invoke();
        }
    }
}
```

## 日志记录

### 1. 简单日志记录

```csharp
using System;
using System.IO;

public class Logger
{
    private readonly string _logFile;
    private readonly object _lock = new object();
    
    public Logger(string logFile = null)
    {
        _logFile = logFile ?? Path.Combine(
            AppDomain.CurrentDomain.BaseDirectory,
            $"TestLog_{DateTime.Now:yyyyMMdd_HHmmss}.txt"
        );
        
        // 写入日志头
        WriteLog("=== 测试日志开始 ===");
        WriteLog($"时间: {DateTime.Now}");
        WriteLog("====================\n");
    }
    
    public void Info(string message)
    {
        WriteLog($"[INFO] {message}");
    }
    
    public void Warning(string message)
    {
        WriteLog($"[WARN] {message}");
    }
    
    public void Error(string message, Exception ex = null)
    {
        WriteLog($"[ERROR] {message}");
        if (ex != null)
        {
            WriteLog($"异常: {ex.Message}");
            WriteLog($"堆栈: {ex.StackTrace}");
        }
    }
    
    public void Debug(string message)
    {
        WriteLog($"[DEBUG] {message}");
    }
    
    private void WriteLog(string message)
    {
        lock (_lock)
        {
            string logMessage = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss.fff}] {message}";
            Console.WriteLine(logMessage);
            
            try
            {
                File.AppendAllText(_logFile, logMessage + Environment.NewLine);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"写入日志文件失败: {ex.Message}");
            }
        }
    }
}

// 使用示例
class LoggerExample
{
    static void Main()
    {
        var logger = new Logger();
        
        logger.Info("启动应用程序");
        
        try
        {
            using var automation = new UIA3Automation();
            var app = Application.Launch("notepad.exe");
            logger.Info($"记事本已启动，进程ID: {app.ProcessId}");
            
            System.Threading.Thread.Sleep(1000);
            
            var window = app.GetMainWindow(automation);
            logger.Info($"获取到主窗口: {window.Title}");
            
            var editBox = window.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit))?.AsTextBox();
            
            if (editBox != null)
            {
                logger.Info("找到编辑框");
                editBox.Text = "测试日志功能";
                logger.Info("文本已输入");
            }
            else
            {
                logger.Warning("未找到编辑框");
            }
            
            System.Threading.Thread.Sleep(2000);
            
            logger.Info("准备关闭应用");
            app.Close();
            logger.Info("应用已关闭");
        }
        catch (Exception ex)
        {
            logger.Error("测试执行失败", ex);
        }
        
        logger.Info("=== 测试日志结束 ===");
    }
}
```

### 2. 增强版日志类（支持日志级别）

```csharp
using System;
using System.IO;

public enum LogLevel
{
    Debug,
    Info,
    Warning,
    Error
}

public class EnhancedLogger
{
    private readonly string _logFile;
    private readonly LogLevel _minLevel;
    private readonly object _lock = new object();
    
    public EnhancedLogger(LogLevel minLevel = LogLevel.Info, string logFile = null)
    {
        _minLevel = minLevel;
        _logFile = logFile ?? Path.Combine(
            AppDomain.CurrentDomain.BaseDirectory,
            "Logs",
            $"Test_{DateTime.Now:yyyyMMdd_HHmmss}.log"
        );
        
        Directory.CreateDirectory(Path.GetDirectoryName(_logFile));
        
        WriteHeader();
    }
    
    public void Debug(string message)
    {
        Log(LogLevel.Debug, message);
    }
    
    public void Info(string message)
    {
        Log(LogLevel.Info, message);
    }
    
    public void Warning(string message)
    {
        Log(LogLevel.Warning, message);
    }
    
    public void Error(string message, Exception ex = null)
    {
        Log(LogLevel.Error, message);
        if (ex != null)
        {
            Log(LogLevel.Error, $"  异常类型: {ex.GetType().Name}");
            Log(LogLevel.Error, $"  异常消息: {ex.Message}");
            Log(LogLevel.Error, $"  堆栈跟踪:\n{ex.StackTrace}");
        }
    }
    
    public void Step(string stepName)
    {
        Log(LogLevel.Info, $">>> 步骤: {stepName}");
    }
    
    public void Assert(string condition, bool result)
    {
        if (result)
        {
            Log(LogLevel.Info, $"✓ 断言通过: {condition}");
        }
        else
        {
            Log(LogLevel.Error, $"✗ 断言失败: {condition}");
        }
    }
    
    private void Log(LogLevel level, string message)
    {
        if (level < _minLevel)
        {
            return;
        }
        
        lock (_lock)
        {
            string levelStr = GetLevelString(level);
            string logMessage = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss.fff}] {levelStr} {message}";
            
            // 控制台输出（带颜色）
            var originalColor = Console.ForegroundColor;
            Console.ForegroundColor = GetLevelColor(level);
            Console.WriteLine(logMessage);
            Console.ForegroundColor = originalColor;
            
            // 文件输出
            try
            {
                File.AppendAllText(_logFile, logMessage + Environment.NewLine);
            }
            catch (Exception ex)
            {
                Console.WriteLine($"写入日志失败: {ex.Message}");
            }
        }
    }
    
    private void WriteHeader()
    {
        Log(LogLevel.Info, "===============================================");
        Log(LogLevel.Info, "自动化测试日志");
        Log(LogLevel.Info, $"开始时间: {DateTime.Now}");
        Log(LogLevel.Info, $"日志级别: {_minLevel}");
        Log(LogLevel.Info, "===============================================\n");
    }
    
    private string GetLevelString(LogLevel level)
    {
        return level switch
        {
            LogLevel.Debug => "[DEBUG]",
            LogLevel.Info => "[INFO ]",
            LogLevel.Warning => "[WARN ]",
            LogLevel.Error => "[ERROR]",
            _ => "[?????]"
        };
    }
    
    private ConsoleColor GetLevelColor(LogLevel level)
    {
        return level switch
        {
            LogLevel.Debug => ConsoleColor.Gray,
            LogLevel.Info => ConsoleColor.White,
            LogLevel.Warning => ConsoleColor.Yellow,
            LogLevel.Error => ConsoleColor.Red,
            _ => ConsoleColor.White
        };
    }
}

// 使用示例
class EnhancedLoggerExample
{
    static void Main()
    {
        var logger = new EnhancedLogger(LogLevel.Debug);
        
        logger.Step("启动应用程序");
        
        try
        {
            using var automation = new UIA3Automation();
            
            logger.Debug("创建 UIA3 自动化引擎");
            
            logger.Info("启动记事本");
            var app = Application.Launch("notepad.exe");
            logger.Info($"进程ID: {app.ProcessId}");
            
            System.Threading.Thread.Sleep(1000);
            
            logger.Step("获取主窗口");
            var window = app.GetMainWindow(automation);
            logger.Info($"窗口标题: {window.Title}");
            logger.Assert("窗口已获取", window != null);
            
            logger.Step("查找编辑框");
            var editBox = window.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit))?.AsTextBox();
            logger.Assert("编辑框已找到", editBox != null);
            
            logger.Step("输入文本");
            editBox.Text = "增强版日志测试";
            logger.Info("文本输入完成");
            
            // 模拟一个警告
            logger.Warning("这是一个警告示例");
            
            System.Threading.Thread.Sleep(2000);
            
            logger.Step("关闭应用");
            app.Close();
            logger.Info("应用已关闭");
        }
        catch (Exception ex)
        {
            logger.Error("测试执行异常", ex);
        }
        
        logger.Info("\n测试完成");
    }
}
```

## 组合使用截图和日志

### 测试辅助类

```csharp
using System;
using System.IO;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Capturing;

public class TestHelper
{
    private readonly EnhancedLogger _logger;
    private readonly Screenshot _screenshot;
    private int _stepCounter = 0;
    
    public TestHelper(string testName = null)
    {
        testName = testName ?? "Test";
        
        string baseDir = Path.Combine(
            AppDomain.CurrentDomain.BaseDirectory,
            "TestResults",
            $"{testName}_{DateTime.Now:yyyyMMdd_HHmmss}"
        );
        
        Directory.CreateDirectory(baseDir);
        
        _logger = new EnhancedLogger(
            LogLevel.Info,
            Path.Combine(baseDir, "test.log")
        );
        
        _screenshot = new Screenshot(Path.Combine(baseDir, "Screenshots"));
        
        _logger.Info($"测试名称: {testName}");
        _logger.Info($"输出目录: {baseDir}");
    }
    
    public void Step(string description, Action action, bool captureOnError = true)
    {
        _stepCounter++;
        string stepName = $"步骤 {_stepCounter}: {description}";
        
        _logger.Step(stepName);
        
        try
        {
            action();
            _logger.Info($"✓ {stepName} - 成功");
        }
        catch (Exception ex)
        {
            _logger.Error($"✗ {stepName} - 失败", ex);
            
            if (captureOnError)
            {
                try
                {
                    _screenshot.CaptureScreen($"Error_Step{_stepCounter}.png");
                }
                catch (Exception screenshotEx)
                {
                    _logger.Warning($"截图失败: {screenshotEx.Message}");
                }
            }
            
            throw;
        }
    }
    
    public void CaptureStep(string description, AutomationElement element = null)
    {
        _logger.Info($"📷 截图: {description}");
        
        if (element != null)
        {
            _screenshot.CaptureElement(element, $"Step{_stepCounter}_{description}.png");
        }
        else
        {
            _screenshot.CaptureScreen($"Step{_stepCounter}_{description}.png");
        }
    }
    
    public void Assert(string condition, bool result)
    {
        _logger.Assert(condition, result);
        
        if (!result)
        {
            _screenshot.CaptureScreen($"AssertFailed_{condition}.png");
            throw new Exception($"断言失败: {condition}");
        }
    }
    
    public void Info(string message) => _logger.Info(message);
    public void Warning(string message) => _logger.Warning(message);
    public void Error(string message, Exception ex = null) => _logger.Error(message, ex);
}

// 使用示例
class TestHelperExample
{
    static void Main()
    {
        var helper = new TestHelper("NotepadTest");
        
        try
        {
            Application app = null;
            Window window = null;
            UIA3Automation automation = null;
            
            helper.Step("初始化自动化引擎", () =>
            {
                automation = new UIA3Automation();
            });
            
            helper.Step("启动记事本", () =>
            {
                app = Application.Launch("notepad.exe");
                System.Threading.Thread.Sleep(1000);
            });
            
            helper.Step("获取主窗口", () =>
            {
                window = app.GetMainWindow(automation);
                helper.Assert("窗口不为空", window != null);
            });
            
            helper.CaptureStep("初始界面", window);
            
            helper.Step("输入文本", () =>
            {
                var editBox = window.FindFirstDescendant(cf => 
                    cf.ByControlType(ControlType.Edit))?.AsTextBox();
                helper.Assert("找到编辑框", editBox != null);
                
                editBox.Text = "这是一个测试";
                helper.Info("文本已输入");
            });
            
            helper.CaptureStep("输入文本后", window);
            
            System.Threading.Thread.Sleep(1000);
            
            helper.Step("关闭应用", () =>
            {
                app.Close();
                System.Threading.Thread.Sleep(500);
                
                // 处理保存对话框
                var desktop = automation.GetDesktop();
                var dialog = desktop.FindFirstChild(cf => cf.ByControlType(ControlType.Window));
                if (dialog != null)
                {
                    helper.CaptureStep("保存对话框", dialog);
                    var noButton = dialog.FindFirstDescendant(cf => cf.ByName("不保存"))?.AsButton();
                    noButton?.Invoke();
                }
            });
            
            helper.Info("\n✓ 测试全部通过");
        }
        catch (Exception ex)
        {
            helper.Error("测试失败", ex);
            throw;
        }
    }
}
```
