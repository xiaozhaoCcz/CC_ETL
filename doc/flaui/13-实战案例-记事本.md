## 记事本页面对象

### NotepadPage.cs

```csharp
using System;
using System.IO;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Definitions;
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;
using FlaUI.UIA3;

namespace NotepadTests.PageObjects
{
    public class NotepadPage : IDisposable
    {
        private readonly UIA3Automation _automation;
        private readonly Application _app;
        private Window _window;
        
        public NotepadPage(string filePath = null)
        {
            _automation = new UIA3Automation();
            
            if (string.IsNullOrEmpty(filePath))
            {
                _app = Application.Launch("notepad.exe");
            }
            else
            {
                _app = Application.Launch("notepad.exe", filePath);
            }
            
            System.Threading.Thread.Sleep(1000);
            _window = _app.GetMainWindow(_automation);
        }
        
        /// <summary>
        /// 获取编辑框
        /// </summary>
        private TextBox GetEditBox()
        {
            var editBox = _window.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit))?.AsTextBox();
            
            if (editBox == null)
            {
                throw new Exception("未找到编辑框");
            }
            
            return editBox;
        }
        
        /// <summary>
        /// 输入文本
        /// </summary>
        public void TypeText(string text)
        {
            _window.SetForeground();
            var editBox = GetEditBox();
            editBox.Text = text;
        }
        
        /// <summary>
        /// 获取文本内容
        /// </summary>
        public string GetText()
        {
            var editBox = GetEditBox();
            return editBox.Text;
        }
        
        /// <summary>
        /// 清空文本
        /// </summary>
        public void ClearText()
        {
            var editBox = GetEditBox();
            editBox.Text = "";
        }
        
        /// <summary>
        /// 追加文本
        /// </summary>
        public void AppendText(string text)
        {
            _window.SetForeground();
            var editBox = GetEditBox();
            editBox.Enter(text);
        }
        
        /// <summary>
        /// 打开文件菜单
        /// </summary>
        private void OpenFileMenu()
        {
            var fileMenu = _window.FindFirstDescendant(cf => 
                cf.ByName("文件").Or(cf.ByName("File")))?.AsMenuItem();
            
            if (fileMenu != null)
            {
                fileMenu.Invoke();
                System.Threading.Thread.Sleep(500);
            }
            else
            {
                // 使用快捷键打开
                Keyboard.Press(VirtualKeyShort.ALT);
                Keyboard.Press(VirtualKeyShort.KEY_F);
                Keyboard.Up(VirtualKeyShort.ALT);
                System.Threading.Thread.Sleep(500);
            }
        }
        
        /// <summary>
        /// 新建文件
        /// </summary>
        public void NewFile()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_N);
            System.Threading.Thread.Sleep(500);
            
            // 处理保存提示
            HandleSaveDialog(false);
        }
        
        /// <summary>
        /// 打开文件
        /// </summary>
        public void OpenFile(string filePath)
        {
            // Ctrl+O
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_O);
            System.Threading.Thread.Sleep(1000);
            
            // 查找打开对话框
            var desktop = _automation.GetDesktop();
            var openDialog = desktop.FindFirstChild(cf => 
                cf.ByName("打开").Or(cf.ByName("Open")));
            
            if (openDialog != null)
            {
                // 输入文件名
                var filenameBox = openDialog.FindFirstDescendant(cf => 
                    cf.ByAutomationId("1148"))?.AsTextBox();
                
                if (filenameBox != null)
                {
                    filenameBox.Text = filePath;
                    System.Threading.Thread.Sleep(300);
                    
                    // 点击打开按钮
                    var openButton = openDialog.FindFirstDescendant(cf => 
                        cf.ByName("打开").Or(cf.ByName("Open")))?.AsButton();
                    openButton?.Invoke();
                    System.Threading.Thread.Sleep(500);
                }
            }
        }
        
        /// <summary>
        /// 保存文件
        /// </summary>
        public void SaveFile(string filePath = null)
        {
            if (filePath == null)
            {
                // Ctrl+S (保存)
                Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_S);
            }
            else
            {
                // Ctrl+Shift+S (另存为) 或使用菜单
                SaveFileAs(filePath);
            }
            
            System.Threading.Thread.Sleep(500);
        }
        
        /// <summary>
        /// 另存为
        /// </summary>
        public void SaveFileAs(string filePath)
        {
            // 打开文件菜单
            OpenFileMenu();
            
            // 点击"另存为"
            var saveAsMenu = _window.FindFirstDescendant(cf => 
                cf.ByName("另存为").Or(cf.ByName("Save as")))?.AsMenuItem();
            saveAsMenu?.Invoke();
            
            System.Threading.Thread.Sleep(1000);
            
            // 查找保存对话框
            var desktop = _automation.GetDesktop();
            var saveDialog = desktop.FindFirstChild(cf => 
                cf.ByName("另存为").Or(cf.ByName("Save As")));
            
            if (saveDialog != null)
            {
                // 输入文件名
                var filenameBox = saveDialog.FindFirstDescendant(cf => 
                    cf.ByAutomationId("1001"))?.AsTextBox();
                
                if (filenameBox != null)
                {
                    filenameBox.Text = filePath;
                    System.Threading.Thread.Sleep(300);
                    
                    // 点击保存按钮
                    var saveButton = saveDialog.FindFirstDescendant(cf => 
                        cf.ByName("保存").Or(cf.ByName("Save")))?.AsButton();
                    saveButton?.Invoke();
                    System.Threading.Thread.Sleep(500);
                }
            }
        }
        
        /// <summary>
        /// 查找文本
        /// </summary>
        public void Find(string textToFind)
        {
            // Ctrl+F
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_F);
            System.Threading.Thread.Sleep(500);
            
            // 查找对话框
            var desktop = _automation.GetDesktop();
            var findDialog = desktop.FindFirstChild(cf => 
                cf.ByName("查找").Or(cf.ByName("Find")));
            
            if (findDialog != null)
            {
                // 输入查找内容
                var searchBox = findDialog.FindFirstDescendant(cf => 
                    cf.ByControlType(ControlType.Edit))?.AsTextBox();
                
                if (searchBox != null)
                {
                    searchBox.Text = textToFind;
                    System.Threading.Thread.Sleep(300);
                    
                    // 点击查找下一个
                    var findNextButton = findDialog.FindFirstDescendant(cf => 
                        cf.ByName("查找下一个").Or(cf.ByName("Find Next")))?.AsButton();
                    findNextButton?.Invoke();
                    System.Threading.Thread.Sleep(300);
                    
                    // 关闭查找对话框
                    findDialog.AsWindow().Close();
                }
            }
        }
        
        /// <summary>
        /// 替换文本
        /// </summary>
        public void Replace(string findText, string replaceText)
        {
            // Ctrl+H
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_H);
            System.Threading.Thread.Sleep(500);
            
            var desktop = _automation.GetDesktop();
            var replaceDialog = desktop.FindFirstChild(cf => 
                cf.ByName("替换").Or(cf.ByName("Replace")));
            
            if (replaceDialog != null)
            {
                var textBoxes = replaceDialog.FindAllDescendants(cf => 
                    cf.ByControlType(ControlType.Edit));
                
                if (textBoxes.Length >= 2)
                {
                    // 查找内容
                    textBoxes[0].AsTextBox().Text = findText;
                    System.Threading.Thread.Sleep(200);
                    
                    // 替换为
                    textBoxes[1].AsTextBox().Text = replaceText;
                    System.Threading.Thread.Sleep(200);
                    
                    // 点击全部替换
                    var replaceAllButton = replaceDialog.FindFirstDescendant(cf => 
                        cf.ByName("全部替换").Or(cf.ByName("Replace All")))?.AsButton();
                    replaceAllButton?.Invoke();
                    System.Threading.Thread.Sleep(500);
                    
                    // 关闭对话框
                    replaceDialog.AsWindow().Close();
                }
            }
        }
        
        /// <summary>
        /// 全选
        /// </summary>
        public void SelectAll()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_A);
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 复制
        /// </summary>
        public void Copy()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_C);
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 粘贴
        /// </summary>
        public void Paste()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_V);
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 撤销
        /// </summary>
        public void Undo()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_Z);
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 获取窗口标题
        /// </summary>
        public string GetWindowTitle()
        {
            return _window.Title;
        }
        
        /// <summary>
        /// 处理保存对话框
        /// </summary>
        private void HandleSaveDialog(bool save)
        {
            System.Threading.Thread.Sleep(500);
            
            var desktop = _automation.GetDesktop();
            var dialog = desktop.FindFirstChild(cf => 
                cf.ByControlType(ControlType.Window));
            
            if (dialog != null && dialog.Name.Contains("记事本"))
            {
                Button button;
                if (save)
                {
                    button = dialog.FindFirstDescendant(cf => 
                        cf.ByName("保存").Or(cf.ByName("Save")))?.AsButton();
                }
                else
                {
                    button = dialog.FindFirstDescendant(cf => 
                        cf.ByName("不保存").Or(cf.ByName("Don't Save")))?.AsButton();
                }
                
                button?.Invoke();
                System.Threading.Thread.Sleep(300);
            }
        }
        
        public void Dispose()
        {
            try
            {
                _app?.Close();
                System.Threading.Thread.Sleep(500);
                HandleSaveDialog(false);
            }
            catch { }
            finally
            {
                _automation?.Dispose();
            }
        }
    }
}
```

```csharp
using Xunit;
using NotepadTests.PageObjects;

namespace NotepadTests.Tests
{
    public class BasicTextOperationsTests
    {
        [Fact]
        public void TestInputText_TypeSimpleText_TextAppears()
        {
            using var notepad = new NotepadPage();
            
            // Act
            notepad.TypeText("Hello World!");
            
            // Assert
            string text = notepad.GetText();
            Assert.Equal("Hello World!", text);
        }
        
        [Fact]
        public void TestInputMultilineText_WithNewlines_FormatsCorrectly()
        {
            using var notepad = new NotepadPage();
            
            // Act
            string inputText = "第一行\n第二行\n第三行";
            notepad.TypeText(inputText);
            
            // Assert
            string text = notepad.GetText();
            Assert.Contains("第一行", text);
            Assert.Contains("第二行", text);
            Assert.Contains("第三行", text);
        }
        
        [Fact]
        public void TestClearText_AfterTyping_TextIsEmpty()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("测试文本");
            
            // Act
            notepad.ClearText();
            
            // Assert
            string text = notepad.GetText();
            Assert.Empty(text);
        }
        
        [Fact]
        public void TestAppendText_AddToExistingText_TextConcatenated()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("Hello ");
            
            // Act
            notepad.AppendText("World!");
            
            // Assert
            string text = notepad.GetText();
            Assert.Contains("Hello World!", text);
        }
    }
}
```

```csharp
using Xunit;
using NotepadTests.PageObjects;

namespace NotepadTests.Tests
{
    public class EditOperationsTests
    {
        [Fact]
        public void TestCopyPaste_CopyText_TextDuplicates()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("测试文本");
            
            // Act
            notepad.SelectAll();
            notepad.Copy();
            notepad.Paste();
            
            // Assert
            string text = notepad.GetText();
            Assert.Equal("测试文本测试文本", text);
        }
        
        [Fact]
        public void TestUndo_AfterTyping_TextReverted()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("原始文本");
            string originalText = notepad.GetText();
            
            // Act
            notepad.AppendText("新增文本");
            notepad.Undo();
            
            // Assert
            string text = notepad.GetText();
            Assert.Equal(originalText, text);
        }
        
        [Fact]
        public void TestFindText_SearchExistingText_TextFound()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("这是一段测试文本，包含关键字keyword在中间。");
            
            // Act
            notepad.Find("keyword");
            
            // Assert
            // 验证查找对话框出现并关闭
            System.Threading.Thread.Sleep(500);
            string text = notepad.GetText();
            Assert.Contains("keyword", text);
        }
        
        [Fact]
        public void TestReplaceText_ReplaceKeyword_TextChanged()
        {
            using var notepad = new NotepadPage();
            
            // Arrange
            notepad.TypeText("Hello Hello Hello");
            
            // Act
            notepad.Replace("Hello", "Hi");
            
            // Assert
            string text = notepad.GetText();
            Assert.Contains("Hi", text);
            Assert.DoesNotContain("Hello", text);
        }
    }
}
```

```csharp
using System;
using System.IO;
using Xunit;
using NotepadTests.PageObjects;

                // Arrange
```

## 集成测试

### ComprehensiveWorkflowTests.cs

```csharp
using System;
using System.IO;
using Xunit;
using Xunit.Abstractions;
using NotepadTests.PageObjects;

namespace NotepadTests.Tests
{
    public class ComprehensiveWorkflowTests
    {
        private readonly ITestOutputHelper _output;
        private readonly string _testDir;
        
        public ComprehensiveWorkflowTests(ITestOutputHelper output)
        {
            _output = output;
            _testDir = Path.Combine(Path.GetTempPath(), "NotepadTests");
            Directory.CreateDirectory(_testDir);
        }
        
        [Fact]
        public void TestCompleteWorkflow_CreateEditSave_Success()
        {
            string filePath = Path.Combine(_testDir, $"workflow_{Guid.NewGuid()}.txt");
            
            try
            {
                using (var notepad = new NotepadPage())
                {
                    _output.WriteLine("步骤 1: 输入初始文本");
                    notepad.TypeText("这是第一段文本。\n");
                    
                    _output.WriteLine("步骤 2: 追加更多文本");
                    notepad.AppendText("这是第二段文本。\n");
                    
                    _output.WriteLine("步骤 3: 全选并复制");
                    notepad.SelectAll();
                    notepad.Copy();
                    
                    _output.WriteLine("步骤 4: 移动到末尾并粘贴");
                    notepad.Paste();
                    
                    _output.WriteLine("步骤 5: 替换文本");
                    notepad.Replace("第一段", "第壹段");
                    
                    _output.WriteLine("步骤 6: 保存文件");
                    notepad.SaveFileAs(filePath);
                    
                    System.Threading.Thread.Sleep(1000);
                    
                    _output.WriteLine("步骤 7: 验证内容");
                    string content = notepad.GetText();
                    Assert.Contains("第壹段", content);
                    Assert.Contains("第二段", content);
                }
                
                // 验证文件已创建
                Assert.True(File.Exists(filePath), "文件应该被创建");
                _output.WriteLine($"✓ 文件已创建: {filePath}");
                
                // 验证文件内容
                string fileContent = File.ReadAllText(filePath);
                Assert.Contains("第壹段", fileContent);
                Assert.Contains("第二段", fileContent);
                _output.WriteLine("✓ 文件内容验证通过");
                
                // 重新打开文件验证
                using (var notepad = new NotepadPage(filePath))
                {
                    System.Threading.Thread.Sleep(1000);
                    string reopenedContent = notepad.GetText();
                    Assert.Contains("第壹段", reopenedContent);
                    _output.WriteLine("✓ 重新打开文件验证通过");
                }
            }
            finally
            {
                // Cleanup
                if (File.Exists(filePath))
                {
                    File.Delete(filePath);
                    _output.WriteLine($"清理: 已删除测试文件");
                }
            }
        }
    }
}
```
