## 代码组织原则

### 1. 页面对象模式（Page Object Pattern）

将页面/窗口的元素和操作封装到单独的类中。

**优点**：
- 提高代码复用性
- 降低维护成本
- 测试代码更清晰

```csharp
// ❌ 不好的做法：测试代码中直接操作元素
[Fact]
public void TestLogin()
{
    var usernameBox = window.FindFirstDescendant(cf => cf.ByAutomationId("username"));
    usernameBox.AsTextBox().Text = "admin";
    
    var passwordBox = window.FindFirstDescendant(cf => cf.ByAutomationId("password"));
    passwordBox.AsTextBox().Text = "123456";
    
    var loginButton = window.FindFirstDescendant(cf => cf.ByName("登录"));
    loginButton.AsButton().Invoke();
}

// ✅ 好的做法：使用页面对象
public class LoginPage
{
    private readonly Window _window;
    
    public LoginPage(Window window)
    {
        _window = window;
    }
    
    public void Login(string username, string password)
    {
        SetUsername(username);
        SetPassword(password);
        ClickLogin();
    }
    
    private void SetUsername(string username)
    {
        var usernameBox = _window.FindFirstDescendant(cf => 
            cf.ByAutomationId("username")).AsTextBox();
        usernameBox.Text = username;
    }
    
    private void SetPassword(string password)
    {
        var passwordBox = _window.FindFirstDescendant(cf => 
            cf.ByAutomationId("password")).AsTextBox();
        passwordBox.Text = password;
    }
    
    private void ClickLogin()
    {
        var loginButton = _window.FindFirstDescendant(cf => 
            cf.ByName("登录")).AsButton();
        loginButton.Invoke();
    }
}

// 测试代码变得简洁
[Fact]
public void TestLogin()
{
    var loginPage = new LoginPage(window);
    loginPage.Login("admin", "123456");
}
```

## 命名规范

### 1. 测试方法命名

使用清晰的命名模式：`Test[功能]_[条件]_[期望结果]`

```csharp
// ✅ 好的命名
[Fact]
public void TestLogin_WithValidCredentials_Success()
{
    // ...
}

[Fact]
public void TestLogin_WithInvalidPassword_ShowsError()
{
    // ...
}

[Fact]
public void TestAddButton_WhenClicked_AddsNewItem()
{
    // ...
}

// ❌ 不好的命名
[Fact]
public void Test1() { }

[Fact]
public void LoginTest() { }

[Fact]
public void TestMethod() { }
```

### 2. 变量命名

```csharp
// ✅ 清晰的变量名
var usernameTextBox = FindElement("UsernameTextBox").AsTextBox();
var loginButton = FindElement("LoginButton").AsButton();
var errorMessage = FindElement("ErrorLabel").Name;

// ❌ 不清晰的变量名
var tb1 = FindElement("UsernameTextBox").AsTextBox();
var btn = FindElement("LoginButton").AsButton();
var msg = FindElement("ErrorLabel").Name;
```

## DRY 原则（Don't Repeat Yourself）

### 1. 提取公共方法

```csharp
// ❌ 代码重复
[Fact]
public void TestAddUser()
{
    var button = window.FindFirstDescendant(cf => cf.ByAutomationId("AddButton")).AsButton();
    button.Invoke();
    System.Threading.Thread.Sleep(500);
}

[Fact]
public void TestDeleteUser()
{
    var button = window.FindFirstDescendant(cf => cf.ByAutomationId("DeleteButton")).AsButton();
    button.Invoke();
    System.Threading.Thread.Sleep(500);
}

// ✅ 提取公共方法
public class TestBase
{
    protected void ClickButton(string automationId)
    {
        var button = window.FindFirstDescendant(cf => 
            cf.ByAutomationId(automationId)).AsButton();
        button.Invoke();
        System.Threading.Thread.Sleep(500);
    }
}

[Fact]
public void TestAddUser()
{
    ClickButton("AddButton");
}

[Fact]
public void TestDeleteUser()
{
    ClickButton("DeleteButton");
}
```

### 2. 使用辅助类

```csharp
public class ElementHelper
{
    private readonly Window _window;
    
    public ElementHelper(Window window)
    {
        _window = window;
    }
    
    public TextBox GetTextBox(string automationId)
    {
        return _window.FindFirstDescendant(cf => 
            cf.ByAutomationId(automationId))?.AsTextBox();
    }
    
    public Button GetButton(string automationId)
    {
        return _window.FindFirstDescendant(cf => 
            cf.ByAutomationId(automationId))?.AsButton();
    }
    
    public void ClickButton(string automationId)
    {
        GetButton(automationId)?.Invoke();
        System.Threading.Thread.Sleep(300);
    }
    
    public void SetTextBoxValue(string automationId, string value)
    {
        var textBox = GetTextBox(automationId);
        if (textBox != null)
        {
            textBox.Text = value;
        }
    }
}

// 使用
var helper = new ElementHelper(window);
helper.SetTextBoxValue("UsernameTextBox", "admin");
helper.ClickButton("LoginButton");
```

## 异常处理

### 1. 优雅的错误处理

```csharp
// ❌ 不处理异常
public void ClickButton(string automationId)
{
    var button = _window.FindFirstDescendant(cf => 
        cf.ByAutomationId(automationId)).AsButton();
    button.Invoke();  // 如果 button 为 null 会抛出异常
}

// ✅ 处理异常
public void ClickButton(string automationId)
{
    try
    {
        var button = _window.FindFirstDescendant(cf => 
            cf.ByAutomationId(automationId))?.AsButton();
        
        if (button == null)
        {
            throw new ElementNotFoundException($"按钮未找到: {automationId}");
        }
        
        if (!button.IsEnabled)
        {
            throw new ElementNotInteractableException($"按钮不可用: {automationId}");
        }
        
        button.Invoke();
    }
    catch (Exception ex)
    {
        // 记录日志
        Logger.Error($"点击按钮失败: {automationId}", ex);
        
        // 截图
        ScreenshotHelper.Capture($"Error_{automationId}");
        
        throw;
    }
}

// 自定义异常
public class ElementNotFoundException : Exception
{
    public ElementNotFoundException(string message) : base(message) { }
}

public class ElementNotInteractableException : Exception
{
    public ElementNotInteractableException(string message) : base(message) { }
}
```

using Newtonsoft.Json;

public class TestDataLoader
{
    public static T LoadFromJson<T>(string filename)
    {
        string json = File.ReadAllText(filename);
        return JsonConvert.DeserializeObject<T>(json);
    }
    
    public static IEnumerable<T> LoadCollectionFromJson<T>(string filename)
    {
        string json = File.ReadAllText(filename);
        return JsonConvert.DeserializeObject<IEnumerable<T>>(json);
    }
}

// users.json
/*
[
  {
    "username": "user1",
    "password": "pass1",
    "email": "user1@test.com"
  },
  {
    "username": "user2",
    "password": "pass2",
    "email": "user2@test.com"
  }
]
*/

// 使用
public class LoginTests
{
    private static IEnumerable<UserData> _testUsers;
    
    static LoginTests()
    {
        _testUsers = TestDataLoader.LoadCollectionFromJson<UserData>("users.json");
    }

    [MemberData(nameof(GetTestUsers))]
    public void TestLogin_MultipleUsers(UserData user)

    }

        foreach (var user in _testUsers)

        }
    }

## 等待策略

### 使用显式等待而非固定等待

```csharp
// ❌ 固定等待
public void ClickButton(string automationId)
{
    var button = FindButton(automationId);
    button.Invoke();
    Thread.Sleep(5000);  // 总是等待 5 秒
}

// ✅ 显式等待
public void ClickButton(string automationId)
{
    var button = FindButton(automationId);
    button.Invoke();
    
    // 等待特定条件
    WaitForElement("ResultLabel", timeout: 5000);
}

public AutomationElement WaitForElement(string automationId, int timeout = 5000)
{
    var startTime = DateTime.Now;
    
    while ((DateTime.Now - startTime).TotalMilliseconds < timeout)
    {
        var element = _window.FindFirstDescendant(cf => 
            cf.ByAutomationId(automationId));
        
        if (element != null)
        {
            return element;
        }
        
        Thread.Sleep(500);
    }
    
    throw new TimeoutException($"元素在 {timeout}ms 内未找到: {automationId}");
}
```

### BasePage.cs

```csharp
public abstract class BasePage
{
    protected readonly Window Window;
    protected readonly WaitHelper Wait;
    protected readonly LogHelper Log;
    
    protected BasePage(Window window)
    {
        Window = window;
        Wait = new WaitHelper(window);
        Log = new LogHelper();
    }
    
    protected AutomationElement FindElement(string automationId, int timeout = 5000)
    {
        return Wait.ForElement(cf => cf.ByAutomationId(automationId), timeout);
    }
    
    protected void ClickButton(string automationId)
    {
        Log.Info($"点击按钮: {automationId}");
        var button = FindElement(automationId).AsButton();
        button.Invoke();
        Thread.Sleep(300);
    }
    
    protected void SetTextBox(string automationId, string text)
    {
        Log.Info($"设置文本框 {automationId} = {text}");
        var textBox = FindElement(automationId).AsTextBox();
        textBox.Text = text;
    }
    
    protected string GetTextBox(string automationId)
    {
        var textBox = FindElement(automationId).AsTextBox();
        return textBox.Text;
    }
}
```

### BaseTest.cs

```csharp
public abstract class BaseTest : IDisposable
{
    protected UIA3Automation Automation { get; private set; }
    protected Application App { get; private set; }
    protected Window MainWindow { get; private set; }
    protected TestConfiguration Config { get; private set; }
    protected ScreenshotHelper Screenshot { get; private set; }
    
    protected BaseTest()
    {
        Config = new TestConfiguration();
        InitializeTest();
    }
    
    private void InitializeTest()
    {
        Automation = new UIA3Automation();
        App = Application.Launch(Config.AppPath);
        Thread.Sleep(1000);
        MainWindow = App.GetMainWindow(Automation);
        
        if (Config.ScreenshotsEnabled)
        {
            Screenshot = new ScreenshotHelper(Config.ScreenshotDirectory);
        }
    }
    
    protected void TakeScreenshot(string name)
    {
        Screenshot?.Capture(MainWindow, name);
    }
    
    public virtual void Dispose()
    {
        App?.Close();
        Automation?.Dispose();
    }
}
```
