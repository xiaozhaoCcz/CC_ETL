## 元素查找

### 基本查找

```csharp
// 按 AutomationId 查找
var element = window.FindFirstDescendant(cf => cf.ByAutomationId("btnSave"));

// 按 Name 查找
var element = window.FindFirstDescendant(cf => cf.ByName("保存"));

// 按 ControlType 查找
var element = window.FindFirstDescendant(cf => cf.ByControlType(ControlType.Button));

// 组合条件
var element = window.FindFirstDescendant(cf => 
    cf.ByName("保存").And(cf.ByControlType(ControlType.Button)));
```

### 等待元素出现

```csharp
public static AutomationElement WaitForElement(
    AutomationElement parent,
    Func<ConditionFactory, ConditionBase> condition,
    int timeoutMs = 5000)
{
    var startTime = DateTime.Now;
    
    while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
    {
        var element = parent.FindFirstDescendant(condition);
        if (element != null)
        {
            return element;
        }
        Thread.Sleep(500);
    }
    
    throw new TimeoutException($"元素在 {timeoutMs}ms 内未找到");
}

// 使用
var button = WaitForElement(window, cf => cf.ByName("保存"), 10000);
```

### 查找所有匹配元素

## 等待和条件

### 等待元素可点击

```csharp
public static AutomationElement WaitForElementClickable(
    AutomationElement parent,
    Func<ConditionFactory, ConditionBase> condition,
    int timeoutMs = 5000)
{
    var startTime = DateTime.Now;
    
    while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
    {
        var element = parent.FindFirstDescendant(condition);
        if (element != null && element.IsEnabled && !element.IsOffscreen)
        {
            return element;
        }
        Thread.Sleep(500);
    }
    
    throw new TimeoutException("元素未变为可点击状态");
}
```

### 等待元素消失

```csharp
public static bool WaitForElementToDisappear(
    AutomationElement parent,
    Func<ConditionFactory, ConditionBase> condition,
    int timeoutMs = 5000)
{
    var startTime = DateTime.Now;
    
    while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
    {
        try
        {
            var element = parent.FindFirstDescendant(condition);
            if (element == null)
            {
                return true;
            }
        }
        catch
        {
            return true;
        }
        
        Thread.Sleep(500);
    }
    
    return false;
}
```

### 等待条件满足

```csharp
public static bool WaitUntil(Func<bool> condition, int timeoutMs = 5000)
{
    var startTime = DateTime.Now;
    
    while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
    {
        if (condition())
        {
            return true;
        }
        Thread.Sleep(500);
    }
    
    return false;
}

// 使用
WaitUntil(() => window.IsEnabled, 10000);
```

## 辅助方法

### 重试机制

```csharp
public static T Retry<T>(Func<T> action, int maxAttempts = 3, int delayMs = 1000)
{
    Exception lastException = null;
    
    for (int i = 0; i < maxAttempts; i++)
    {
        try
        {
            return action();
        }
        catch (Exception ex)
        {
            lastException = ex;
            Console.WriteLine($"尝试 {i + 1}/{maxAttempts} 失败: {ex.Message}");
            
            if (i < maxAttempts - 1)
            {
                Thread.Sleep(delayMs);
            }
        }
    }
    
    throw new Exception($"操作在 {maxAttempts} 次尝试后仍失败", lastException);
}

// 使用
var element = Retry(() => 
    window.FindFirstDescendant(cf => cf.ByName("按钮")) 
    ?? throw new Exception("元素未找到"),
    maxAttempts: 3
);
```

### 打印元素树

```csharp
public static void PrintElementTree(AutomationElement element, int maxDepth = 5)
{
    PrintElementRecursive(element, 0, maxDepth);
}

private static void PrintElementRecursive(AutomationElement element, int depth, int maxDepth)
{
    if (depth > maxDepth) return;
    
    string indent = new string(' ', depth * 2);
    
    try
    {
        Console.WriteLine($"{indent}[{element.ControlType}] " +
                         $"Name: \"{element.Name}\", " +
                         $"ID: \"{element.AutomationId}\"");
        
        var children = element.FindAllChildren();
        foreach (var child in children)
        {
            PrintElementRecursive(child, depth + 1, maxDepth);
        }
    }
    catch (Exception ex)
    {
        Console.WriteLine($"{indent}[ERROR] {ex.Message}");
    }
}
```

## 页面对象模式模板

### 基础页面类

```csharp
public abstract class BasePage
{
    protected readonly Window Window;
    
    protected BasePage(Window window)
    {
        Window = window;
    }
    
    protected AutomationElement FindElement(string automationId, int timeoutMs = 5000)
    {
        return WaitForElement(Window, cf => cf.ByAutomationId(automationId), timeoutMs);
    }
    
    protected void ClickButton(string automationId)
    {
        var button = FindElement(automationId).AsButton();
        button.Invoke();
        Thread.Sleep(300);
    }
    
    protected void SetTextBox(string automationId, string text)
    {
        var textBox = FindElement(automationId).AsTextBox();
        textBox.Text = text;
    }
    
    protected string GetTextBox(string automationId)
    {
        var textBox = FindElement(automationId).AsTextBox();
        return textBox.Text;
    }
    
    private AutomationElement WaitForElement(
        AutomationElement parent,
        Func<ConditionFactory, ConditionBase> condition,
        int timeoutMs)
    {
        var startTime = DateTime.Now;
        
        while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
        {
            var element = parent.FindFirstDescendant(condition);
            if (element != null)
            {
                return element;
            }
            Thread.Sleep(500);
        }
        
        throw new TimeoutException("元素未找到");
    }
}
```

### 具体页面类

```csharp
public class LoginPage : BasePage
{
    public LoginPage(Window window) : base(window) { }
    
    public void SetUsername(string username)
    {
        SetTextBox("txtUsername", username);
    }
    
    public void SetPassword(string password)
    {
        SetTextBox("txtPassword", password);
    }
    
    public void ClickLogin()
    {
        ClickButton("btnLogin");
    }
    
    public void Login(string username, string password)
    {
        SetUsername(username);
        SetPassword(password);
        ClickLogin();
    }
}
```
