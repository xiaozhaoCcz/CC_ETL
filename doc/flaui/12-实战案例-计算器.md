## 页面对象模式

### CalculatorPage.cs

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

namespace CalculatorTests.PageObjects
{
    public class CalculatorPage : IDisposable
    {
        private readonly UIA3Automation _automation;
        private readonly Application _app;
        private readonly Window _window;
        
        public CalculatorPage()
        {
            _automation = new UIA3Automation();
            _app = Application.Launch("calc.exe");
            System.Threading.Thread.Sleep(1500);
            _window = _app.GetMainWindow(_automation);
        }
        
        // 数字按钮（中文系统）
        private readonly string[] _numbersCN = { "零", "一", "二", "三", "四", "五", "六", "七", "八", "九" };
        
        // 数字按钮（英文系统）
        private readonly string[] _numbersEN = { "Zero", "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine" };
        
        /// <summary>
        /// 点击数字按钮
        /// </summary>
        public void ClickNumber(int number)
        {
            if (number < 0 || number > 9)
            {
                throw new ArgumentException("数字必须在 0-9 之间");
            }
            
            var buttonCN = FindButton(_numbersCN[number]);
            var buttonEN = FindButton(_numbersEN[number]);
            
            var button = buttonCN ?? buttonEN;
            if (button == null)
            {
                throw new Exception($"未找到数字按钮: {number}");
            }
            
            button.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 点击加号
        /// </summary>
        public void ClickPlus()
        {
            var button = FindButton("加") ?? FindButton("Plus");
            button?.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 点击减号
        /// </summary>
        public void ClickMinus()
        {
            var button = FindButton("减") ?? FindButton("Minus");
            button?.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 点击乘号
        /// </summary>
        public void ClickMultiply()
        {
            var button = FindButton("乘") ?? FindButton("Multiply by");
            button?.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 点击除号
        /// </summary>
        public void ClickDivide()
        {
            var button = FindButton("除") ?? FindButton("Divide by");
            button?.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 点击等号
        /// </summary>
        public void ClickEquals()
        {
            var button = FindButton("等于") ?? FindButton("Equals");
            button?.Invoke();
            System.Threading.Thread.Sleep(500);
        }
        
        /// <summary>
        /// 点击清除
        /// </summary>
        public void ClickClear()
        {
            var button = FindButton("清除") ?? FindButton("Clear");
            button?.Invoke();
            System.Threading.Thread.Sleep(300);
        }
        
        /// <summary>
        /// 获取计算结果
        /// </summary>
        public string GetResult()
        {
            var resultElement = _window.FindFirstDescendant(cf => 
                cf.ByAutomationId("CalculatorResults"));
            
            if (resultElement == null)
            {
                throw new Exception("未找到结果元素");
            }
            
            // 结果文本格式：" 显示为 5 "
            string text = resultElement.Name;
            
            // 提取数字
            var parts = text.Split(new[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
            return parts.Length > 0 ? parts[parts.Length - 1] : "";
        }
        
        /// <summary>
        /// 输入整数
        /// </summary>
        public void InputNumber(int number)
        {
            string numStr = Math.Abs(number).ToString();
            
            foreach (char digit in numStr)
            {
                int d = digit - '0';
                ClickNumber(d);
            }
        }
        
        /// <summary>
        /// 执行加法
        /// </summary>
        public string Add(int a, int b)
        {
            ClickClear();
            InputNumber(a);
            ClickPlus();
            InputNumber(b);
            ClickEquals();
            return GetResult();
        }
        
        /// <summary>
        /// 执行减法
        /// </summary>
        public string Subtract(int a, int b)
        {
            ClickClear();
            InputNumber(a);
            ClickMinus();
            InputNumber(b);
            ClickEquals();
            return GetResult();
        }
        
        /// <summary>
        /// 执行乘法
        /// </summary>
        public string Multiply(int a, int b)
        {
            ClickClear();
            InputNumber(a);
            ClickMultiply();
            InputNumber(b);
            ClickEquals();
            return GetResult();
        }
        
        /// <summary>
        /// 执行除法
        /// </summary>
        public string Divide(int a, int b)
        {
            ClickClear();
            InputNumber(a);
            ClickDivide();
            InputNumber(b);
            ClickEquals();
            return GetResult();
        }
        
        private Button FindButton(string buttonName)
        {
            return _window.FindFirstDescendant(cf => 
                cf.ByName(buttonName).And(cf.ByControlType(ControlType.Button)))?.AsButton();
        }
        
        public void Dispose()
        {
            _app?.Close();
            _automation?.Dispose();
        }
    }
}
```

```csharp
using Xunit;
using CalculatorTests.PageObjects;

namespace CalculatorTests.Tests
{
    public class BasicOperationsTests
    {

        public void TestAddition_TwoPositiveNumbers_ReturnsCorrectSum()

            using var calculator = new CalculatorPage();
            
            // Act
            string result = calculator.Add(5, 3);
            
            // Assert
            Assert.Equal("8", result);
        }
        
        [Fact]
        public void TestSubtraction_TwoPositiveNumbers_ReturnsCorrectDifference()
        {
            using var calculator = new CalculatorPage();
            
            // Act
            string result = calculator.Subtract(10, 4);
            
            // Assert
            Assert.Equal("6", result);
        }
        
        [Fact]
        public void TestMultiplication_TwoPositiveNumbers_ReturnsCorrectProduct()
        {
            using var calculator = new CalculatorPage();
            
            // Act
            string result = calculator.Multiply(6, 7);
            
            // Assert
            Assert.Equal("42", result);
        }
        
        [Fact]
        public void TestDivision_TwoPositiveNumbers_ReturnsCorrectQuotient()
        {
            using var calculator = new CalculatorPage();
            
            // Act
            string result = calculator.Divide(20, 4);
            
            // Assert
            Assert.Equal("5", result);
        }
        
        [Theory]
        [InlineData(0, 5, "5")]
        [InlineData(10, 0, "10")]
        [InlineData(100, 50, "150")]
        public void TestAddition_VariousInputs_ReturnsCorrectResults(int a, int b, string expected)
        {
            using var calculator = new CalculatorPage();
            
            // Act
            string result = calculator.Add(a, b);
            
            // Assert
            Assert.Equal(expected, result);
        }
        
        [Fact]
        public void TestClearButton_AfterCalculation_ResetsCalculator()
        {
            using var calculator = new CalculatorPage();
            
            // Arrange
            calculator.Add(5, 3);
            
            // Act
            calculator.ClickClear();
            calculator.InputNumber(2);
            calculator.ClickEquals();
            
            // Assert
            string result = calculator.GetResult();
            Assert.Equal("2", result);
        }
    }
}
```

### AdvancedOperationsTests.cs

```csharp
using Xunit;
using CalculatorTests.PageObjects;

namespace CalculatorTests.Tests
{
    public class AdvancedOperationsTests
    {
        [Fact]
        public void TestChainedOperations_MultipleAdditions_ReturnsCorrectSum()
        {
            using var calculator = new CalculatorPage();
            
            // 1 + 2 + 3 = 6
            calculator.ClickClear();
            calculator.ClickNumber(1);
            calculator.ClickPlus();
            calculator.ClickNumber(2);
            calculator.ClickPlus();
            calculator.ClickNumber(3);
            calculator.ClickEquals();
            
            string result = calculator.GetResult();
            Assert.Equal("6", result);
        }
        
        [Fact]
        public void TestMixedOperations_AddThenMultiply_ReturnsCorrectResult()
        {
            using var calculator = new CalculatorPage();
            
            // (5 + 3) = 8, then 8 * 2 = 16
            calculator.Add(5, 3);
            calculator.ClickMultiply();
            calculator.InputNumber(2);
            calculator.ClickEquals();
            
            string result = calculator.GetResult();
            Assert.Equal("16", result);
        }
        
        [Fact]
        public void TestLargeNumbers_Multiplication_HandlesCorrectly()
        {
            using var calculator = new CalculatorPage();
            
            // 999 * 999
            string result = calculator.Multiply(999, 999);
            
            Assert.Equal("998,001", result);
        }
        
        [Fact]
        public void TestDivisionByZero_ShowsError()
        {
            using var calculator = new CalculatorPage();
            
            // 5 / 0
            calculator.ClickClear();
            calculator.InputNumber(5);
            calculator.ClickDivide();
            calculator.InputNumber(0);
            calculator.ClickEquals();
            
            string result = calculator.GetResult();
            
            // 计算器会显示错误或特殊提示
            Assert.True(result.Contains("无法") || result.Contains("Cannot") || result == "");
        }
    }
}
```

## 增强版实现

### 带截图和日志的测试类

```csharp
using System;
using System.IO;
using Xunit;
using Xunit.Abstractions;
using FlaUI.Core.Capturing;
using CalculatorTests.PageObjects;

namespace CalculatorTests.Tests
{
    public class CalculatorTestsWithLogging : IDisposable
    {
        private readonly ITestOutputHelper _output;
        private readonly CalculatorPage _calculator;
        private readonly string _screenshotDir;
        private int _stepCounter = 0;
        
        public CalculatorTestsWithLogging(ITestOutputHelper output)
        {
            _output = output;
            _calculator = new CalculatorPage();
            
            _screenshotDir = Path.Combine(
                AppDomain.CurrentDomain.BaseDirectory,
                "Screenshots",
                DateTime.Now.ToString("yyyyMMdd_HHmmss")
            );
            Directory.CreateDirectory(_screenshotDir);
            
            Log("测试初始化完成");
        }
        
        [Fact]
        public void TestAdditionWithScreenshots()
        {
            try
            {
                Step("清除计算器", () => _calculator.ClickClear());
                
                Step("输入数字 7", () => _calculator.InputNumber(7));
                
                Step("点击加号", () => _calculator.ClickPlus());
                
                Step("输入数字 5", () => _calculator.InputNumber(5));
                
                Step("点击等号", () => _calculator.ClickEquals());
                
                Step("验证结果", () =>
                {
                    string result = _calculator.GetResult();
                    Log($"计算结果: {result}");
                    Assert.Equal("12", result);
                });
                
                Log("✓ 测试通过");
            }
            catch (Exception ex)
            {
                Log($"✗ 测试失败: {ex.Message}");
                TakeScreenshot("TestFailed");
                throw;
            }
        }
        
        private void Step(string description, Action action)
        {
            _stepCounter++;
            Log($"步骤 {_stepCounter}: {description}");
            
            try
            {
                action();
                TakeScreenshot($"Step{_stepCounter}_{description}");
            }
            catch (Exception ex)
            {
                Log($"步骤失败: {ex.Message}");
                TakeScreenshot($"Step{_stepCounter}_Error");
                throw;
            }
        }
        
        private void Log(string message)
        {
            _output.WriteLine($"[{DateTime.Now:HH:mm:ss.fff}] {message}");
        }
        
        private void TakeScreenshot(string name)
        {
            try
            {
                // 清理文件名中的非法字符
                name = string.Join("_", name.Split(Path.GetInvalidFileNameChars()));
                
                var filename = Path.Combine(_screenshotDir, $"{name}.png");
                var capture = Capture.Screen();
                capture.ToFile(filename);
                
                Log($"截图已保存: {filename}");
            }
            catch (Exception ex)
            {
                Log($"截图失败: {ex.Message}");
            }
        }
        
        public void Dispose()
        {
            Log("测试清理");
            _calculator?.Dispose();
        }
    }
}
```

## 运行测试

### 命令行运行

```bash
# 运行所有测试
dotnet test

# 运行特定测试类
dotnet test --filter "FullyQualifiedName~BasicOperationsTests"

# 运行特定测试方法
dotnet test --filter "FullyQualifiedName~TestAddition"

# 生成测试报告
dotnet test --logger "trx;LogFileName=TestResults.xml"

# 详细输出
dotnet test --logger "console;verbosity=detailed"
```

### Visual Studio 中运行

1. 打开测试资源管理器（Test Explorer）
2. 点击"运行全部"或选择特定测试
3. 查看测试结果和输出
