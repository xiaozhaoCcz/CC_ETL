# 07 - 鼠标和键盘操作

本章介绍如何使用 FlaUI 模拟鼠标和键盘操作，实现更灵活的自动化控制。

---

## 鼠标操作

### 1. 基本鼠标操作

FlaUI 提供了 `Mouse` 类来模拟鼠标操作。

```csharp
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;

// 移动鼠标到指定坐标
Mouse.MoveTo(500, 300);

// 单击左键
Mouse.Click();

// 单击右键
Mouse.RightClick();

// 双击
Mouse.DoubleClick();

// 按下鼠标按钮
Mouse.Down(MouseButton.Left);

// 释放鼠标按钮
Mouse.Up(MouseButton.Left);

// 拖拽操作
Mouse.Drag(TimeSpan.FromMilliseconds(500), 
    new Point(100, 100), 
    new Point(300, 300));
```

### 2. 点击元素

```csharp
using FlaUI.Core;
using FlaUI.Core.Input;
using FlaUI.UIA3;

using var automation = new UIA3Automation();
var app = Application.Launch("calc.exe");
var window = app.GetMainWindow(automation);

// 方式1：使用元素的 Click 方法
var button = window.FindFirstDescendant(cf => cf.ByName("一"))?.AsButton();
button?.Click();

// 方式2：通过元素位置点击
var bounds = button.BoundingRectangle;
var centerX = bounds.X + bounds.Width / 2;
var centerY = bounds.Y + bounds.Height / 2;
Mouse.MoveTo((int)centerX, (int)centerY);
Mouse.Click();

// 方式3：直接在元素上点击
button?.Click(moveMouse: true);  // 移动鼠标后点击
```

### 3. 右键点击

```csharp
// 右键点击元素
var element = window.FindFirstDescendant(cf => cf.ByName("文件"));

// 移动到元素并右键点击
var bounds = element.BoundingRectangle;
Mouse.MoveTo((int)(bounds.X + bounds.Width / 2), (int)(bounds.Y + bounds.Height / 2));
Mouse.RightClick();

System.Threading.Thread.Sleep(500);

// 选择上下文菜单项
var menuItem = window.FindFirstDescendant(cf => cf.ByName("新建"));
menuItem?.Click();
```

### 4. 拖拽操作

```csharp
using FlaUI.Core.Input;
using System;

// 拖拽元素从一个位置到另一个位置
var sourceElement = window.FindFirstDescendant(cf => cf.ByName("源元素"));
var targetElement = window.FindFirstDescendant(cf => cf.ByName("目标元素"));

// 获取源和目标的中心点
var sourceBounds = sourceElement.BoundingRectangle;
var targetBounds = targetElement.BoundingRectangle;

var sourceCenter = new Point(
    (int)(sourceBounds.X + sourceBounds.Width / 2),
    (int)(sourceBounds.Y + sourceBounds.Height / 2)
);

var targetCenter = new Point(
    (int)(targetBounds.X + targetBounds.Width / 2),
    (int)(targetBounds.Y + targetBounds.Height / 2)
);

// 执行拖拽
Mouse.Drag(TimeSpan.FromMilliseconds(500), sourceCenter, targetCenter);
```

### 完整示例：鼠标操作

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.Core.Input;
using FlaUI.UIA3;

class MouseExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("mspaint.exe");  // 打开画图
        System.Threading.Thread.Sleep(2000);
        
        var window = app.GetMainWindow(automation);
        window.SetForeground();
        
        // 查找画布区域
        var canvas = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.Pane).And(cf.ByName("Canvas")));
        
        if (canvas != null)
        {
            var bounds = canvas.BoundingRectangle;
            
            // 在画布上绘制
            int startX = (int)(bounds.X + 100);
            int startY = (int)(bounds.Y + 100);
            
            // 移动到起点
            Mouse.MoveTo(startX, startY);
            
            // 绘制矩形
            Mouse.Down(MouseButton.Left);
            Mouse.MoveTo(startX + 200, startY);
            System.Threading.Thread.Sleep(100);
            Mouse.MoveTo(startX + 200, startY + 150);
            System.Threading.Thread.Sleep(100);
            Mouse.MoveTo(startX, startY + 150);
            System.Threading.Thread.Sleep(100);
            Mouse.MoveTo(startX, startY);
            Mouse.Up(MouseButton.Left);
            
            Console.WriteLine("绘制完成");
        }
        
        System.Threading.Thread.Sleep(3000);
        app.Close();
        
        // 处理保存对话框
        System.Threading.Thread.Sleep(500);
        var saveDialog = automation.GetDesktop()
            .FindFirstChild(cf => cf.ByControlType(ControlType.Window));
        var noButton = saveDialog?.FindFirstDescendant(cf => 
            cf.ByName("不保存"))?.AsButton();
        noButton?.Invoke();
    }
}
```

## 键盘操作

### 1. 基本键盘操作

FlaUI 提供了 `Keyboard` 类来模拟键盘输入。

```csharp
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;

// 输入文本
Keyboard.Type("Hello World");

// 输入单个字符
Keyboard.Type("A");

// 按下按键
Keyboard.Press(VirtualKeyShort.ENTER);

// 按下组合键
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_C);  // Ctrl+C

// 按下按键（不释放）
Keyboard.Down(VirtualKeyShort.SHIFT);

// 释放按键
Keyboard.Up(VirtualKeyShort.SHIFT);
```

### 2. 输入文本

```csharp
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.Core.Input;
using FlaUI.UIA3;

using var automation = new UIA3Automation();
var app = Application.Launch("notepad.exe");
System.Threading.Thread.Sleep(1000);

var window = app.GetMainWindow(automation);
window.SetForeground();

// 方式1：直接使用键盘输入
Keyboard.Type("Hello World!");

// 方式2：使用文本框的 Enter 方法
var textBox = window.FindFirstDescendant(cf => 
    cf.ByControlType(ControlType.Edit))?.AsTextBox();
textBox?.Enter("使用 Enter 方法输入");

// 方式3：设置 Text 属性
textBox.Text = "使用 Text 属性设置";
```

### 3. 特殊按键


### 4. 组合键

```csharp
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;

// Ctrl+C (复制)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_C);

// Ctrl+V (粘贴)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_V);

// Ctrl+X (剪切)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_X);

// Ctrl+A (全选)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_A);

// Ctrl+Z (撤销)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_Z);

// Ctrl+S (保存)
Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_S);

// Alt+F4 (关闭窗口)
Keyboard.TypeSimultaneously(VirtualKeyShort.ALT, VirtualKeyShort.F4);

// Shift+Tab
Keyboard.TypeSimultaneously(VirtualKeyShort.SHIFT, VirtualKeyShort.TAB);

// Ctrl+Shift+N (多个修饰键)
Keyboard.Down(VirtualKeyShort.CONTROL);
Keyboard.Down(VirtualKeyShort.SHIFT);
Keyboard.Press(VirtualKeyShort.KEY_N);
Keyboard.Up(VirtualKeyShort.SHIFT);
Keyboard.Up(VirtualKeyShort.CONTROL);
```

### 完整示例：键盘操作

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;
using FlaUI.UIA3;

class KeyboardExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        window.SetForeground();
        
        Console.WriteLine("输入文本...");
        Keyboard.Type("第一行文本");
        System.Threading.Thread.Sleep(300);
        
        // 回车换行
        Keyboard.Press(VirtualKeyShort.ENTER);
        System.Threading.Thread.Sleep(300);
        
        Keyboard.Type("第二行文本");
        System.Threading.Thread.Sleep(300);
        
        // 全选
        Console.WriteLine("执行全选 (Ctrl+A)...");
        Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_A);
        System.Threading.Thread.Sleep(300);
        
        // 复制
        Console.WriteLine("执行复制 (Ctrl+C)...");
        Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_C);
        System.Threading.Thread.Sleep(300);
        
        // 移动到末尾
        Keyboard.Press(VirtualKeyShort.END);
        System.Threading.Thread.Sleep(300);
        
        // 回车
        Keyboard.Press(VirtualKeyShort.ENTER);
        
        // 粘贴
        Console.WriteLine("执行粘贴 (Ctrl+V)...");
        Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.KEY_V);
        System.Threading.Thread.Sleep(1000);
        
        // 不保存关闭
        Console.WriteLine("关闭窗口 (Alt+F4)...");
        Keyboard.TypeSimultaneously(VirtualKeyShort.ALT, VirtualKeyShort.F4);
        System.Threading.Thread.Sleep(500);
        
        // 按 N 键（不保存）
        Keyboard.Press(VirtualKeyShort.KEY_N);
        
        Console.WriteLine("操作完成");
    }
}
```

## 混合使用鼠标和键盘

### 示例 1：选择文本

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;
using FlaUI.UIA3;

class SelectTextExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        window.SetForeground();
        
        // 输入多行文本
        Keyboard.Type("第一行文本\n第二行文本\n第三行文本");
        System.Threading.Thread.Sleep(500);
        
        // 移动到开头
        Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, VirtualKeyShort.HOME);
        System.Threading.Thread.Sleep(300);
        
        // 使用 Shift+方向键选择文本
        Console.WriteLine("选择第一行...");
        Keyboard.Down(VirtualKeyShort.SHIFT);
        Keyboard.Press(VirtualKeyShort.END);
        Keyboard.Up(VirtualKeyShort.SHIFT);
        System.Threading.Thread.Sleep(1000);
        
        // 使用 Shift+Ctrl+方向键选择单词
        Keyboard.Press(VirtualKeyShort.HOME);  // 取消选择
        Keyboard.TypeSimultaneously(VirtualKeyShort.SHIFT, VirtualKeyShort.CONTROL, VirtualKeyShort.RIGHT);
        System.Threading.Thread.Sleep(1000);
        
        app.Close();
        System.Threading.Thread.Sleep(500);
        Keyboard.Press(VirtualKeyShort.KEY_N);  // 不保存
    }
}
```

### 示例 2：右键菜单操作

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.Core.Input;

class ContextMenuExample
```

## 辅助类封装

### 键盘辅助类

```csharp
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;
using System.Threading;

public static class KeyboardHelper
{
    private static int _defaultDelay = 100;
    
    /// <summary>
    /// 设置默认延迟时间（毫秒）
    /// </summary>
    public static void SetDefaultDelay(int milliseconds)
    {
        _defaultDelay = milliseconds;
    }
    
    /// <summary>
    /// 输入文本（带延迟）
    /// </summary>
    public static void TypeWithDelay(string text, int delay = -1)
    {
        Keyboard.Type(text);
        Thread.Sleep(delay < 0 ? _defaultDelay : delay);
    }
    
    /// <summary>
    /// 按下按键（带延迟）
    /// </summary>
    public static void PressWithDelay(VirtualKeyShort key, int delay = -1)
    {
        Keyboard.Press(key);
        Thread.Sleep(delay < 0 ? _defaultDelay : delay);
    }
    
    /// <summary>
    /// Ctrl+字母组合键
    /// </summary>
    public static void CtrlKey(char key, int delay = -1)
    {
        var vk = (VirtualKeyShort)Enum.Parse(typeof(VirtualKeyShort), $"KEY_{key}");
        Keyboard.TypeSimultaneously(VirtualKeyShort.CONTROL, vk);
        Thread.Sleep(delay < 0 ? _defaultDelay : delay);
    }
    
    /// <summary>
    /// 常用快捷键
    /// </summary>
    public static class Shortcuts
    {
        public static void Copy() => KeyboardHelper.CtrlKey('C');
        public static void Paste() => KeyboardHelper.CtrlKey('V');
        public static void Cut() => KeyboardHelper.CtrlKey('X');
        public static void SelectAll() => KeyboardHelper.CtrlKey('A');
        public static void Undo() => KeyboardHelper.CtrlKey('Z');
        public static void Redo() => KeyboardHelper.CtrlKey('Y');
        public static void Save() => KeyboardHelper.CtrlKey('S');
        public static void Find() => KeyboardHelper.CtrlKey('F');
        
        public static void Enter() => KeyboardHelper.PressWithDelay(VirtualKeyShort.ENTER);
        public static void Escape() => KeyboardHelper.PressWithDelay(VirtualKeyShort.ESCAPE);
        public static void Tab() => KeyboardHelper.PressWithDelay(VirtualKeyShort.TAB);
        
        public static void CloseWindow()
        {
            Keyboard.TypeSimultaneously(VirtualKeyShort.ALT, VirtualKeyShort.F4);
            Thread.Sleep(_defaultDelay);
        }
    }
}

// 使用示例
class KeyboardHelperExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        window.SetForeground();
        
        // 使用辅助类
        KeyboardHelper.SetDefaultDelay(300);
        KeyboardHelper.TypeWithDelay("测试文本");
        KeyboardHelper.Shortcuts.SelectAll();
        KeyboardHelper.Shortcuts.Copy();
        KeyboardHelper.Shortcuts.Enter();
        KeyboardHelper.Shortcuts.Paste();
        
        System.Threading.Thread.Sleep(2000);
        KeyboardHelper.Shortcuts.CloseWindow();
        System.Threading.Thread.Sleep(500);
        KeyboardHelper.PressWithDelay(VirtualKeyShort.KEY_N);
    }
}
```

### 鼠标辅助类

```csharp
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Input;
using FlaUI.Core.WindowsAPI;
using System;
using System.Threading;

public static class MouseHelper
{
    private static int _defaultDelay = 100;
    
    public static void SetDefaultDelay(int milliseconds)
    {
        _defaultDelay = milliseconds;
    }
    
    /// <summary>
    /// 移动鼠标到元素中心
    /// </summary>
    public static void MoveToElement(AutomationElement element)
    {
        var bounds = element.BoundingRectangle;
        var centerX = (int)(bounds.X + bounds.Width / 2);
        var centerY = (int)(bounds.Y + bounds.Height / 2);
        Mouse.MoveTo(centerX, centerY);
        Thread.Sleep(_defaultDelay);
    }
    
    /// <summary>
    /// 点击元素中心
    /// </summary>
    public static void ClickElement(AutomationElement element, bool useRightClick = false)
    {
        MoveToElement(element);
        if (useRightClick)
        {
            Mouse.RightClick();
        }
        else
        {
            Mouse.Click();
        }
        Thread.Sleep(_defaultDelay);
    }
    
    /// <summary>
    /// 双击元素
    /// </summary>
    public static void DoubleClickElement(AutomationElement element)
    {
        MoveToElement(element);
        Mouse.DoubleClick();
        Thread.Sleep(_defaultDelay);
    }
    
    /// <summary>
    /// 拖拽元素
    /// </summary>
    public static void DragElement(AutomationElement source, AutomationElement target, int durationMs = 500)
    {
        var sourceBounds = source.BoundingRectangle;
        var targetBounds = target.BoundingRectangle;
        
        var sourceCenter = new Point(
            (int)(sourceBounds.X + sourceBounds.Width / 2),
            (int)(sourceBounds.Y + sourceBounds.Height / 2)
        );
        
        var targetCenter = new Point(
            (int)(targetBounds.X + targetBounds.Width / 2),
            (int)(targetBounds.Y + targetBounds.Height / 2)
        );
        
        Mouse.Drag(TimeSpan.FromMilliseconds(durationMs), sourceCenter, targetCenter);
        Thread.Sleep(_defaultDelay);
    }
}
```
