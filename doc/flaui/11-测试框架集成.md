# 11 - 测试框架集成

本章介绍如何将 FlaUI 与流行的 .NET 测试框架（xUnit、NUnit、MSTest）集成，编写结构化的自动化测试。

---

## 为什么使用测试框架？

### 优势

1. **结构化测试**：组织测试用例
2. **断言支持**：验证测试结果
3. **测试生命周期**：Setup/Teardown 管理
4. **测试报告**：自动生成测试报告
5. **CI/CD 集成**：轻松集成到持续集成流程
6. **并行执行**：提高测试效率

---

## xUnit 集成

### 1. 安装 xUnit

```bash
dotnet add package xunit
dotnet add package xunit.runner.visualstudio
dotnet add package Microsoft.NET.Test.Sdk
```

### 2. 基础测试类

```csharp
using System;
using Xunit;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

namespace FlaUITests
{
    public class NotepadTests : IDisposable
    {
        private readonly UIA3Automation _automation;
        private Application _app;
        private Window _window;
        
        public NotepadTests()
        {
            // 测试初始化
            _automation = new UIA3Automation();
            _app = Application.Launch("notepad.exe");
            System.Threading.Thread.Sleep(1000);
            _window = _app.GetMainWindow(_automation);
        }
        
        [Fact]
        public void TestWindowTitle()
        {
            // 验证窗口标题
            Assert.Contains("记事本", _window.Title);
        }
        
        [Fact]
        public void TestInputText()
        {
            // 查找编辑框
            var editBox = _window.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit))?.AsTextBox();
            
            Assert.NotNull(editBox);
            
            // 输入文本
            editBox.Text = "Hello xUnit!";
            
            // 验证文本
            Assert.Equal("Hello xUnit!", editBox.Text);
        }
        
        [Fact]
        public void TestEditBoxIsEnabled()
        {
            var editBox = _window.FindFirstDescendant(cf => 
                cf.ByControlType(ControlType.Edit));
            
            Assert.NotNull(editBox);
            Assert.True(editBox.IsEnabled);
        }
        
        public void Dispose()
        {
            // 测试清理
            try
            {
                _app?.Close();
                System.Threading.Thread.Sleep(500);
                
                // 处理保存对话框
                var desktop = _automation.GetDesktop();
                var dialog = desktop.FindFirstChild(cf => cf.ByControlType(ControlType.Window));
                var noButton = dialog?.FindFirstDescendant(cf => 
                    cf.ByName("不保存").Or(cf.ByName("Don't Save")))?.AsButton();
                noButton?.Invoke();
            }
            catch { }
            finally
            {
                _automation?.Dispose();
            }
        }
    }
}
```

### 3. 使用 Theory 进行参数化测试

```csharp
using Xunit;

public class CalculatorTests : IDisposable
{
    private readonly UIA3Automation _automation;
    private Application _app;
    private Window _window;
    
    public CalculatorTests()
    {
        _automation = new UIA3Automation();
        _app = Application.Launch("calc.exe");
        System.Threading.Thread.Sleep(1500);
        _window = _app.GetMainWindow(_automation);
    }
    
    [Theory]
    [InlineData("一", "二", "加", "等于", "3")]  // 1 + 2 = 3
    [InlineData("五", "三", "减", "等于", "2")]  // 5 - 3 = 2
    public void TestCalculation(string num1, string num2, string op, string equals, string expectedResult)
    {
        // 点击第一个数字
        ClickButton(num1);
        
        // 点击操作符
        ClickButton(op);
        
        // 点击第二个数字
        ClickButton(num2);
        
        // 点击等号
        ClickButton(equals);
        
        System.Threading.Thread.Sleep(500);
        
        // 验证结果
        var result = _window.FindFirstDescendant(cf => cf.ByAutomationId("CalculatorResults"));
        Assert.NotNull(result);
        Assert.Contains(expectedResult, result.Name);
    }
    
    private void ClickButton(string buttonName)
    {
        var button = _window.FindFirstDescendant(cf => 
            cf.ByName(buttonName).And(cf.ByControlType(ControlType.Button)))?.AsButton();
        Assert.NotNull(button);
        button.Invoke();
        System.Threading.Thread.Sleep(300);
    }
    
    public void Dispose()
    {
        _app?.Close();
        _automation?.Dispose();
    }
}
```

### 4. 使用 Collection Fixture 共享上下文

using Xunit;

// 定义共享上下文
public class AppFixture : IDisposable
{

    {
        App?.Close();
        System.Threading.Thread.Sleep(500);

    }
}

// 定义 Collection
[CollectionDefinition("App collection")]
public class AppCollection : ICollectionFixture<AppFixture>

}

## 测试基类封装

### 通用测试基类

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.UIA3;

public abstract class TestBase : IDisposable
{
    protected UIA3Automation Automation { get; private set; }
    protected Application App { get; private set; }
    protected Window MainWindow { get; private set; }
    
    protected void LaunchApp(string appPath, int waitMs = 1000)
    {
        Automation = new UIA3Automation();
        App = Application.Launch(appPath);
        System.Threading.Thread.Sleep(waitMs);
        MainWindow = App.GetMainWindow(Automation);
    }
    
    protected AutomationElement FindElement(Func<ConditionFactory, ConditionBase> condition, int timeoutMs = 5000)
    {
        var startTime = DateTime.Now;
        
        while ((DateTime.Now - startTime).TotalMilliseconds < timeoutMs)
        {
            var element = MainWindow.FindFirstDescendant(condition);
            if (element != null)
            {
                return element;
            }
            System.Threading.Thread.Sleep(500);
        }
        
        throw new Exception($"元素在 {timeoutMs}ms 内未找到");
    }
    
    protected void CloseApp()
    {
        try
        {
            App?.Close();
            System.Threading.Thread.Sleep(500);
        }
        catch { }
    }
    
    public virtual void Dispose()
    {
        CloseApp();
        Automation?.Dispose();
    }
}

// 使用示例（xUnit）
public class NotepadTestsWithBase : TestBase
{
    public NotepadTestsWithBase()
    {
        LaunchApp("notepad.exe");
    }
    
    [Fact]
    public void TestInputText()
    {
        var editBox = FindElement(cf => cf.ByControlType(ControlType.Edit))?.AsTextBox();
        Assert.NotNull(editBox);
        
        editBox.Text = "测试文本";
        Assert.Equal("测试文本", editBox.Text);
    }
}
```
