# 05 - 常用控件操作

本章详细介绍如何使用 FlaUI 操作各种常见的 Windows 控件。

---

## 按钮 (Button)

### 基本操作

```csharp
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

// 查找按钮
var button = window.FindFirstDescendant(cf => 
    cf.ByName("确定").And(cf.ByControlType(ControlType.Button)))?.AsButton();

// 点击按钮
button?.Invoke();

// 或使用 Click 方法
button?.Click();

// 检查按钮状态
if (button != null)
{
    Console.WriteLine($"按钮是否可用: {button.IsEnabled}");
    Console.WriteLine($"按钮是否可见: {!button.IsOffscreen}");
}
```

### 完整示例

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class ButtonExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("calc.exe");
        System.Threading.Thread.Sleep(1500);
        
        var window = app.GetMainWindow(automation);
        
        // 查找并点击数字按钮
        ClickButton(window, "一", automation);  // 中文系统
        ClickButton(window, "One", automation); // 英文系统
        
        System.Threading.Thread.Sleep(2000);
        app.Close();
    }
    
    static void ClickButton(AutomationElement window, string buttonName, AutomationBase automation)
    {
        var button = window.FindFirstDescendant(cf => 
            cf.ByName(buttonName).And(cf.ByControlType(ControlType.Button)))?.AsButton();
        
        if (button != null && button.IsEnabled)
        {
            button.Invoke();
            Console.WriteLine($"已点击按钮: {buttonName}");
        }
    }
}
```

## 文本框 (TextBox)

### 基本操作

```csharp
// 查找文本框
var textBox = window.FindFirstDescendant(cf => 
    cf.ByControlType(ControlType.Edit))?.AsTextBox();

// 输入文本（清空后输入）
textBox.Text = "Hello World";

// 追加文本（不清空）
textBox.Enter("Additional Text");

// 获取文本
string text = textBox.Text;
Console.WriteLine($"文本内容: {text}");

// 检查是否只读
bool isReadOnly = textBox.IsReadOnly;
```

### 完整示例：记事本文本操作

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class TextBoxExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        
        // 查找编辑框
        var editBox = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.Edit))?.AsTextBox();
        
        if (editBox != null)
        {
            // 输入文本
            editBox.Text = "第一行文本\n";
            System.Threading.Thread.Sleep(500);
            
            // 追加文本
            editBox.Enter("第二行文本\n");
            System.Threading.Thread.Sleep(500);
            
            // 读取文本
            string content = editBox.Text;
            Console.WriteLine($"文本内容:\n{content}");
            
            // 清空文本
            editBox.Text = "";
            Console.WriteLine("文本已清空");
        }
        
        System.Threading.Thread.Sleep(2000);
        
        // 关闭时不保存
        app.Close();
        System.Threading.Thread.Sleep(500);
        
        // 处理"是否保存"对话框
        var saveDialog = window.FrameworkAutomationElement.Automation.GetDesktop()
            .FindFirstChild(cf => cf.ByControlType(ControlType.Window));
        var noButton = saveDialog?.FindFirstDescendant(cf => 
            cf.ByName("不保存").Or(cf.ByName("Don't Save")))?.AsButton();
        noButton?.Invoke();
    }
}
```

## 复选框 (CheckBox)

### 基本操作

```csharp
// 查找复选框
var checkBox = window.FindFirstDescendant(cf => 
    cf.ByName("记住密码").And(cf.ByControlType(ControlType.CheckBox)))?.AsCheckBox();

// 勾选
checkBox.IsChecked = true;

// 取消勾选
checkBox.IsChecked = false;

// 切换状态
checkBox.Toggle();

// 获取状态
bool? isChecked = checkBox.IsChecked;
if (isChecked == true)
{
    Console.WriteLine("复选框已勾选");
}
else if (isChecked == false)
{
    Console.WriteLine("复选框未勾选");
}
else
{
    Console.WriteLine("复选框状态不确定");
}
```

### 完整示例

```csharp
using System;
using FlaUI.Core.AutomationElements;

class CheckBoxExample
{
    static void Main()
    {
        // 假设已获取window
        var checkBox = window.FindFirstDescendant(cf => 
            cf.ByName("同意条款"))?.AsCheckBox();
        
        if (checkBox != null)
        {
            Console.WriteLine($"初始状态: {checkBox.IsChecked}");
            
            // 勾选
            checkBox.IsChecked = true;
            Console.WriteLine($"勾选后: {checkBox.IsChecked}");
            
            System.Threading.Thread.Sleep(1000);
            
            // 取消勾选
            checkBox.IsChecked = false;
            Console.WriteLine($"取消后: {checkBox.IsChecked}");
        }
    }
}
```

## 单选按钮 (RadioButton)

### 基本操作

```csharp
// 查找单选按钮
var radioButton = window.FindFirstDescendant(cf => 
    cf.ByName("男").And(cf.ByControlType(ControlType.RadioButton)))?.AsRadioButton();

// 选中
radioButton.IsChecked = true;

// 或使用 Select 方法
radioButton.Select();

// 获取状态
```

### 完整示例：单选按钮组

```csharp
using System;
using System.Linq;
using FlaUI.Core.Definitions;

class RadioButtonExample
{
    static void Main()
    {
        // 假设已获取window
        
        // 查找所有单选按钮
        var radioButtons = window.FindAllDescendants(cf => 
            cf.ByControlType(ControlType.RadioButton));
        
        Console.WriteLine($"找到 {radioButtons.Length} 个单选按钮");
        
        // 打印所有单选按钮
        foreach (var rb in radioButtons)
        {
            var radioButton = rb.AsRadioButton();
            Console.WriteLine($"选项: {radioButton.Name}, 选中: {radioButton.IsChecked}");
        }
        
        // 选中特定选项
        var maleOption = radioButtons.FirstOrDefault(rb => rb.Name == "男")?.AsRadioButton();
        if (maleOption != null)
        {
            maleOption.Select();
            Console.WriteLine($"已选中: {maleOption.Name}");
        }
    }
}
```

## 下拉框 (ComboBox)

### 基本操作

```csharp
// 查找下拉框
var comboBox = window.FindFirstDescendant(cf => 
    cf.ByControlType(ControlType.ComboBox))?.AsComboBox();

// 展开下拉框
comboBox.Expand();

// 选择项（按文本）
comboBox.Select("选项1");

// 选择项（按索引）
comboBox.Select(0);  // 选择第一项

// 获取当前选中项
var selectedItem = comboBox.SelectedItem;
Console.WriteLine($"当前选中: {selectedItem?.Name}");

// 获取所有选项
var items = comboBox.Items;
foreach (var item in items)
{
    Console.WriteLine($"选项: {item.Name}");
}

// 折叠下拉框
comboBox.Collapse();
```

### 完整示例

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.Definitions;
using FlaUI.UIA3;

class ComboBoxExample
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        
        // 假设应用有下拉框
        // var app = Application.Launch("YourApp.exe");
        // var window = app.GetMainWindow(automation);
        
        var comboBox = window.FindFirstDescendant(cf => 
            cf.ByAutomationId("cityComboBox"))?.AsComboBox();
        
        if (comboBox != null)
        {
            // 展开并显示所有选项
            comboBox.Expand();
            System.Threading.Thread.Sleep(500);
            
            var items = comboBox.Items;
            Console.WriteLine($"下拉框包含 {items.Length} 个选项:");
            foreach (var item in items)
            {
                Console.WriteLine($"  - {item.Name}");
            }
            
            // 选择特定选项
            comboBox.Select("北京");
            Console.WriteLine($"\n已选择: {comboBox.SelectedItem?.Name}");
            
            System.Threading.Thread.Sleep(1000);
            
            // 选择另一个选项
            comboBox.Select(2);  // 选择第3项
            Console.WriteLine($"已选择: {comboBox.SelectedItem?.Name}");
        }
    }
}
```

```csharp
using System;
using FlaUI.Core.Definitions;

class ListBoxExample
{

    {

            }
        }
    }
}
```

## 树形控件 (Tree)

### 基本操作

```csharp
// 查找树形控件
var tree = window.FindFirstDescendant(cf => 
    cf.ByControlType(ControlType.Tree))?.AsTree();

// 获取所有根节点
var items = tree.Items;

// 展开节点
var node = tree.Items[0].AsTreeItem();
node.Expand();

// 折叠节点
node.Collapse();

// 选择节点
node.Select();

// 获取子节点
var children = node.Items;
foreach (var child in children)
{
    Console.WriteLine($"子节点: {child.Name}");
}
```

### 完整示例：遍历树形结构

```csharp
using System;
using FlaUI.Core.AutomationElements;

class TreeExample
{
    static void Main()
    {
        // 假设已获取window
        var tree = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.Tree))?.AsTree();
        
        if (tree != null)
        {
            Console.WriteLine("=== 树形结构 ===");
            TraverseTree(tree.Items, 0);
        }
    }
    
    static void TraverseTree(TreeItem[] items, int depth)
    {
        string indent = new string(' ', depth * 2);
        
        foreach (var item in items)
        {
            Console.WriteLine($"{indent}- {item.Name}");
            
            // 展开节点以获取子节点
            try
            {
                item.Expand();
                System.Threading.Thread.Sleep(200);
                
                var children = item.Items;
                if (children.Length > 0)
                {
                    TraverseTree(children, depth + 1);
                }
                
                item.Collapse();
            }
            catch (Exception ex)
            {
                Console.WriteLine($"{indent}  无法展开: {ex.Message}");
            }
        }
    }
}
```

```csharp
// 查找表格
```

### 完整示例

```csharp
using System;
using FlaUI.Core.Definitions;

class DataGridExample
{
    static void Main()
    {
        // 假设已获取window
        var grid = window.FindFirstDescendant(cf => 
            cf.ByControlType(ControlType.DataGrid))?.AsDataGridView();
        
        if (grid != null)
        {
            var rows = grid.Rows;
            var columns = grid.Columns;
            
            Console.WriteLine($"表格大小: {rows.Length} 行 × {columns.Length} 列\n");
            
            // 打印表头
            Console.Write("索引\t");
            foreach (var col in columns)
            {
                Console.Write($"{col.Name}\t");
            }
            Console.WriteLine();
            
            // 打印表格内容
            for (int i = 0; i < rows.Length; i++)
            {
                Console.Write($"{i}\t");
                
                var cells = rows[i].Cells;
                foreach (var cell in cells)
                {
                    Console.Write($"{cell.Name}\t");
                }
                Console.WriteLine();
            }
        }
    }
}
```

## 封装常用操作

### 控件操作辅助类

```csharp
using System;
using FlaUI.Core;
using FlaUI.Core.AutomationElements;
using FlaUI.Core.Definitions;

public class ControlHelper
{
    private readonly Window _window;
    
    public ControlHelper(Window window)
    {
        _window = window ?? throw new ArgumentNullException(nameof(window));
    }
    
    /// <summary>
    /// 点击按钮
    /// </summary>
    public bool ClickButton(string name = null, string automationId = null)
    {
        var button = FindButton(name, automationId);
        if (button != null && button.IsEnabled)
        {
            button.Invoke();
            return true;
        }
        return false;
    }
    
    /// <summary>
    /// 输入文本
    /// </summary>
    public bool SetText(string text, string name = null, string automationId = null)
    {
        var textBox = FindTextBox(name, automationId);
        if (textBox != null)
        {
            textBox.Text = text;
            return true;
        }
        return false;
    }
    
    /// <summary>
    /// 勾选复选框
    /// </summary>
    public bool CheckCheckBox(bool check, string name = null, string automationId = null)
    {
        var checkBox = FindCheckBox(name, automationId);
        if (checkBox != null)
        {
            checkBox.IsChecked = check;
            return true;
        }
        return false;
    }
    
    /// <summary>
    /// 选择下拉框项
    /// </summary>
    public bool SelectComboBox(string itemName, string comboBoxName = null, string automationId = null)
    {
        var comboBox = FindComboBox(comboBoxName, automationId);
        if (comboBox != null)
        {
            comboBox.Select(itemName);
            return true;
        }
        return false;
    }
    
    // 私有查找方法
    private Button FindButton(string name, string automationId)
    {
        if (!string.IsNullOrEmpty(automationId))
        {
            return _window.FindFirstDescendant(cf => cf.ByAutomationId(automationId))?.AsButton();
        }
        return _window.FindFirstDescendant(cf => 
            cf.ByName(name).And(cf.ByControlType(ControlType.Button)))?.AsButton();
    }
    
    private TextBox FindTextBox(string name, string automationId)
    {
        if (!string.IsNullOrEmpty(automationId))
        {
            return _window.FindFirstDescendant(cf => cf.ByAutomationId(automationId))?.AsTextBox();
        }
        return _window.FindFirstDescendant(cf => 
            cf.ByName(name).And(cf.ByControlType(ControlType.Edit)))?.AsTextBox();
    }
    
    private CheckBox FindCheckBox(string name, string automationId)
    {
        if (!string.IsNullOrEmpty(automationId))
        {
            return _window.FindFirstDescendant(cf => cf.ByAutomationId(automationId))?.AsCheckBox();
        }
        return _window.FindFirstDescendant(cf => 
            cf.ByName(name).And(cf.ByControlType(ControlType.CheckBox)))?.AsCheckBox();
    }
    
    private ComboBox FindComboBox(string name, string automationId)
    {
        if (!string.IsNullOrEmpty(automationId))
        {
            return _window.FindFirstDescendant(cf => cf.ByAutomationId(automationId))?.AsComboBox();
        }
        return _window.FindFirstDescendant(cf => 
            cf.ByName(name).And(cf.ByControlType(ControlType.ComboBox)))?.AsComboBox();
    }
}

// 使用示例
class Program
{
    static void Main()
    {
        using var automation = new UIA3Automation();
        var app = Application.Launch("notepad.exe");
        System.Threading.Thread.Sleep(1000);
        
        var window = app.GetMainWindow(automation);
        var helper = new ControlHelper(window);
        
        // 使用封装的方法
        helper.SetText("使用辅助类输入文本", automationId: "15");
        
        Console.ReadKey();
        app.Close();
    }
}
```
