## 17.1 需求分析

## 17.2 实现代码

### 17.2.1 Calculator.java

```java
package com.example.javafx.calculator;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.TextField;
import javafx.scene.input.KeyEvent;
import javafx.scene.layout.*;
import javafx.stage.Stage;

public class Calculator extends Application {

    private TextField display;
    private String operator = "";
    private double firstOperand = 0;
    private boolean startNewNumber = true;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        root.setStyle("-fx-background-color: #2c3e50;");

        // 显示屏
        display = new TextField("0");
        display.setEditable(false);
        display.setAlignment(Pos.CENTER_RIGHT);
        display.setStyle(
            "-fx-font-size: 32px; " +
            "-fx-font-weight: bold; " +
            "-fx-background-color: #34495e; " +
            "-fx-text-fill: white; " +
            "-fx-padding: 15; " +
            "-fx-border-radius: 5; " +
            "-fx-background-radius: 5;"
        );
        display.setPrefHeight(80);
        root.setTop(display);
        BorderPane.setMargin(display, new Insets(0, 0, 10, 0));

        // 按钮面板
        GridPane buttonPanel = createButtonPanel();
        root.setCenter(buttonPanel);

        Scene scene = new Scene(root, 350, 500);
        
        // 键盘输入支持
        scene.addEventFilter(KeyEvent.KEY_TYPED, this::handleKeyPress);
        scene.setOnKeyPressed(e -> {
            switch (e.getCode()) {
                case ENTER:
                    calculate();
                    break;
                case BACK_SPACE:
                    if (!display.getText().equals("0")) {
                        String text = display.getText();
                        display.setText(text.length() > 1 ? 
                            text.substring(0, text.length() - 1) : "0");
                    }
                    break;
                case ESCAPE:
                    allClear();
                    break;
            }
        });

        primaryStage.setTitle("计算器");
        primaryStage.setScene(scene);
        primaryStage.setResizable(false);
        primaryStage.show();
    }

    private GridPane createButtonPanel() {
        GridPane grid = new GridPane();
        grid.setHgap(8);
        grid.setVgap(8);
        grid.setPadding(new Insets(10, 0, 0, 0));

        // 按钮定义
        String[][] buttons = {
            {"AC", "C", "%", "÷"},
            {"7", "8", "9", "×"},
            {"4", "5", "6", "-"},
            {"1", "2", "3", "+"},
            {"0", "0", ".", "="}
        };

        for (int row = 0; row < buttons.length; row++) {
            for (int col = 0; col < buttons[row].length; col++) {
                String buttonText = buttons[row][col];
                Button button = createButton(buttonText);
                
                // 特殊处理：0按钮占两列
                if (row == 4 && col == 0) {
                    grid.add(button, col, row, 2, 1);
                    col++; // 跳过下一列
                } else {
                    grid.add(button, col, row);
                }
            }
        }

        // 设置列约束（使按钮等宽）
        for (int i = 0; i < 4; i++) {
            ColumnConstraints colConstraints = new ColumnConstraints();
            colConstraints.setPercentWidth(25);
            colConstraints.setHgrow(Priority.ALWAYS);
            grid.getColumnConstraints().add(colConstraints);
        }

        // 设置行约束
        for (int i = 0; i < 5; i++) {
            RowConstraints rowConstraints = new RowConstraints();
            rowConstraints.setPercentHeight(20);
            rowConstraints.setVgrow(Priority.ALWAYS);
            grid.getRowConstraints().add(rowConstraints);
        }

        return grid;
    }

    private Button createButton(String text) {
        Button button = new Button(text);
        button.setMaxSize(Double.MAX_VALUE, Double.MAX_VALUE);
        button.setStyle(getButtonStyle(text));

        button.setOnAction(e -> handleButton(text));

        // 悬停效果
        button.setOnMouseEntered(e -> 
            button.setStyle(getButtonStyle(text) + "-fx-opacity: 0.8;")
        );
        button.setOnMouseExited(e -> 
            button.setStyle(getButtonStyle(text))
        );

        return button;
    }

    private String getButtonStyle(String text) {
        String baseStyle = 
            "-fx-font-size: 24px; " +
            "-fx-font-weight: bold; " +
            "-fx-cursor: hand; " +
            "-fx-border-radius: 5; " +
            "-fx-background-radius: 5; ";

        if (text.matches("[0-9]")) {
            // 数字按钮 - 深灰色
            return baseStyle + 
                "-fx-background-color: #3a3a3a; " +
                "-fx-text-fill: white;";
        } else if (text.equals("=")) {
            // 等号 - 蓝色
            return baseStyle + 
                "-fx-background-color: #3498db; " +
                "-fx-text-fill: white;";
        } else if (text.matches("[+\\-×÷%]")) {
            // 运算符 - 橙色
            return baseStyle + 
                "-fx-background-color: #e67e22; " +
                "-fx-text-fill: white;";
        } else {
            // 功能按钮 - 浅灰色
            return baseStyle + 
                "-fx-background-color: #7f8c8d; " +
                "-fx-text-fill: white;";
        }
    }

    private void handleButton(String text) {
        switch (text) {
            case "AC":
                allClear();
                break;
            case "C":
                clear();
                break;
            case "=":
                calculate();
                break;
            case "+":
            case "-":
            case "×":
            case "÷":
            case "%":
                setOperator(text);
                break;
            case ".":
                addDecimalPoint();
                break;
            default:
                addDigit(text);
                break;
        }
    }

    private void addDigit(String digit) {
        if (startNewNumber) {
            display.setText(digit);
            startNewNumber = false;
        } else {
            String current = display.getText();
            if (current.equals("0")) {
                display.setText(digit);
            } else {
                display.setText(current + digit);
            }
        }
    }

    private void addDecimalPoint() {
        String current = display.getText();
        if (startNewNumber) {
            display.setText("0.");
            startNewNumber = false;
        } else if (!current.contains(".")) {
            display.setText(current + ".");
        }
    }

    private void setOperator(String op) {
        if (!operator.isEmpty() && !startNewNumber) {
            calculate();
        }
        firstOperand = Double.parseDouble(display.getText());
        operator = op;
        startNewNumber = true;
    }

    private void calculate() {
        if (operator.isEmpty()) {
            return;
        }

        double secondOperand = Double.parseDouble(display.getText());
        double result = 0;

        switch (operator) {
            case "+":
                result = firstOperand + secondOperand;
                break;
            case "-":
                result = firstOperand - secondOperand;
                break;
            case "×":
                result = firstOperand * secondOperand;
                break;
            case "÷":
                if (secondOperand != 0) {
                    result = firstOperand / secondOperand;
                } else {
                    display.setText("错误");
                    operator = "";
                    startNewNumber = true;
                    return;
                }
                break;
            case "%":
                result = firstOperand % secondOperand;
                break;
        }

        // 格式化结果
        String resultText;
        if (result == (long) result) {
            resultText = String.format("%d", (long) result);
        } else {
            resultText = String.format("%.8f", result);
            // 移除末尾的0
            resultText = resultText.replaceAll("0+$", "").replaceAll("\\.$", "");
        }

        display.setText(resultText);
        operator = "";
        startNewNumber = true;
    }

    private void clear() {
        display.setText("0");
        startNewNumber = true;
    }

    private void allClear() {
        display.setText("0");
        operator = "";
        firstOperand = 0;
        startNewNumber = true;
    }

    private void handleKeyPress(KeyEvent event) {
        String character = event.getCharacter();
        if (character.matches("[0-9]")) {
            addDigit(character);
        } else if (character.equals(".")) {
            addDecimalPoint();
        } else if (character.equals("+") || character.equals("-") || 
                   character.equals("*") || character.equals("/")) {
            String op = character.equals("*") ? "×" : 
                       character.equals("/") ? "÷" : character;
            setOperator(op);
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 17.3 功能说明

### 17.3.1 核心功能

1. **数字输入**：
   - 点击数字按钮输入
   - 支持键盘数字键输入
   - 自动处理前导零

2. **小数点**：
   - 只能输入一个小数点
   - 自动补充"0."

3. **运算符**：
   - 支持+、-、×、÷、%
   - 连续计算（不按等号）

4. **清除功能**：
   - C：清除当前输入
   - AC：全部清除
   - Backspace：删除最后一位
   - ESC：全部清除

5. **计算**：
   - =按钮或Enter键
   - 除零错误处理
   - 结果格式化

### 17.3.2 用户体验优化

1. **视觉反馈**：
   - 按钮悬停效果
   - 不同类型按钮不同颜色
   - 现代化深色主题

2. **键盘支持**：
   - 数字键：0-9
   - 运算符：+ - * /
   - Enter：计算
   - Backspace：删除
   - ESC：清空

3. **错误处理**：
   - 除零显示错误
   - 自动清除错误状态

// 添加按钮：sin, cos, tan, log, sqrt, x², x³
private void scientificCalculation(String function) {
    double value = Double.parseDouble(display.getText());
    double result = switch (function) {
        case "sin" -> Math.sin(Math.toRadians(value));
        case "cos" -> Math.cos(Math.toRadians(value));
        case "tan" -> Math.tan(Math.toRadians(value));
        case "log" -> Math.log10(value);
        case "√" -> Math.sqrt(value);
        case "x²" -> value * value;
        case "x³" -> value * value * value;
        default -> value;
    };
    display.setText(String.valueOf(result));
    startNewNumber = true;
}
