# 12-图表和可视化

## 12.1 JavaFX图表概述

JavaFX提供了强大的图表API，用于数据可视化。所有图表都继承自`Chart`类。

### 12.1.1 图表类型

- **LineChart**：折线图
- **AreaChart**：面积图
- **BarChart**：柱状图
- **PieChart**：饼图
- **ScatterChart**：散点图
- **BubbleChart**：气泡图

## 12.2 折线图（LineChart）

折线图用于显示数据随时间或类别的变化趋势。

## 12.4 饼图（PieChart）

```java
package com.example.javafx.charts;

import javafx.application.Application;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.geometry.Side;
import javafx.scene.Scene;
import javafx.scene.chart.PieChart;
import javafx.scene.control.Label;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;

public class PieChartDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        // 创建数据
        ObservableList<PieChart.Data> pieChartData = FXCollections.observableArrayList(
            new PieChart.Data("Java", 35),
            new PieChart.Data("Python", 25),
            new PieChart.Data("JavaScript", 20),
            new PieChart.Data("C++", 12),
            new PieChart.Data("其他", 8)
        );
        
        // 创建饼图
        PieChart pieChart = new PieChart(pieChartData);
        pieChart.setTitle("编程语言使用占比");
        pieChart.setLegendSide(Side.RIGHT);
        
        // 标签显示
        Label captionLabel = new Label("");
        captionLabel.setStyle("-fx-font-size: 16px; -fx-font-weight: bold;");
        
        // 鼠标悬停显示详细信息
        for (PieChart.Data data : pieChartData) {
            data.getNode().setOnMouseEntered(e -> {
                captionLabel.setText(String.format("%s: %.0f%%", 
                    data.getName(), 
                    (data.getPieValue() / 100 * 100)));
            });
        }
        
        BorderPane root = new BorderPane();
        root.setCenter(pieChart);
        root.setBottom(captionLabel);
        
        Scene scene = new Scene(root, 800, 600);
        primaryStage.setTitle("饼图示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 12.6 散点图（ScatterChart）

```java
package com.example.javafx.charts;

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.chart.NumberAxis;
import javafx.scene.chart.ScatterChart;
import javafx.scene.chart.XYChart;
import javafx.stage.Stage;

import java.util.Random;

public class ScatterChartDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        NumberAxis xAxis = new NumberAxis(0, 100, 10);
        xAxis.setLabel("成绩");
        
        NumberAxis yAxis = new NumberAxis(0, 100, 10);
        yAxis.setLabel("出勤率");
        
        ScatterChart<Number, Number> scatterChart = new ScatterChart<>(xAxis, yAxis);
        scatterChart.setTitle("成绩与出勤率关系");
        
        // 生成随机数据
        XYChart.Series<Number, Number> series = new XYChart.Series<>();
        series.setName("学生数据");
        
        Random random = new Random();
        for (int i = 0; i < 50; i++) {
            double attendance = 60 + random.nextDouble() * 40;
            double score = attendance - 10 + random.nextDouble() * 20;
            series.getData().add(new XYChart.Data<>(score, attendance));
        }
        
        scatterChart.getData().add(series);
        
        Scene scene = new Scene(scatterChart, 800, 600);
        primaryStage.setTitle("散点图示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 12.7 动态图表

实时更新数据的动态图表。

```java
package com.example.javafx.charts;

import javafx.animation.Animation;
import javafx.animation.KeyFrame;
import javafx.animation.Timeline;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.chart.LineChart;
import javafx.scene.chart.NumberAxis;
import javafx.scene.chart.XYChart;
import javafx.stage.Stage;
import javafx.util.Duration;

import java.util.Random;

public class DynamicChartDemo extends Application {

    private int time = 0;
    private XYChart.Series<Number, Number> series;
    private Random random = new Random();

    @Override
    public void start(Stage primaryStage) {
        NumberAxis xAxis = new NumberAxis(0, 60, 5);
        xAxis.setLabel("时间（秒）");
        xAxis.setAnimated(false);
        
        NumberAxis yAxis = new NumberAxis(0, 100, 10);
        yAxis.setLabel("数值");
        
        LineChart<Number, Number> lineChart = new LineChart<>(xAxis, yAxis);
        lineChart.setTitle("实时数据监控");
        lineChart.setAnimated(false);
        lineChart.setCreateSymbols(false);
        
        series = new XYChart.Series<>();
        series.setName("传感器数据");
        lineChart.getData().add(series);
        
        // 定时更新数据
        Timeline timeline = new Timeline(new KeyFrame(Duration.millis(100), e -> {
            time++;
            
            // 添加新数据点
            double value = 50 + random.nextDouble() * 50;
            series.getData().add(new XYChart.Data<>(time / 10.0, value));
            
            // 保持最多显示60秒的数据
            if (series.getData().size() > 600) {
                series.getData().remove(0);
            }
            
            // 更新X轴范围
            if (time / 10.0 > 60) {
                xAxis.setLowerBound(time / 10.0 - 60);
                xAxis.setUpperBound(time / 10.0);
            }
        }));
        
        timeline.setCycleCount(Animation.INDEFINITE);
        timeline.play();
        
        Scene scene = new Scene(lineChart, 800, 600);
        primaryStage.setTitle("动态图表示例");
        primaryStage.setScene(scene);
        primaryStage.setOnCloseRequest(e -> timeline.stop());
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 12.9 综合仪表盘

```java
package com.example.javafx.charts;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.chart.*;
import javafx.scene.layout.GridPane;
import javafx.stage.Stage;

public class DashboardDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        GridPane grid = new GridPane();
        grid.setHgap(10);
        grid.setVgap(10);
        grid.setPadding(new Insets(10));
        grid.setStyle("-fx-background-color: #f5f5f5;");
        
        // 左上：折线图
        LineChart<Number, Number> lineChart = createLineChart();
        grid.add(lineChart, 0, 0);
        
        // 右上：柱状图
        BarChart<String, Number> barChart = createBarChart();
        grid.add(barChart, 1, 0);
        
        // 左下：饼图
        PieChart pieChart = createPieChart();
        grid.add(pieChart, 0, 1);
        
        // 右下：面积图
        AreaChart<Number, Number> areaChart = createAreaChart();
        grid.add(areaChart, 1, 1);
        
        Scene scene = new Scene(grid, 1200, 800);
        primaryStage.setTitle("数据仪表盘");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private LineChart<Number, Number> createLineChart() {
        NumberAxis xAxis = new NumberAxis();
        NumberAxis yAxis = new NumberAxis();
        LineChart<Number, Number> chart = new LineChart<>(xAxis, yAxis);
        chart.setTitle("销售趋势");
        chart.setPrefSize(580, 380);
        
        XYChart.Series<Number, Number> series = new XYChart.Series<>();
        series.setName("本月");
        for (int i = 1; i <= 7; i++) {
            series.getData().add(new XYChart.Data<>(i, Math.random() * 100));
        }
        chart.getData().add(series);
        return chart;
    }

    private BarChart<String, Number> createBarChart() {
        CategoryAxis xAxis = new CategoryAxis();
        NumberAxis yAxis = new NumberAxis();
        BarChart<String, Number> chart = new BarChart<>(xAxis, yAxis);
        chart.setTitle("产品对比");
        chart.setPrefSize(580, 380);
        
        XYChart.Series<String, Number> series = new XYChart.Series<>();
        series.getData().add(new XYChart.Data<>("A", 80));
        series.getData().add(new XYChart.Data<>("B", 65));
        series.getData().add(new XYChart.Data<>("C", 90));
        chart.getData().add(series);
        return chart;
    }

    private PieChart createPieChart() {
        PieChart chart = new PieChart();
        chart.setTitle("市场份额");
        chart.setPrefSize(580, 380);
        chart.getData().addAll(
            new PieChart.Data("产品A", 35),
            new PieChart.Data("产品B", 25),
            new PieChart.Data("产品C", 20),
            new PieChart.Data("其他", 20)
        );
        return chart;
    }

    private AreaChart<Number, Number> createAreaChart() {
        NumberAxis xAxis = new NumberAxis();
        NumberAxis yAxis = new NumberAxis();
        AreaChart<Number, Number> chart = new AreaChart<>(xAxis, yAxis);
        chart.setTitle("流量统计");
        chart.setPrefSize(580, 380);
        
        XYChart.Series<Number, Number> series = new XYChart.Series<>();
        for (int i = 1; i <= 7; i++) {
            series.getData().add(new XYChart.Data<>(i, Math.random() * 1000));
        }
        chart.getData().add(series);
        return chart;
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```
