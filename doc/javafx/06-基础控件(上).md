# 06-基础控件(上)

## 6.1 控件概述

JavaFX提供了丰富的UI控件（Control），所有控件都继承自`javafx.scene.control.Control`类。本章介绍常用的基础控件。

## 6.2 Label（标签）

Label用于显示文本或图像，是最基础的控件。

### 6.2.1 基本使用

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.ContentDisplay;
import javafx.scene.control.Label;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.layout.VBox;
import javafx.scene.paint.Color;
import javafx.scene.text.Font;
import javafx.scene.text.FontWeight;
import javafx.stage.Stage;

public class LabelDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        root.setAlignment(Pos.CENTER_LEFT);
        
        // 基本标签
        Label basicLabel = new Label("这是一个基本标签");
        
        // 设置字体
        Label fontLabel = new Label("自定义字体标签");
        fontLabel.setFont(Font.font("Arial", FontWeight.BOLD, 18));
        
        // 设置颜色
        Label colorLabel = new Label("彩色标签");
        colorLabel.setTextFill(Color.BLUE);
        colorLabel.setStyle("-fx-background-color: yellow; -fx-padding: 10;");
        
        // 长文本换行
        Label wrapLabel = new Label("这是一个很长的文本标签，当文本超过标签宽度时，" +
                                   "可以设置自动换行来显示完整内容。");
        wrapLabel.setWrapText(true);
        wrapLabel.setMaxWidth(300);
        
        // 带图标的标签
        try {
            ImageView icon = new ImageView(
                new Image("https://openjfx.io/images/logo.png")
            );
            icon.setFitWidth(32);
            icon.setFitHeight(32);
            
            Label iconLabel = new Label("带图标的标签", icon);
            iconLabel.setContentDisplay(ContentDisplay.LEFT);  // 图标在左侧
            root.getChildren().add(iconLabel);
        } catch (Exception e) {
            System.out.println("加载图标失败: " + e.getMessage());
        }
        
        root.getChildren().addAll(basicLabel, fontLabel, colorLabel, wrapLabel);
        
        Scene scene = new Scene(root, 450, 350);
        primaryStage.setTitle("Label 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.3 Button（按钮）

Button是最常用的交互控件。

### 6.3.1 基本使用

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ButtonDemo extends Application {

    private int clickCount = 0;

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        root.setAlignment(Pos.CENTER);
        
        Label label = new Label("点击次数: 0");
        label.setStyle("-fx-font-size: 16px;");
        
        // 基本按钮
        Button basicBtn = new Button("基本按钮");
        basicBtn.setOnAction(e -> {
            clickCount++;
            label.setText("点击次数: " + clickCount);
        });
        
        // 样式按钮
        Button styledBtn = new Button("样式按钮");
        styledBtn.setStyle(
            "-fx-background-color: #4CAF50; " +
            "-fx-text-fill: white; " +
            "-fx-font-size: 14px; " +
            "-fx-padding: 10 20; " +
            "-fx-background-radius: 5;"
        );
        
        // 禁用按钮
        Button disabledBtn = new Button("禁用按钮");
        disabledBtn.setDisable(true);
        
        // 默认按钮（回车触发）
        Button defaultBtn = new Button("默认按钮（回车触发）");
        defaultBtn.setDefaultButton(true);
        defaultBtn.setOnAction(e -> 
            System.out.println("默认按钮被触发")
        );
        
        // 取消按钮（ESC触发）
        Button cancelBtn = new Button("取消按钮（ESC触发）");
        cancelBtn.setCancelButton(true);
        cancelBtn.setOnAction(e -> 
            System.out.println("取消按钮被触发")
        );
        
        root.getChildren().addAll(label, basicBtn, styledBtn, disabledBtn, 
                                 defaultBtn, cancelBtn);
        
        Scene scene = new Scene(root, 400, 400);
        primaryStage.setTitle("Button 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.4 TextField（文本框）

TextField用于单行文本输入。

### 6.4.1 基本使用

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;
import javafx.scene.layout.GridPane;
import javafx.stage.Stage;

public class TextFieldDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        GridPane grid = new GridPane();
        grid.setPadding(new Insets(20));
        grid.setHgap(10);
        grid.setVgap(15);
        
        // 基本文本框
        Label basicLabel = new Label("基本文本框:");
        TextField basicField = new TextField();
        basicField.setPromptText("请输入文本");
        grid.add(basicLabel, 0, 0);
        grid.add(basicField, 1, 0);
        
        // 带默认值的文本框
        Label defaultLabel = new Label("默认值:");
        TextField defaultField = new TextField("默认文本");
        grid.add(defaultLabel, 0, 1);
        grid.add(defaultField, 1, 1);
        
        // 只读文本框
        Label readonlyLabel = new Label("只读:");
        TextField readonlyField = new TextField("只读文本");
        readonlyField.setEditable(false);
        grid.add(readonlyLabel, 0, 2);
        grid.add(readonlyField, 1, 2);
        
        // 限制字符数
        Label limitLabel = new Label("限制10字符:");
        TextField limitField = new TextField();
        limitField.textProperty().addListener((obs, oldVal, newVal) -> {
            if (newVal.length() > 10) {
                limitField.setText(oldVal);
            }
        });
        grid.add(limitLabel, 0, 3);
        grid.add(limitField, 1, 3);
        
        // 数字输入
        Label numberLabel = new Label("只能输入数字:");
        TextField numberField = new TextField();
        numberField.textProperty().addListener((obs, oldVal, newVal) -> {
            if (!newVal.matches("\\d*")) {
                numberField.setText(newVal.replaceAll("[^\\d]", ""));
            }
        });
        grid.add(numberLabel, 0, 4);
        grid.add(numberField, 1, 4);
        
        // 结果显示
        Label resultLabel = new Label();
        resultLabel.setStyle("-fx-font-weight: bold;");
        grid.add(resultLabel, 0, 5, 2, 1);
        
        // 提交按钮
        Button submitBtn = new Button("提交");
        submitBtn.setOnAction(e -> {
            String text = basicField.getText();
            resultLabel.setText("输入的内容: " + text);
        });
        grid.add(submitBtn, 1, 6);
        
        Scene scene = new Scene(grid, 450, 350);
        primaryStage.setTitle("TextField 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.5 TextArea（文本区域）

TextArea用于多行文本输入。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.TextArea;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class TextAreaDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(15));
        
        Label label = new Label("文本编辑器:");
        BorderPane.setMargin(label, new Insets(0, 0, 10, 0));
        root.setTop(label);
        
        // 文本区域
        TextArea textArea = new TextArea();
        textArea.setPromptText("在这里输入多行文本...");
        textArea.setWrapText(true);  // 自动换行
        textArea.setPrefRowCount(10);
        root.setCenter(textArea);
        
        // 底部按钮
        HBox buttonBox = new HBox(10);
        buttonBox.setPadding(new Insets(10, 0, 0, 0));
        
        Button clearBtn = new Button("清空");
        clearBtn.setOnAction(e -> textArea.clear());
        
        Button countBtn = new Button("字符统计");
        countBtn.setOnAction(e -> {
            int length = textArea.getText().length();
            int lines = textArea.getText().split("\n").length;
            System.out.println("字符数: " + length + ", 行数: " + lines);
        });
        
        Button copyBtn = new Button("复制");
        copyBtn.setOnAction(e -> textArea.copy());
        
        Button pasteBtn = new Button("粘贴");
        pasteBtn.setOnAction(e -> textArea.paste());
        
        buttonBox.getChildren().addAll(clearBtn, countBtn, copyBtn, pasteBtn);
        root.setBottom(buttonBox);
        
        Scene scene = new Scene(root, 500, 400);
        primaryStage.setTitle("TextArea 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.6 PasswordField（密码框）

PasswordField是TextField的子类，用于密码输入，输入的字符会被隐藏。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class PasswordFieldDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        GridPane grid = new GridPane();
        grid.setAlignment(Pos.CENTER);
        grid.setPadding(new Insets(25));
        grid.setHgap(10);
        grid.setVgap(10);
        
        Label titleLabel = new Label("登录");
        titleLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold;");
        grid.add(titleLabel, 0, 0, 2, 1);
        
        // 用户名
        Label userLabel = new Label("用户名:");
        grid.add(userLabel, 0, 1);
        TextField userField = new TextField();
        userField.setPromptText("请输入用户名");
        grid.add(userField, 1, 1);
        
        // 密码
        Label pwLabel = new Label("密码:");
        grid.add(pwLabel, 0, 2);
        PasswordField pwField = new PasswordField();
        pwField.setPromptText("请输入密码");
        grid.add(pwField, 1, 2);
        
        // 显示密码复选框
        CheckBox showPwCb = new CheckBox("显示密码");
        TextField pwTextField = new TextField();
        pwTextField.setManaged(false);
        pwTextField.setVisible(false);
        pwTextField.setPromptText("请输入密码");
        grid.add(pwTextField, 1, 2);
        
        // 绑定密码显示/隐藏
        pwTextField.managedProperty().bind(showPwCb.selectedProperty());
        pwTextField.visibleProperty().bind(showPwCb.selectedProperty());
        pwField.managedProperty().bind(showPwCb.selectedProperty().not());
        pwField.visibleProperty().bind(showPwCb.selectedProperty().not());
        pwTextField.textProperty().bindBidirectional(pwField.textProperty());
        
        grid.add(showPwCb, 1, 3);
        
        // 记住密码
        CheckBox rememberCb = new CheckBox("记住密码");
        grid.add(rememberCb, 1, 4);
        
        // 按钮
        HBox buttonBox = new HBox(15);
        buttonBox.setAlignment(Pos.CENTER_RIGHT);
        
        Button loginBtn = new Button("登录");
        loginBtn.setDefaultButton(true);
        loginBtn.setOnAction(e -> {
            String username = userField.getText();
            String password = pwField.getText();
            
            if (username.isEmpty() || password.isEmpty()) {
                showAlert(Alert.AlertType.WARNING, "请填写用户名和密码");
            } else if (username.equals("admin") && password.equals("123456")) {
                showAlert(Alert.AlertType.INFORMATION, "登录成功！");
            } else {
                showAlert(Alert.AlertType.ERROR, "用户名或密码错误");
            }
        });
        
        Button cancelBtn = new Button("取消");
        cancelBtn.setOnAction(e -> primaryStage.close());
        
        buttonBox.getChildren().addAll(loginBtn, cancelBtn);
        grid.add(buttonBox, 0, 5, 2, 1);
        
        Scene scene = new Scene(grid, 400, 300);
        primaryStage.setTitle("PasswordField 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void showAlert(Alert.AlertType type, String message) {
        Alert alert = new Alert(type);
        alert.setTitle(type == Alert.AlertType.ERROR ? "错误" : 
                      type == Alert.AlertType.WARNING ? "警告" : "提示");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.7 CheckBox（复选框）

CheckBox允许用户进行多选。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.CheckBox;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class CheckBoxDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        Label titleLabel = new Label("选择你喜欢的水果:");
        titleLabel.setStyle("-fx-font-size: 16px; -fx-font-weight: bold;");
        
        // 创建复选框
        CheckBox appleCb = new CheckBox("苹果");
        CheckBox bananaCb = new CheckBox("香蕉");
        CheckBox orangeCb = new CheckBox("橙子");
        CheckBox grapeCb = new CheckBox("葡萄");
        
        // 设置默认选中
        appleCb.setSelected(true);
        
        // 添加监听器
        appleCb.selectedProperty().addListener((obs, oldVal, newVal) -> {
            System.out.println("苹果: " + (newVal ? "选中" : "未选中"));
        });
        
        // 全选复选框
        CheckBox selectAllCb = new CheckBox("全选");
        selectAllCb.setStyle("-fx-font-weight: bold;");
        selectAllCb.setOnAction(e -> {
            boolean selected = selectAllCb.isSelected();
            appleCb.setSelected(selected);
            bananaCb.setSelected(selected);
            orangeCb.setSelected(selected);
            grapeCb.setSelected(selected);
        });
        
        // 结果标签
        Label resultLabel = new Label();
        resultLabel.setStyle("-fx-font-size: 14px; -fx-text-fill: blue;");
        
        // 提交按钮
        Button submitBtn = new Button("提交");
        submitBtn.setOnAction(e -> {
            StringBuilder sb = new StringBuilder("你选择了: ");
            if (appleCb.isSelected()) sb.append("苹果 ");
            if (bananaCb.isSelected()) sb.append("香蕉 ");
            if (orangeCb.isSelected()) sb.append("橙子 ");
            if (grapeCb.isSelected()) sb.append("葡萄 ");
            
            if (sb.length() == 7) {  // "你选择了: " 长度为7
                sb.append("(无)");
            }
            resultLabel.setText(sb.toString());
        });
        
        root.getChildren().addAll(titleLabel, appleCb, bananaCb, orangeCb, grapeCb,
                                 new Label("---"), selectAllCb, submitBtn, resultLabel);
        
        Scene scene = new Scene(root, 350, 400);
        primaryStage.setTitle("CheckBox 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.8 RadioButton（单选按钮）

RadioButton用于互斥选择，同一组内只能选一个。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class RadioButtonDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        // 性别选择
        Label genderLabel = new Label("性别:");
        genderLabel.setStyle("-fx-font-size: 16px; -fx-font-weight: bold;");
        
        ToggleGroup genderGroup = new ToggleGroup();
        
        RadioButton maleRb = new RadioButton("男");
        maleRb.setToggleGroup(genderGroup);
        maleRb.setSelected(true);  // 默认选中
        
        RadioButton femaleRb = new RadioButton("女");
        femaleRb.setToggleGroup(genderGroup);
        
        RadioButton otherRb = new RadioButton("其他");
        otherRb.setToggleGroup(genderGroup);
        
        // 监听选择变化
        genderGroup.selectedToggleProperty().addListener((obs, oldToggle, newToggle) -> {
            if (newToggle != null) {
                RadioButton selected = (RadioButton) newToggle;
                System.out.println("选择了: " + selected.getText());
            }
        });
        
        Separator separator = new Separator();
        
        // 学历选择
        Label eduLabel = new Label("学历:");
        eduLabel.setStyle("-fx-font-size: 16px; -fx-font-weight: bold;");
        
        ToggleGroup eduGroup = new ToggleGroup();
        
        RadioButton highSchoolRb = new RadioButton("高中");
        highSchoolRb.setToggleGroup(eduGroup);
        
        RadioButton bachelorRb = new RadioButton("本科");
        bachelorRb.setToggleGroup(eduGroup);
        bachelorRb.setSelected(true);
        
        RadioButton masterRb = new RadioButton("硕士");
        masterRb.setToggleGroup(eduGroup);
        
        RadioButton doctorRb = new RadioButton("博士");
        doctorRb.setToggleGroup(eduGroup);
        
        // 结果标签
        Label resultLabel = new Label();
        resultLabel.setStyle("-fx-font-size: 14px; -fx-text-fill: blue;");
        
        // 提交按钮
        Button submitBtn = new Button("提交");
        submitBtn.setOnAction(e -> {
            RadioButton selectedGender = (RadioButton) genderGroup.getSelectedToggle();
            RadioButton selectedEdu = (RadioButton) eduGroup.getSelectedToggle();
            
            String result = String.format("性别: %s, 学历: %s",
                                         selectedGender.getText(),
                                         selectedEdu.getText());
            resultLabel.setText(result);
        });
        
        root.getChildren().addAll(
            genderLabel, maleRb, femaleRb, otherRb,
            separator,
            eduLabel, highSchoolRb, bachelorRb, masterRb, doctorRb,
            submitBtn, resultLabel
        );
        
        Scene scene = new Scene(root, 350, 450);
        primaryStage.setTitle("RadioButton 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 6.9 小结

本章学习了JavaFX的基础控件（上）：

1. ✅ **Label**：显示文本和图像
2. ✅ **Button**：用户交互按钮
3. ✅ **TextField**：单行文本输入
4. ✅ **TextArea**：多行文本输入
5. ✅ **PasswordField**：密码输入
6. ✅ **CheckBox**：多选框
7. ✅ **RadioButton**：单选按钮

这些是构建JavaFX应用最常用的基础控件。
