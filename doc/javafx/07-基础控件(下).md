## 7.1 ComboBox（下拉框）

ComboBox提供下拉选择功能，用户可以从预定义的选项中选择。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.collections.FXCollections;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.ComboBox;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ComboBoxDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        // 基本ComboBox
        Label label1 = new Label("选择城市:");
        ComboBox<String> cityCombo = new ComboBox<>();
        cityCombo.getItems().addAll("北京", "上海", "广州", "深圳", "杭州");
        cityCombo.setValue("北京");  // 设置默认值
        
        // 使用ObservableList
        Label label2 = new Label("选择编程语言:");
        ComboBox<String> langCombo = new ComboBox<>(
            FXCollections.observableArrayList(
                "Java", "Python", "JavaScript", "C++", "Go"
            )
        );
        
        // 可编辑的ComboBox
        Label label3 = new Label("选择或输入框架:");
        ComboBox<String> frameworkCombo = new ComboBox<>();
        frameworkCombo.setEditable(true);  // 允许用户输入
        frameworkCombo.getItems().addAll("Spring", "Hibernate", "MyBatis", "JavaFX");
        frameworkCombo.setPromptText("选择或输入...");
        
        // 监听选择变化
        cityCombo.setOnAction(e -> {
            String selected = cityCombo.getValue();
            System.out.println("选择的城市: " + selected);
        });
        
        // 结果标签
        Label resultLabel = new Label();
        resultLabel.setStyle("-fx-font-size: 14px; -fx-text-fill: blue;");
        
        Button submitBtn = new Button("提交");
        submitBtn.setOnAction(e -> {
            String city = cityCombo.getValue();
            String lang = langCombo.getValue();
            String framework = frameworkCombo.getValue();
            
            resultLabel.setText(String.format("城市: %s, 语言: %s, 框架: %s",
                               city != null ? city : "未选择",
                               lang != null ? lang : "未选择",
                               framework != null ? framework : "未选择"));
        });
        
        root.getChildren().addAll(label1, cityCombo, label2, langCombo,
                                 label3, frameworkCombo, submitBtn, resultLabel);
        
        Scene scene = new Scene(root, 400, 400);
        primaryStage.setTitle("ComboBox 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 7.2 ListView（列表视图）

ListView用于显示可滚动的项目列表。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class ListViewDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(15));
        
        // 创建ListView
        ObservableList<String> items = FXCollections.observableArrayList(
            "Java", "Python", "JavaScript", "C++", "Go", "Rust",
            "TypeScript", "Kotlin", "Swift", "PHP"
        );
        
        ListView<String> listView = new ListView<>(items);
        listView.setPrefHeight(300);
        
        // 设置选择模式
        listView.getSelectionModel().setSelectionMode(SelectionMode.MULTIPLE);
        
        root.setCenter(listView);
        
        // 底部按钮
        HBox buttonBox = new HBox(10);
        buttonBox.setPadding(new Insets(10, 0, 0, 0));
        
        Button addBtn = new Button("添加");
        addBtn.setOnAction(e -> {
            TextInputDialog dialog = new TextInputDialog();
            dialog.setTitle("添加项目");
            dialog.setHeaderText("添加新项目");
            dialog.setContentText("请输入:");
            dialog.showAndWait().ifPresent(text -> {
                if (!text.isEmpty()) {
                    items.add(text);
                }
            });
        });
        
        Button removeBtn = new Button("删除选中");
        removeBtn.setOnAction(e -> {
            ObservableList<String> selected = 
                listView.getSelectionModel().getSelectedItems();
            items.removeAll(selected);
        });
        
        Button clearBtn = new Button("清空");
        clearBtn.setOnAction(e -> items.clear());
        
        Button printBtn = new Button("打印选中");
        printBtn.setOnAction(e -> {
            ObservableList<String> selected = 
                listView.getSelectionModel().getSelectedItems();
            System.out.println("选中的项: " + selected);
        });
        
        buttonBox.getChildren().addAll(addBtn, removeBtn, clearBtn, printBtn);
        root.setBottom(buttonBox);
        
        Scene scene = new Scene(root, 400, 450);
        primaryStage.setTitle("ListView 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 7.3 TableView（表格视图）

TableView用于显示表格数据，是最复杂也最强大的控件之一。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.beans.property.SimpleIntegerProperty;
import javafx.beans.property.SimpleStringProperty;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

public class TableViewDemo extends Application {

    public static class Student {
        private final SimpleIntegerProperty id;
        private final SimpleStringProperty name;
        private final SimpleIntegerProperty age;
        private final SimpleStringProperty major;

        public Student(int id, String name, int age, String major) {
            this.id = new SimpleIntegerProperty(id);
            this.name = new SimpleStringProperty(name);
            this.age = new SimpleIntegerProperty(age);
            this.major = new SimpleStringProperty(major);
        }

        public int getId() { return id.get(); }
        public void setId(int value) { id.set(value); }
        public SimpleIntegerProperty idProperty() { return id; }

        public String getName() { return name.get(); }
        public void setName(String value) { name.set(value); }
        public SimpleStringProperty nameProperty() { return name; }

        public int getAge() { return age.get(); }
        public void setAge(int value) { age.set(value); }
        public SimpleIntegerProperty ageProperty() { return age; }

        public String getMajor() { return major.get(); }
        public void setMajor(String value) { major.set(value); }
        public SimpleStringProperty majorProperty() { return major; }
    }

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(15));
        
        // 创建TableView
        TableView<Student> tableView = new TableView<>();
        
        // 创建列
        TableColumn<Student, Integer> idCol = new TableColumn<>("学号");
        idCol.setCellValueFactory(new PropertyValueFactory<>("id"));
        idCol.setPrefWidth(80);
        
        TableColumn<Student, String> nameCol = new TableColumn<>("姓名");
        nameCol.setCellValueFactory(new PropertyValueFactory<>("name"));
        nameCol.setPrefWidth(120);
        
        TableColumn<Student, Integer> ageCol = new TableColumn<>("年龄");
        ageCol.setCellValueFactory(new PropertyValueFactory<>("age"));
        ageCol.setPrefWidth(80);
        
        TableColumn<Student, String> majorCol = new TableColumn<>("专业");
        majorCol.setCellValueFactory(new PropertyValueFactory<>("major"));
        majorCol.setPrefWidth(150);
        
        tableView.getColumns().addAll(idCol, nameCol, ageCol, majorCol);
        
        // 添加数据
        ObservableList<Student> data = FXCollections.observableArrayList(
            new Student(1001, "张三", 20, "计算机科学"),
            new Student(1002, "李四", 21, "软件工程"),
            new Student(1003, "王五", 19, "人工智能"),
            new Student(1004, "赵六", 22, "数据科学"),
            new Student(1005, "钱七", 20, "网络安全")
        );
        
        tableView.setItems(data);
        
        root.setCenter(tableView);
        
        // 底部按钮
        HBox buttonBox = new HBox(10);
        buttonBox.setPadding(new Insets(10, 0, 0, 0));
        
        Button addBtn = new Button("添加");
        addBtn.setOnAction(e -> {
            int nextId = data.size() + 1001;
            data.add(new Student(nextId, "新学生", 20, "未指定"));
        });
        
        Button removeBtn = new Button("删除选中");
        removeBtn.setOnAction(e -> {
            Student selected = tableView.getSelectionModel().getSelectedItem();
            if (selected != null) {
                data.remove(selected);
            }
        });
        
        Button editBtn = new Button("编辑选中");
        editBtn.setOnAction(e -> {
            Student selected = tableView.getSelectionModel().getSelectedItem();
            if (selected != null) {
                // 这里可以打开编辑对话框
                System.out.println("编辑: " + selected.getName());
            }
        });
        
        buttonBox.getChildren().addAll(addBtn, removeBtn, editBtn);
        root.setBottom(buttonBox);
        
        Scene scene = new Scene(root, 550, 450);
        primaryStage.setTitle("TableView 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 7.4 TreeView（树形视图）

TreeView用于显示层次结构的数据。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.TreeItem;
import javafx.scene.control.TreeView;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;

public class TreeViewDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(15));
        
        // 创建根节点
        TreeItem<String> rootItem = new TreeItem<>("项目");
        rootItem.setExpanded(true);
        
        // 创建子节点
        TreeItem<String> srcItem = new TreeItem<>("src");
        TreeItem<String> mainItem = new TreeItem<>("main");
        TreeItem<String> javaItem = new TreeItem<>("java");
        TreeItem<String> packageItem = new TreeItem<>("com.example");
        
        TreeItem<String> class1 = new TreeItem<>("Main.java");
        TreeItem<String> class2 = new TreeItem<>("App.java");
        TreeItem<String> class3 = new TreeItem<>("Utils.java");
        
        packageItem.getChildren().addAll(class1, class2, class3);
        javaItem.getChildren().add(packageItem);
        mainItem.getChildren().add(javaItem);
        
        TreeItem<String> resourcesItem = new TreeItem<>("resources");
        TreeItem<String> config = new TreeItem<>("application.properties");
        resourcesItem.getChildren().add(config);
        mainItem.getChildren().add(resourcesItem);
        
        srcItem.getChildren().add(mainItem);
        
        TreeItem<String> testItem = new TreeItem<>("test");
        TreeItem<String> testJava = new TreeItem<>("java");
        testItem.getChildren().add(testJava);
        srcItem.getChildren().add(testItem);
        
        TreeItem<String> pomItem = new TreeItem<>("pom.xml");
        TreeItem<String> readmeItem = new TreeItem<>("README.md");
        
        rootItem.getChildren().addAll(srcItem, pomItem, readmeItem);
        
        // 创建TreeView
        TreeView<String> treeView = new TreeView<>(rootItem);
        
        // 监听选择变化
        treeView.getSelectionModel().selectedItemProperty().addListener(
            (obs, oldVal, newVal) -> {
                if (newVal != null) {
                    System.out.println("选中: " + newVal.getValue());
                }
            }
        );
        
        root.setCenter(treeView);
        
        Scene scene = new Scene(root, 400, 500);
        primaryStage.setTitle("TreeView 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 7.5 MenuBar（菜单栏）

MenuBar用于创建应用程序的菜单系统。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.application.Platform;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.input.KeyCode;
import javafx.scene.input.KeyCodeCombination;
import javafx.scene.input.KeyCombination;
import javafx.scene.layout.BorderPane;
import javafx.stage.Stage;

public class MenuBarDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        
        // 创建菜单栏
        MenuBar menuBar = new MenuBar();
        
        // 文件菜单
        Menu fileMenu = new Menu("文件(_F)");
        fileMenu.setMnemonicParsing(true);
        
        MenuItem newItem = new MenuItem("新建");
        newItem.setAccelerator(new KeyCodeCombination(KeyCode.N, KeyCombination.CONTROL_DOWN));
        newItem.setOnAction(e -> System.out.println("新建文件"));
        
        MenuItem openItem = new MenuItem("打开...");
        openItem.setAccelerator(new KeyCodeCombination(KeyCode.O, KeyCombination.CONTROL_DOWN));
        openItem.setOnAction(e -> System.out.println("打开文件"));
        
        MenuItem saveItem = new MenuItem("保存");
        saveItem.setAccelerator(new KeyCodeCombination(KeyCode.S, KeyCombination.CONTROL_DOWN));
        saveItem.setOnAction(e -> System.out.println("保存文件"));
        
        MenuItem saveAsItem = new MenuItem("另存为...");
        
        SeparatorMenuItem separator1 = new SeparatorMenuItem();
        
        MenuItem exitItem = new MenuItem("退出");
        exitItem.setOnAction(e -> Platform.exit());
        
        fileMenu.getItems().addAll(newItem, openItem, saveItem, saveAsItem, 
                                   separator1, exitItem);
        
        // 编辑菜单
        Menu editMenu = new Menu("编辑(_E)");
        
        MenuItem undoItem = new MenuItem("撤销");
        undoItem.setAccelerator(new KeyCodeCombination(KeyCode.Z, KeyCombination.CONTROL_DOWN));
        
        MenuItem redoItem = new MenuItem("重做");
        redoItem.setAccelerator(new KeyCodeCombination(KeyCode.Y, KeyCombination.CONTROL_DOWN));
        
        SeparatorMenuItem separator2 = new SeparatorMenuItem();
        
        MenuItem cutItem = new MenuItem("剪切");
        cutItem.setAccelerator(new KeyCodeCombination(KeyCode.X, KeyCombination.CONTROL_DOWN));
        
        MenuItem copyItem = new MenuItem("复制");
        copyItem.setAccelerator(new KeyCodeCombination(KeyCode.C, KeyCombination.CONTROL_DOWN));
        
        MenuItem pasteItem = new MenuItem("粘贴");
        pasteItem.setAccelerator(new KeyCodeCombination(KeyCode.V, KeyCombination.CONTROL_DOWN));
        
        editMenu.getItems().addAll(undoItem, redoItem, separator2, cutItem, copyItem, pasteItem);
        
        // 视图菜单（带复选框）
        Menu viewMenu = new Menu("视图(_V)");
        
        CheckMenuItem toolbarItem = new CheckMenuItem("工具栏");
        toolbarItem.setSelected(true);
        
        CheckMenuItem statusBarItem = new CheckMenuItem("状态栏");
        statusBarItem.setSelected(true);
        
        viewMenu.getItems().addAll(toolbarItem, statusBarItem);
        
        // 帮助菜单（带单选按钮）
        Menu helpMenu = new Menu("帮助(_H)");
        
        MenuItem aboutItem = new MenuItem("关于");
        aboutItem.setOnAction(e -> {
            Alert alert = new Alert(Alert.AlertType.INFORMATION);
            alert.setTitle("关于");
            alert.setHeaderText("应用程序信息");
            alert.setContentText("版本: 1.0.0\n作者: JavaFX开发者");
            alert.showAndWait();
        });
        
        MenuItem helpContentItem = new MenuItem("帮助内容");
        
        helpMenu.getItems().addAll(helpContentItem, aboutItem);
        
        // 添加菜单到菜单栏
        menuBar.getMenus().addAll(fileMenu, editMenu, viewMenu, helpMenu);
        
        root.setTop(menuBar);
        
        // 中心内容
        TextArea textArea = new TextArea();
        textArea.setPromptText("在这里输入内容...");
        root.setCenter(textArea);
        
        Scene scene = new Scene(root, 600, 400);
        primaryStage.setTitle("MenuBar 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 7.6 ProgressBar 和 ProgressIndicator

用于显示进度的控件。

```java
package com.example.javafx.controls;

import javafx.application.Application;
import javafx.concurrent.Task;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ProgressDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setPadding(new Insets(30));
        root.setAlignment(Pos.CENTER);
        
        Label titleLabel = new Label("进度条示例");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        
        // 进度条
        ProgressBar progressBar = new ProgressBar(0);
        progressBar.setPrefWidth(300);
        
        // 进度指示器（圆形）
        ProgressIndicator progressIndicator = new ProgressIndicator(0);
        
        // 进度标签
        Label progressLabel = new Label("0%");
        
        // 开始按钮
        Button startBtn = new Button("开始任务");
        startBtn.setOnAction(e -> {
            startBtn.setDisable(true);
            
            // 创建后台任务
            Task<Void> task = new Task<Void>() {
                @Override
                protected Void call() throws Exception {
                    for (int i = 0; i <= 100; i++) {
                        updateProgress(i, 100);
                        updateMessage(i + "%");
                        Thread.sleep(50);  // 模拟耗时操作
                    }
                    return null;
                }
            };
            
            // 绑定进度
            progressBar.progressProperty().bind(task.progressProperty());
            progressIndicator.progressProperty().bind(task.progressProperty());
            progressLabel.textProperty().bind(task.messageProperty());
            
            // 任务完成后
            task.setOnSucceeded(event -> {
                startBtn.setDisable(false);
                Alert alert = new Alert(Alert.AlertType.INFORMATION);
                alert.setTitle("完成");
                alert.setContentText("任务执行完成！");
                alert.showAndWait();
            });
            
            // 在新线程中执行任务
            new Thread(task).start();
        });
        
        root.getChildren().addAll(titleLabel, progressBar, progressIndicator, 
                                 progressLabel, startBtn);
        
        Scene scene = new Scene(root, 400, 350);
        primaryStage.setTitle("Progress 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```
