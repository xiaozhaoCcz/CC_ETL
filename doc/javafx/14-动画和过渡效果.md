# 14-动画和过渡效果

## 14.1 动画基础

JavaFX提供了丰富的动画API，使界面更加生动。

### 14.1.1 Timeline动画

```java
package com.example.javafx.animation;

import javafx.animation.KeyFrame;
import javafx.animation.Timeline;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import javafx.util.Duration;

public class TimelineDemo extends Application {

    private int count = 0;

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setAlignment(Pos.CENTER);
        
        Label countLabel = new Label("0");
        countLabel.setStyle("-fx-font-size: 48px; -fx-font-weight: bold;");
        
        Button startBtn = new Button("开始计数");
        Button stopBtn = new Button("停止");
        stopBtn.setDisable(true);
        
        // 创建Timeline动画
        Timeline timeline = new Timeline(
            new KeyFrame(Duration.millis(100), e -> {
                count++;
                countLabel.setText(String.valueOf(count));
            })
        );
        timeline.setCycleCount(Timeline.INDEFINITE);
        
        startBtn.setOnAction(e -> {
            timeline.play();
            startBtn.setDisable(true);
            stopBtn.setDisable(false);
        });
        
        stopBtn.setOnAction(e -> {
            timeline.stop();
            startBtn.setDisable(false);
            stopBtn.setDisable(true);
        });
        
        root.getChildren().addAll(countLabel, startBtn, stopBtn);
        
        Scene scene = new Scene(root, 400, 300);
        primaryStage.setTitle("Timeline动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.2 TransLate过渡（移动）

```java
package com.example.javafx.animation;

import javafx.animation.TranslateTransition;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.stage.Stage;
import javafx.util.Duration;

public class TranslateDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        Pane root = new Pane();
        root.setStyle("-fx-background-color: lightgray;");
        
        // 创建圆形
        Circle circle = new Circle(50);
        circle.setFill(Color.BLUE);
        circle.setCenterX(100);
        circle.setCenterY(200);
        
        // 创建移动动画
        TranslateTransition translate = new TranslateTransition();
        translate.setNode(circle);
        translate.setDuration(Duration.seconds(2));
        translate.setByX(600);  // 向右移动600像素
        translate.setCycleCount(TranslateTransition.INDEFINITE);  // 无限循环
        translate.setAutoReverse(true);  // 自动反向
        
        // 控制按钮
        Button playBtn = new Button("播放");
        playBtn.setLayoutX(350);
        playBtn.setLayoutY(20);
        playBtn.setOnAction(e -> translate.play());
        
        Button pauseBtn = new Button("暂停");
        pauseBtn.setLayoutX(420);
        pauseBtn.setLayoutY(20);
        pauseBtn.setOnAction(e -> translate.pause());
        
        Button stopBtn = new Button("停止");
        stopBtn.setLayoutX(490);
        stopBtn.setLayoutY(20);
        stopBtn.setOnAction(e -> translate.stop());
        
        root.getChildren().addAll(circle, playBtn, pauseBtn, stopBtn);
        
        Scene scene = new Scene(root, 800, 400);
        primaryStage.setTitle("移动动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.3 Rotate过渡（旋转）

```java
package com.example.javafx.animation;

import javafx.animation.RotateTransition;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Rectangle;
import javafx.stage.Stage;
import javafx.util.Duration;

public class RotateDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        StackPane root = new StackPane();
        
        // 创建矩形
        Rectangle rect = new Rectangle(150, 100);
        rect.setFill(Color.CORAL);
        
        // 创建旋转动画
        RotateTransition rotate = new RotateTransition();
        rotate.setNode(rect);
        rotate.setDuration(Duration.seconds(3));
        rotate.setByAngle(360);  // 旋转360度
        rotate.setCycleCount(RotateTransition.INDEFINITE);
        
        // 按钮
        Button btn = new Button("开始旋转");
        btn.setOnAction(e -> {
            if (rotate.getStatus() == javafx.animation.Animation.Status.RUNNING) {
                rotate.pause();
                btn.setText("继续旋转");
            } else {
                rotate.play();
                btn.setText("暂停");
            }
        });
        
        StackPane.setAlignment(btn, Pos.BOTTOM_CENTER);
        StackPane.setMargin(btn, new javafx.geometry.Insets(0, 0, 20, 0));
        
        root.getChildren().addAll(rect, btn);
        
        Scene scene = new Scene(root, 400, 350);
        primaryStage.setTitle("旋转动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.5 Fade过渡（淡入淡出）

```java
package com.example.javafx.animation;

import javafx.animation.FadeTransition;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import javafx.util.Duration;

public class FadeDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(30);
        root.setAlignment(Pos.CENTER);
        
        Label label = new Label("淡入淡出效果");
        label.setStyle("-fx-font-size: 32px; -fx-font-weight: bold;");
        
        // 创建淡入淡出动画
        FadeTransition fade = new FadeTransition();
        fade.setNode(label);
        fade.setDuration(Duration.seconds(1.5));
        fade.setFromValue(1.0);  // 完全不透明
        fade.setToValue(0.1);    // 接近透明
        fade.setCycleCount(FadeTransition.INDEFINITE);
        fade.setAutoReverse(true);
        fade.play();
        
        Button btn = new Button("重新开始");
        btn.setOnAction(e -> fade.playFromStart());
        
        root.getChildren().addAll(label, btn);
        
        Scene scene = new Scene(root, 500, 300);
        primaryStage.setTitle("淡入淡出动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.6 Path动画（路径动画）

```java
package com.example.javafx.animation;

import javafx.animation.PathTransition;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.scene.shape.CubicCurveTo;
import javafx.scene.shape.MoveTo;
import javafx.scene.shape.Path;
import javafx.stage.Stage;
import javafx.util.Duration;

public class PathAnimationDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        Pane root = new Pane();
        root.setStyle("-fx-background-color: lightgray;");
        
        // 创建路径
        Path path = new Path();
        path.getElements().add(new MoveTo(50, 200));
        path.getElements().add(new CubicCurveTo(150, 50, 350, 350, 750, 200));
        path.setStroke(Color.BLUE);
        path.setStrokeWidth(2);
        path.setFill(null);
        
        // 创建移动的圆形
        Circle circle = new Circle(20);
        circle.setFill(Color.RED);
        
        // 创建路径动画
        PathTransition pathTransition = new PathTransition();
        pathTransition.setNode(circle);
        pathTransition.setPath(path);
        pathTransition.setDuration(Duration.seconds(4));
        pathTransition.setCycleCount(PathTransition.INDEFINITE);
        pathTransition.setAutoReverse(true);
        pathTransition.play();
        
        root.getChildren().addAll(path, circle);
        
        Scene scene = new Scene(root, 800, 400);
        primaryStage.setTitle("路径动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.7 组合动画

```java
package com.example.javafx.animation;

import javafx.animation.*;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Rectangle;
import javafx.stage.Stage;
import javafx.util.Duration;

public class CombinedAnimationDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        StackPane root = new StackPane();
        
        // 创建矩形
        Rectangle rect = new Rectangle(100, 100);
        rect.setFill(Color.DODGERBLUE);
        
        // 1. 移动动画
        TranslateTransition translate = new TranslateTransition(Duration.seconds(2), rect);
        translate.setToX(200);
        translate.setToY(100);
        
        // 2. 旋转动画
        RotateTransition rotate = new RotateTransition(Duration.seconds(2), rect);
        rotate.setByAngle(360);
        
        // 3. 缩放动画
        ScaleTransition scale = new ScaleTransition(Duration.seconds(2), rect);
        scale.setToX(1.5);
        scale.setToY(1.5);
        
        // 4. 淡入淡出
        FadeTransition fade = new FadeTransition(Duration.seconds(2), rect);
        fade.setFromValue(1.0);
        fade.setToValue(0.3);
        
        // 并行动画（同时执行）
        ParallelTransition parallel = new ParallelTransition(
            translate, rotate, scale, fade
        );
        parallel.setCycleCount(ParallelTransition.INDEFINITE);
        parallel.setAutoReverse(true);
        
        // 按钮
        Button btn = new Button("播放组合动画");
        btn.setOnAction(e -> parallel.play());
        StackPane.setAlignment(btn, Pos.BOTTOM_CENTER);
        StackPane.setMargin(btn, new javafx.geometry.Insets(0, 0, 20, 0));
        
        root.getChildren().addAll(rect, btn);
        
        Scene scene = new Scene(root, 600, 400);
        primaryStage.setTitle("组合动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.8 顺序动画

```java
package com.example.javafx.animation;

import javafx.animation.*;
import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.stage.Stage;
import javafx.util.Duration;

public class SequentialAnimationDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        Pane root = new Pane();
        root.setStyle("-fx-background-color: lightgray;");
        
        Circle circle = new Circle(30);
        circle.setFill(Color.CRIMSON);
        circle.setCenterX(100);
        circle.setCenterY(100);
        
        // 动画1：向右移动
        TranslateTransition move1 = new TranslateTransition(Duration.seconds(1), circle);
        move1.setToX(500);
        
        // 动画2：向下移动
        TranslateTransition move2 = new TranslateTransition(Duration.seconds(1), circle);
        move2.setToY(200);
        
        // 动画3：向左移动
        TranslateTransition move3 = new TranslateTransition(Duration.seconds(1), circle);
        move3.setToX(0);
        
        // 动画4：向上移动
        TranslateTransition move4 = new TranslateTransition(Duration.seconds(1), circle);
        move4.setToY(0);
        
        // 顺序动画（依次执行）
        SequentialTransition sequential = new SequentialTransition(
            move1, move2, move3, move4
        );
        sequential.setCycleCount(SequentialTransition.INDEFINITE);
        sequential.play();
        
        root.getChildren().add(circle);
        
        Scene scene = new Scene(root, 700, 400);
        primaryStage.setTitle("顺序动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.9 实战：加载动画

```java
package com.example.javafx.animation;

import javafx.animation.FadeTransition;
import javafx.animation.RotateTransition;
import javafx.animation.ScaleTransition;
import javafx.application.Application;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.control.ProgressIndicator;
import javafx.scene.layout.StackPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;
import javafx.util.Duration;

public class LoadingAnimationDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(30);
        root.setAlignment(Pos.CENTER);
        root.setStyle("-fx-background-color: white;");
        
        // 进度指示器
        ProgressIndicator progress = new ProgressIndicator();
        progress.setPrefSize(80, 80);
        
        // 旋转动画
        RotateTransition rotate = new RotateTransition(Duration.seconds(2), progress);
        rotate.setByAngle(360);
        rotate.setCycleCount(RotateTransition.INDEFINITE);
        rotate.play();
        
        // 文本
        Label label = new Label("加载中...");
        label.setStyle("-fx-font-size: 18px;");
        
        // 淡入淡出动画
        FadeTransition fade = new FadeTransition(Duration.seconds(1), label);
        fade.setFromValue(1.0);
        fade.setToValue(0.3);
        fade.setCycleCount(FadeTransition.INDEFINITE);
        fade.setAutoReverse(true);
        fade.play();
        
        root.getChildren().addAll(progress, label);
        
        Scene scene = new Scene(root, 400, 300);
        primaryStage.setTitle("加载动画");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 14.10 动画控制

```java
// 播放控制
animation.play();           // 播放
animation.pause();          // 暂停
animation.stop();           // 停止
animation.playFromStart();  // 从头播放

// 状态查询
Animation.Status status = animation.getStatus();
// RUNNING, PAUSED, STOPPED

// 速率控制
animation.setRate(2.0);  // 2倍速
animation.setRate(0.5);  // 0.5倍速

// 延迟
animation.setDelay(Duration.seconds(1));

// 监听
animation.setOnFinished(e -> {
    System.out.println("动画完成");
});
```
