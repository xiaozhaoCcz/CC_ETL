# 15-文件和数据操作

## 15.1 FileChooser（文件选择器）

FileChooser用于选择文件进行打开或保存。

### 15.1.1 打开文件

```java
package com.example.javafx.file;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.TextArea;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.io.*;

public class FileChooserDemo extends Application {

    private TextArea textArea;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 文本区域
        textArea = new TextArea();
        textArea.setWrapText(true);
        root.setCenter(textArea);
        
        // 按钮栏
        HBox buttonBar = new HBox(10);
        buttonBar.setPadding(new Insets(10, 0, 0, 0));
        
        Button openBtn = new Button("打开文件");
        openBtn.setOnAction(e -> openFile(primaryStage));
        
        Button saveBtn = new Button("保存文件");
        saveBtn.setOnAction(e -> saveFile(primaryStage));
        
        buttonBar.getChildren().addAll(openBtn, saveBtn);
        root.setBottom(buttonBar);
        
        Scene scene = new Scene(root, 600, 400);
        primaryStage.setTitle("文件选择器示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void openFile(Stage stage) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("打开文件");
        
        // 设置初始目录
        fileChooser.setInitialDirectory(new File(System.getProperty("user.home")));
        
        // 设置文件过滤器
        fileChooser.getExtensionFilters().addAll(
            new FileChooser.ExtensionFilter("文本文件", "*.txt"),
            new FileChooser.ExtensionFilter("Java文件", "*.java"),
            new FileChooser.ExtensionFilter("所有文件", "*.*")
        );
        
        File file = fileChooser.showOpenDialog(stage);
        
        if (file != null) {
            try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
                StringBuilder content = new StringBuilder();
                String line;
                while ((line = reader.readLine()) != null) {
                    content.append(line).append("\n");
                }
                textArea.setText(content.toString());
                stage.setTitle("文件选择器 - " + file.getName());
            } catch (IOException e) {
                showError("读取文件失败: " + e.getMessage());
            }
        }
    }

    private void saveFile(Stage stage) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("保存文件");
        fileChooser.setInitialFileName("untitled.txt");
        
        fileChooser.getExtensionFilters().addAll(
            new FileChooser.ExtensionFilter("文本文件", "*.txt"),
            new FileChooser.ExtensionFilter("所有文件", "*.*")
        );
        
        File file = fileChooser.showSaveDialog(stage);
        
        if (file != null) {
            try (BufferedWriter writer = new BufferedWriter(new FileWriter(file))) {
                writer.write(textArea.getText());
                stage.setTitle("文件选择器 - " + file.getName());
            } catch (IOException e) {
                showError("保存文件失败: " + e.getMessage());
            }
        }
    }

    private void showError(String message) {
        javafx.scene.control.Alert alert = new javafx.scene.control.Alert(
            javafx.scene.control.Alert.AlertType.ERROR
        );
        alert.setTitle("错误");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 15.2 DirectoryChooser（目录选择器）

```java
package com.example.javafx.file;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.ListView;
import javafx.scene.layout.BorderPane;
import javafx.stage.DirectoryChooser;
import javafx.stage.Stage;

import java.io.File;

public class DirectoryChooserDemo extends Application {

    private ListView<String> listView;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 文件列表
        listView = new ListView<>();
        root.setCenter(listView);
        
        // 选择目录按钮
        Button chooseBtn = new Button("选择目录");
        chooseBtn.setOnAction(e -> chooseDirectory(primaryStage));
        BorderPane.setMargin(chooseBtn, new Insets(10, 0, 0, 0));
        root.setBottom(chooseBtn);
        
        Scene scene = new Scene(root, 500, 400);
        primaryStage.setTitle("目录选择器示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void chooseDirectory(Stage stage) {
        DirectoryChooser directoryChooser = new DirectoryChooser();
        directoryChooser.setTitle("选择目录");
        directoryChooser.setInitialDirectory(
            new File(System.getProperty("user.home"))
        );
        
        File directory = directoryChooser.showDialog(stage);
        
        if (directory != null) {
            listView.getItems().clear();
            listView.getItems().add("目录: " + directory.getAbsolutePath());
            listView.getItems().add("---文件列表---");
            
            File[] files = directory.listFiles();
            if (files != null) {
                for (File file : files) {
                    String type = file.isDirectory() ? "[DIR] " : "[FILE] ";
                    listView.getItems().add(type + file.getName());
                }
            }
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 15.3 Properties配置文件

```java
package com.example.javafx.file;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.GridPane;
import javafx.stage.Stage;

import java.io.*;
import java.util.Properties;

public class PropertiesDemo extends Application {

    private static final String CONFIG_FILE = "app.properties";
    private TextField hostField;
    private TextField portField;
    private TextField usernameField;
    private PasswordField passwordField;

    @Override
    public void start(Stage primaryStage) {
        GridPane grid = new GridPane();
        grid.setHgap(10);
        grid.setVgap(10);
        grid.setPadding(new Insets(20));
        
        Label titleLabel = new Label("应用配置");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        grid.add(titleLabel, 0, 0, 2, 1);
        
        // 配置项
        grid.add(new Label("服务器:"), 0, 1);
        hostField = new TextField();
        grid.add(hostField, 1, 1);
        
        grid.add(new Label("端口:"), 0, 2);
        portField = new TextField();
        grid.add(portField, 1, 2);
        
        grid.add(new Label("用户名:"), 0, 3);
        usernameField = new TextField();
        grid.add(usernameField, 1, 3);
        
        grid.add(new Label("密码:"), 0, 4);
        passwordField = new PasswordField();
        grid.add(passwordField, 1, 4);
        
        // 按钮
        Button loadBtn = new Button("加载配置");
        loadBtn.setOnAction(e -> loadConfig());
        
        Button saveBtn = new Button("保存配置");
        saveBtn.setOnAction(e -> saveConfig());
        
        javafx.scene.layout.HBox buttonBox = new javafx.scene.layout.HBox(10);
        buttonBox.getChildren().addAll(loadBtn, saveBtn);
        grid.add(buttonBox, 0, 5, 2, 1);
        
        Scene scene = new Scene(grid, 400, 300);
        primaryStage.setTitle("配置文件管理");
        primaryStage.setScene(scene);
        primaryStage.show();
        
        // 自动加载配置
        loadConfig();
    }

    private void loadConfig() {
        Properties props = new Properties();
        
        try (InputStream input = new FileInputStream(CONFIG_FILE)) {
            props.load(input);
            
            hostField.setText(props.getProperty("host", "localhost"));
            portField.setText(props.getProperty("port", "8080"));
            usernameField.setText(props.getProperty("username", ""));
            passwordField.setText(props.getProperty("password", ""));
            
            showInfo("配置加载成功");
        } catch (IOException e) {
            System.out.println("配置文件不存在，使用默认值");
        }
    }

    private void saveConfig() {
        Properties props = new Properties();
        props.setProperty("host", hostField.getText());
        props.setProperty("port", portField.getText());
        props.setProperty("username", usernameField.getText());
        props.setProperty("password", passwordField.getText());
        
        try (OutputStream output = new FileOutputStream(CONFIG_FILE)) {
            props.store(output, "Application Configuration");
            showInfo("配置保存成功");
        } catch (IOException e) {
            showError("保存配置失败: " + e.getMessage());
        }
    }

    private void showInfo(String message) {
        Alert alert = new Alert(Alert.AlertType.INFORMATION);
        alert.setTitle("提示");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    private void showError(String message) {
        Alert alert = new Alert(Alert.AlertType.ERROR);
        alert.setTitle("错误");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 15.4 JSON数据处理

使用Gson库处理JSON数据。

**pom.xml添加依赖**：
```xml
<dependency>
    <groupId>com.google.code.gson</groupId>
    <artifactId>gson</artifactId>
    <version>2.10.1</version>
</dependency>
```

```java
package com.example.javafx.file;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

import java.io.*;
import java.util.ArrayList;
import java.util.List;

public class JsonDemo extends Application {

    private static final String JSON_FILE = "students.json";
    private TextArea textArea;
    private Gson gson;

    // 学生类
    static class Student {
        private String name;
        private int age;
        private String major;

        public Student(String name, int age, String major) {
            this.name = name;
            this.age = age;
            this.major = major;
        }

        @Override
        public String toString() {
            return String.format("姓名: %s, 年龄: %d, 专业: %s", name, age, major);
        }
    }

    @Override
    public void start(Stage primaryStage) {
        gson = new GsonBuilder().setPrettyPrinting().create();
        
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 文本区域
        textArea = new TextArea();
        textArea.setWrapText(true);
        root.setCenter(textArea);
        
        // 按钮栏
        HBox buttonBar = new HBox(10);
        buttonBar.setPadding(new Insets(10, 0, 0, 0));
        
        Button createBtn = new Button("创建示例数据");
        createBtn.setOnAction(e -> createSampleData());
        
        Button saveBtn = new Button("保存到JSON");
        saveBtn.setOnAction(e -> saveToJson());
        
        Button loadBtn = new Button("从JSON加载");
        loadBtn.setOnAction(e -> loadFromJson());
        
        buttonBar.getChildren().addAll(createBtn, saveBtn, loadBtn);
        root.setBottom(buttonBar);
        
        Scene scene = new Scene(root, 600, 400);
        primaryStage.setTitle("JSON数据处理");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void createSampleData() {
        List<Student> students = new ArrayList<>();
        students.add(new Student("张三", 20, "计算机科学"));
        students.add(new Student("李四", 21, "软件工程"));
        students.add(new Student("王五", 19, "人工智能"));
        
        String json = gson.toJson(students);
        textArea.setText(json);
    }

    private void saveToJson() {
        try (Writer writer = new FileWriter(JSON_FILE)) {
            writer.write(textArea.getText());
            showInfo("保存成功: " + JSON_FILE);
        } catch (IOException e) {
            showError("保存失败: " + e.getMessage());
        }
    }

    private void loadFromJson() {
        try (Reader reader = new FileReader(JSON_FILE)) {
            Student[] students = gson.fromJson(reader, Student[].class);
            
            StringBuilder sb = new StringBuilder();
            sb.append("加载的学生数据:\n\n");
            for (Student student : students) {
                sb.append(student.toString()).append("\n");
            }
            
            textArea.setText(sb.toString());
        } catch (IOException e) {
            showError("加载失败: " + e.getMessage());
        }
    }

    private void showInfo(String message) {
        Alert alert = new Alert(Alert.AlertType.INFORMATION);
        alert.setTitle("提示");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    private void showError(String message) {
        Alert alert = new Alert(Alert.AlertType.ERROR);
        alert.setTitle("错误");
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 15.5 数据库操作

使用JDBC连接数据库（以H2为例）。

**pom.xml添加依赖**：
```xml
<dependency>
    <groupId>com.h2database</groupId>
    <artifactId>h2</artifactId>
    <version>2.2.224</version>
</dependency>
```

```java
package com.example.javafx.file;

import javafx.application.Application;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.Stage;

import java.sql.*;

public class DatabaseDemo extends Application {

    private static final String DB_URL = "jdbc:h2:./testdb";
    private TableView<Student> tableView;
    private ObservableList<Student> data;

    public static class Student {
        private int id;
        private String name;
        private int age;

        public Student(int id, String name, int age) {
            this.id = id;
            this.name = name;
            this.age = age;
        }

        public int getId() { return id; }
        public String getName() { return name; }
        public int getAge() { return age; }
    }

    @Override
    public void start(Stage primaryStage) {
        data = FXCollections.observableArrayList();
        
        // 初始化数据库
        initDatabase();
        
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 创建表格
        tableView = new TableView<>(data);
        
        TableColumn<Student, Integer> idCol = new TableColumn<>("ID");
        idCol.setCellValueFactory(new PropertyValueFactory<>("id"));
        
        TableColumn<Student, String> nameCol = new TableColumn<>("姓名");
        nameCol.setCellValueFactory(new PropertyValueFactory<>("name"));
        
        TableColumn<Student, Integer> ageCol = new TableColumn<>("年龄");
        ageCol.setCellValueFactory(new PropertyValueFactory<>("age"));
        
        tableView.getColumns().addAll(idCol, nameCol, ageCol);
        root.setCenter(tableView);
        
        // 按钮栏
        HBox buttonBar = new HBox(10);
        buttonBar.setPadding(new Insets(10, 0, 0, 0));
        
        Button loadBtn = new Button("加载数据");
        loadBtn.setOnAction(e -> loadData());
        
        Button addBtn = new Button("添加记录");
        addBtn.setOnAction(e -> addRecord());
        
        buttonBar.getChildren().addAll(loadBtn, addBtn);
        root.setBottom(buttonBar);
        
        Scene scene = new Scene(root, 500, 400);
        primaryStage.setTitle("数据库操作示例");
        primaryStage.setScene(scene);
        primaryStage.show();
        
        // 初始加载数据
        loadData();
    }

    private void initDatabase() {
        try (Connection conn = DriverManager.getConnection(DB_URL);
             Statement stmt = conn.createStatement()) {
            
            String sql = "CREATE TABLE IF NOT EXISTS students (" +
                        "id INT PRIMARY KEY AUTO_INCREMENT, " +
                        "name VARCHAR(50), " +
                        "age INT)";
            stmt.execute(sql);
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    private void loadData() {
        data.clear();
        
        try (Connection conn = DriverManager.getConnection(DB_URL);
             Statement stmt = conn.createStatement();
             ResultSet rs = stmt.executeQuery("SELECT * FROM students")) {
            
            while (rs.next()) {
                data.add(new Student(
                    rs.getInt("id"),
                    rs.getString("name"),
                    rs.getInt("age")
                ));
            }
            
        } catch (SQLException e) {
            showError("加载数据失败: " + e.getMessage());
        }
    }

    private void addRecord() {
        TextInputDialog dialog = new TextInputDialog();
        dialog.setTitle("添加学生");
        dialog.setHeaderText("输入学生信息");
        dialog.setContentText("姓名,年龄:");
        
        dialog.showAndWait().ifPresent(input -> {
            String[] parts = input.split(",");
            if (parts.length == 2) {
                try (Connection conn = DriverManager.getConnection(DB_URL);
                     PreparedStatement pstmt = conn.prepareStatement(
                         "INSERT INTO students (name, age) VALUES (?, ?)")) {
                    
                    pstmt.setString(1, parts[0].trim());
                    pstmt.setInt(2, Integer.parseInt(parts[1].trim()));
                    pstmt.executeUpdate();
                    
                    loadData();
                    showInfo("添加成功");
                    
                } catch (SQLException | NumberFormatException e) {
                    showError("添加失败: " + e.getMessage());
                }
            }
        });
    }

    private void showInfo(String message) {
        Alert alert = new Alert(Alert.AlertType.INFORMATION);
        alert.setContentText(message);
        alert.showAndWait();
    }

    private void showError(String message) {
        Alert alert = new Alert(Alert.AlertType.ERROR);
        alert.setContentText(message);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```
