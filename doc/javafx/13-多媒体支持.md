# 13-多媒体支持

## 13.1 图像处理

JavaFX提供了强大的图像处理功能。

### 13.1.1 加载和显示图像

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.io.File;

public class ImageDemo extends Application {

    private ImageView imageView;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 创建ImageView
        imageView = new ImageView();
        imageView.setPreserveRatio(true);
        imageView.setFitWidth(600);
        imageView.setFitHeight(400);
        imageView.setStyle("-fx-border-color: #ddd; -fx-border-width: 1;");
        
        root.setCenter(imageView);
        
        // 按钮工具栏
        HBox toolBar = new HBox(10);
        toolBar.setAlignment(Pos.CENTER);
        toolBar.setPadding(new Insets(10));
        
        Button loadBtn = new Button("加载图片");
        loadBtn.setOnAction(e -> loadImage(primaryStage));
        
        Button rotateBtn = new Button("旋转90°");
        rotateBtn.setOnAction(e -> {
            imageView.setRotate(imageView.getRotate() + 90);
        });
        
        Button scaleBtn = new Button("放大");
        scaleBtn.setOnAction(e -> {
            imageView.setScaleX(imageView.getScaleX() * 1.2);
            imageView.setScaleY(imageView.getScaleY() * 1.2);
        });
        
        Button resetBtn = new Button("重置");
        resetBtn.setOnAction(e -> {
            imageView.setRotate(0);
            imageView.setScaleX(1);
            imageView.setScaleY(1);
        });
        
        toolBar.getChildren().addAll(loadBtn, rotateBtn, scaleBtn, resetBtn);
        root.setBottom(toolBar);
        
        Scene scene = new Scene(root, 800, 600);
        primaryStage.setTitle("图像查看器");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void loadImage(Stage stage) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("选择图片");
        fileChooser.getExtensionFilters().addAll(
            new FileChooser.ExtensionFilter("图片文件", 
                "*.png", "*.jpg", "*.jpeg", "*.gif", "*.bmp")
        );
        
        File file = fileChooser.showOpenDialog(stage);
        if (file != null) {
            Image image = new Image(file.toURI().toString());
            imageView.setImage(image);
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

### 13.1.2 图像效果

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.Slider;
import javafx.scene.effect.*;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ImageEffectDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(10));
        
        // 加载示例图片
        Image image = new Image(getClass().getResourceAsStream("/images/sample.jpg"));
        ImageView imageView = new ImageView(image);
        imageView.setFitWidth(600);
        imageView.setPreserveRatio(true);
        
        root.setCenter(imageView);
        
        // 效果控制面板
        VBox controlPanel = new VBox(15);
        controlPanel.setPadding(new Insets(10));
        controlPanel.setPrefWidth(200);
        
        // 模糊效果
        Button blurBtn = new Button("模糊效果");
        blurBtn.setPrefWidth(180);
        blurBtn.setOnAction(e -> {
            GaussianBlur blur = new GaussianBlur();
            blur.setRadius(10);
            imageView.setEffect(blur);
        });
        
        // 阴影效果
        Button shadowBtn = new Button("阴影效果");
        shadowBtn.setPrefWidth(180);
        shadowBtn.setOnAction(e -> {
            DropShadow shadow = new DropShadow();
            shadow.setRadius(15);
            shadow.setOffsetX(5);
            shadow.setOffsetY(5);
            imageView.setEffect(shadow);
        });
        
        // 反射效果
        Button reflectionBtn = new Button("反射效果");
        reflectionBtn.setPrefWidth(180);
        reflectionBtn.setOnAction(e -> {
            Reflection reflection = new Reflection();
            reflection.setFraction(0.6);
            imageView.setEffect(reflection);
        });
        
        // 发光效果
        Button glowBtn = new Button("发光效果");
        glowBtn.setPrefWidth(180);
        glowBtn.setOnAction(e -> {
            Glow glow = new Glow();
            glow.setLevel(0.9);
            imageView.setEffect(glow);
        });
        
        // 亮度调节
        Label brightnessLabel = new Label("亮度调节");
        Slider brightnessSlider = new Slider(-1, 1, 0);
        brightnessSlider.setShowTickLabels(true);
        brightnessSlider.valueProperty().addListener((obs, oldVal, newVal) -> {
            ColorAdjust colorAdjust = new ColorAdjust();
            colorAdjust.setBrightness(newVal.doubleValue());
            imageView.setEffect(colorAdjust);
        });
        
        // 清除效果
        Button clearBtn = new Button("清除效果");
        clearBtn.setPrefWidth(180);
        clearBtn.setOnAction(e -> imageView.setEffect(null));
        
        controlPanel.getChildren().addAll(
            blurBtn, shadowBtn, reflectionBtn, glowBtn,
            brightnessLabel, brightnessSlider, clearBtn
        );
        
        root.setRight(controlPanel);
        
        Scene scene = new Scene(root, 850, 600);
        primaryStage.setTitle("图像效果演示");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 13.2 音频播放

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.control.Slider;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.scene.layout.VBox;
import javafx.scene.media.Media;
import javafx.scene.media.MediaPlayer;
import javafx.stage.FileChooser;
import javafx.stage.Stage;
import javafx.util.Duration;

import java.io.File;

public class AudioPlayerDemo extends Application {

    private MediaPlayer mediaPlayer;
    private Label statusLabel;
    private Label timeLabel;
    private Slider progressSlider;
    private Slider volumeSlider;
    private boolean isPlaying = false;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        root.setPadding(new Insets(20));
        
        // 顶部信息
        VBox topBox = new VBox(10);
        topBox.setAlignment(Pos.CENTER);
        
        Label titleLabel = new Label("音频播放器");
        titleLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold;");
        
        statusLabel = new Label("请加载音频文件");
        statusLabel.setStyle("-fx-font-size: 14px;");
        
        topBox.getChildren().addAll(titleLabel, statusLabel);
        root.setTop(topBox);
        
        // 中心控制区
        VBox centerBox = new VBox(20);
        centerBox.setAlignment(Pos.CENTER);
        centerBox.setPadding(new Insets(30));
        
        // 进度条
        HBox progressBox = new HBox(10);
        progressBox.setAlignment(Pos.CENTER);
        
        timeLabel = new Label("00:00 / 00:00");
        progressSlider = new Slider();
        progressSlider.setPrefWidth(400);
        progressSlider.setDisable(true);
        
        progressBox.getChildren().addAll(timeLabel, progressSlider);
        
        // 音量控制
        HBox volumeBox = new HBox(10);
        volumeBox.setAlignment(Pos.CENTER);
        
        Label volumeLabel = new Label("音量:");
        volumeSlider = new Slider(0, 1, 0.5);
        volumeSlider.setPrefWidth(200);
        
        volumeBox.getChildren().addAll(volumeLabel, volumeSlider);
        
        centerBox.getChildren().addAll(progressBox, volumeBox);
        root.setCenter(centerBox);
        
        // 底部按钮
        HBox buttonBox = new HBox(15);
        buttonBox.setAlignment(Pos.CENTER);
        buttonBox.setPadding(new Insets(20));
        
        Button loadBtn = new Button("📁 加载");
        Button playBtn = new Button("▶ 播放");
        Button pauseBtn = new Button("⏸ 暂停");
        Button stopBtn = new Button("⏹ 停止");
        
        playBtn.setDisable(true);
        pauseBtn.setDisable(true);
        stopBtn.setDisable(true);
        
        loadBtn.setOnAction(e -> loadAudio(primaryStage, playBtn, pauseBtn, stopBtn));
        playBtn.setOnAction(e -> play(playBtn, pauseBtn));
        pauseBtn.setOnAction(e -> pause(playBtn, pauseBtn));
        stopBtn.setOnAction(e -> stop(playBtn, pauseBtn));
        
        buttonBox.getChildren().addAll(loadBtn, playBtn, pauseBtn, stopBtn);
        root.setBottom(buttonBox);
        
        Scene scene = new Scene(root, 600, 400);
        primaryStage.setTitle("音频播放器");
        primaryStage.setScene(scene);
        primaryStage.setOnCloseRequest(e -> {
            if (mediaPlayer != null) {
                mediaPlayer.dispose();
            }
        });
        primaryStage.show();
    }

    private void loadAudio(Stage stage, Button playBtn, Button pauseBtn, Button stopBtn) {
        FileChooser fileChooser = new FileChooser();
        fileChooser.setTitle("选择音频文件");
        fileChooser.getExtensionFilters().add(
            new FileChooser.ExtensionFilter("音频文件", "*.mp3", "*.wav", "*.m4a")
        );
        
        File file = fileChooser.showOpenDialog(stage);
        if (file != null) {
            if (mediaPlayer != null) {
                mediaPlayer.dispose();
            }
            
            Media media = new Media(file.toURI().toString());
            mediaPlayer = new MediaPlayer(media);
            
            // 绑定音量
            mediaPlayer.volumeProperty().bind(volumeSlider.valueProperty());
            
            // 更新进度
            mediaPlayer.currentTimeProperty().addListener((obs, oldTime, newTime) -> {
                if (!progressSlider.isValueChanging()) {
                    progressSlider.setValue(newTime.toSeconds());
                }
                updateTimeLabel();
            });
            
            // 媒体就绪
            mediaPlayer.setOnReady(() -> {
                Duration duration = mediaPlayer.getTotalDuration();
                progressSlider.setMax(duration.toSeconds());
                progressSlider.setDisable(false);
                statusLabel.setText("已加载: " + file.getName());
                playBtn.setDisable(false);
                stopBtn.setDisable(false);
            });
            
            // 进度条拖动
            progressSlider.valueChangingProperty().addListener((obs, wasChanging, isChanging) -> {
                if (!isChanging) {
                    mediaPlayer.seek(Duration.seconds(progressSlider.getValue()));
                }
            });
            
            // 播放结束
            mediaPlayer.setOnEndOfMedia(() -> {
                stop(playBtn, pauseBtn);
                statusLabel.setText("播放完成");
            });
        }
    }

    private void play(Button playBtn, Button pauseBtn) {
        if (mediaPlayer != null) {
            mediaPlayer.play();
            isPlaying = true;
            playBtn.setDisable(true);
            pauseBtn.setDisable(false);
            statusLabel.setText("正在播放");
        }
    }

    private void pause(Button playBtn, Button pauseBtn) {
        if (mediaPlayer != null) {
            mediaPlayer.pause();
            isPlaying = false;
            playBtn.setDisable(false);
            pauseBtn.setDisable(true);
            statusLabel.setText("已暂停");
        }
    }

    private void stop(Button playBtn, Button pauseBtn) {
        if (mediaPlayer != null) {
            mediaPlayer.stop();
            isPlaying = false;
            playBtn.setDisable(false);
            pauseBtn.setDisable(true);
            progressSlider.setValue(0);
            statusLabel.setText("已停止");
        }
    }

    private void updateTimeLabel() {
        if (mediaPlayer != null) {
            Duration current = mediaPlayer.getCurrentTime();
            Duration total = mediaPlayer.getTotalDuration();
            timeLabel.setText(formatTime(current) + " / " + formatTime(total));
        }
    }

    private String formatTime(Duration duration) {
        int minutes = (int) duration.toMinutes();
        int seconds = (int) duration.toSeconds() % 60;
        return String.format("%02d:%02d", minutes, seconds);
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 13.3 视频播放

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Slider;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.scene.media.Media;
import javafx.scene.media.MediaPlayer;
import javafx.scene.media.MediaView;
import javafx.stage.FileChooser;
import javafx.stage.Stage;

import java.io.File;

public class VideoPlayerDemo extends Application {

    private MediaPlayer mediaPlayer;
    private MediaView mediaView;

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        
        // 视频显示区域
        mediaView = new MediaView();
        mediaView.setPreserveRatio(true);
        root.setCenter(mediaView);
        
        // 控制栏
        HBox controlBar = new HBox(10);
        controlBar.setAlignment(Pos.CENTER);
        controlBar.setPadding(new Insets(10));
        controlBar.setStyle("-fx-background-color: #333;");
        
        Button loadBtn = new Button("打开");
        Button playBtn = new Button("播放");
        Button pauseBtn = new Button("暂停");
        Button stopBtn = new Button("停止");
        
        Slider volumeSlider = new Slider(0, 1, 0.5);
        volumeSlider.setPrefWidth(100);
        
        playBtn.setDisable(true);
        pauseBtn.setDisable(true);
        stopBtn.setDisable(true);
        
        loadBtn.setOnAction(e -> {
            FileChooser fileChooser = new FileChooser();
            fileChooser.getExtensionFilters().add(
                new FileChooser.ExtensionFilter("视频文件", "*.mp4", "*.avi", "*.mov")
            );
            File file = fileChooser.showOpenDialog(primaryStage);
            
            if (file != null) {
                if (mediaPlayer != null) {
                    mediaPlayer.dispose();
                }
                
                Media media = new Media(file.toURI().toString());
                mediaPlayer = new MediaPlayer(media);
                mediaView.setMediaPlayer(mediaPlayer);
                
                mediaPlayer.volumeProperty().bind(volumeSlider.valueProperty());
                
                playBtn.setDisable(false);
                pauseBtn.setDisable(false);
                stopBtn.setDisable(false);
            }
        });
        
        playBtn.setOnAction(e -> mediaPlayer.play());
        pauseBtn.setOnAction(e -> mediaPlayer.pause());
        stopBtn.setOnAction(e -> mediaPlayer.stop());
        
        controlBar.getChildren().addAll(
            loadBtn, playBtn, pauseBtn, stopBtn,
            new javafx.scene.control.Label("音量:"), volumeSlider
        );
        
        root.setBottom(controlBar);
        
        Scene scene = new Scene(root, 800, 600);
        primaryStage.setTitle("视频播放器");
        primaryStage.setScene(scene);
        primaryStage.setOnCloseRequest(e -> {
            if (mediaPlayer != null) {
                mediaPlayer.dispose();
            }
        });
        primaryStage.show();
        
        // 绑定视频大小到窗口大小
        mediaView.fitWidthProperty().bind(scene.widthProperty());
        mediaView.fitHeightProperty().bind(scene.heightProperty().subtract(60));
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 13.4 Canvas绘图

Canvas提供了强大的2D绘图API。

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.canvas.Canvas;
import javafx.scene.canvas.GraphicsContext;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.stage.Stage;

public class CanvasDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        Canvas canvas = new Canvas(800, 600);
        GraphicsContext gc = canvas.getGraphicsContext2D();
        
        // 绘制背景
        gc.setFill(Color.LIGHTGRAY);
        gc.fillRect(0, 0, 800, 600);
        
        // 绘制矩形
        gc.setFill(Color.BLUE);
        gc.fillRect(50, 50, 200, 150);
        
        // 绘制圆形
        gc.setFill(Color.RED);
        gc.fillOval(300, 50, 150, 150);
        
        // 绘制文本
        gc.setFill(Color.BLACK);
        gc.setFont(javafx.scene.text.Font.font(24));
        gc.fillText("Canvas 绘图示例", 500, 100);
        
        // 绘制线条
        gc.setStroke(Color.GREEN);
        gc.setLineWidth(5);
        gc.strokeLine(50, 250, 750, 250);
        
        // 绘制路径
        gc.beginPath();
        gc.moveTo(100, 350);
        gc.lineTo(200, 450);
        gc.lineTo(100, 550);
        gc.closePath();
        gc.setFill(Color.ORANGE);
        gc.fill();
        
        StackPane root = new StackPane(canvas);
        Scene scene = new Scene(root, 800, 600);
        primaryStage.setTitle("Canvas 绘图");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 13.5 WebView

WebView可以在JavaFX中嵌入网页。

```java
package com.example.javafx.media;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.TextField;
import javafx.scene.layout.BorderPane;
import javafx.scene.layout.HBox;
import javafx.scene.web.WebEngine;
import javafx.scene.web.WebView;
import javafx.stage.Stage;

public class WebViewDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        BorderPane root = new BorderPane();
        
        // 创建WebView
        WebView webView = new WebView();
        WebEngine webEngine = webView.getEngine();
        
        // 地址栏
        HBox toolbar = new HBox(10);
        toolbar.setPadding(new Insets(10));
        
        TextField urlField = new TextField("https://www.example.com");
        urlField.setPrefWidth(600);
        
        Button goBtn = new Button("访问");
        Button backBtn = new Button("后退");
        Button forwardBtn = new Button("前进");
        Button refreshBtn = new Button("刷新");
        
        goBtn.setOnAction(e -> webEngine.load(urlField.getText()));
        backBtn.setOnAction(e -> webEngine.getHistory().go(-1));
        forwardBtn.setOnAction(e -> webEngine.getHistory().go(1));
        refreshBtn.setOnAction(e -> webEngine.reload());
        
        toolbar.getChildren().addAll(backBtn, forwardBtn, refreshBtn, urlField, goBtn);
        
        root.setTop(toolbar);
        root.setCenter(webView);
        
        // 加载初始页面
        webEngine.load("https://www.example.com");
        
        Scene scene = new Scene(root, 1000, 700);
        primaryStage.setTitle("内置浏览器");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```
