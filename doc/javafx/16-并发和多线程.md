# 16-并发和多线程

## 16.1 JavaFX线程模型

JavaFX应用程序采用单线程模型，所有UI操作必须在JavaFX Application Thread上执行。

### 16.1.1 线程规则

```
规则1: 只能在JavaFX Application Thread上访问活跃的Scene Graph
规则2: 耗时操作必须在后台线程执行
规则3: 使用Platform.runLater()从后台线程更新UI
```

### 16.1.2 线程问题示例

```java
package com.example.javafx.concurrency;

import javafx.application.Application;
import javafx.application.Platform;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ThreadProblemDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        Label statusLabel = new Label("状态: 就绪");
        
        // 错误示例：在JavaFX线程执行耗时操作
        Button wrongBtn = new Button("错误方式（UI会冻结）");
        wrongBtn.setOnAction(e -> {
            statusLabel.setText("状态: 处理中...");
            
            // 错误！在JavaFX线程执行耗时操作
            try {
                Thread.sleep(3000);  // 模拟耗时操作
            } catch (InterruptedException ex) {
                ex.printStackTrace();
            }
            
            statusLabel.setText("状态: 完成");
        });
        
        // 正确示例：在后台线程执行耗时操作
        Button correctBtn = new Button("正确方式（UI不会冻结）");
        correctBtn.setOnAction(e -> {
            statusLabel.setText("状态: 处理中...");
            
            // 正确！在后台线程执行
            new Thread(() -> {
                try {
                    Thread.sleep(3000);  // 模拟耗时操作
                    
                    // 使用Platform.runLater()更新UI
                    Platform.runLater(() -> {
                        statusLabel.setText("状态: 完成");
                    });
                } catch (InterruptedException ex) {
                    ex.printStackTrace();
                }
            }).start();
        });
        
        // 检查当前线程
        Button checkThreadBtn = new Button("检查当前线程");
        checkThreadBtn.setOnAction(e -> {
            System.out.println("当前线程: " + Thread.currentThread().getName());
            System.out.println("是否JavaFX线程: " + Platform.isFxApplicationThread());
        });
        
        root.getChildren().addAll(statusLabel, wrongBtn, correctBtn, checkThreadBtn);
        
        Scene scene = new Scene(root, 400, 250);
        primaryStage.setTitle("线程问题演示");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 16.2 Task类

Task是JavaFX提供的用于执行后台任务的类。

### 16.2.1 基本使用

```java
package com.example.javafx.concurrency;

import javafx.application.Application;
import javafx.concurrent.Task;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class TaskDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setPadding(new Insets(30));
        root.setAlignment(Pos.CENTER);
        
        Label titleLabel = new Label("Task 示例");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        
        // 进度条
        ProgressBar progressBar = new ProgressBar(0);
        progressBar.setPrefWidth(300);
        
        // 进度标签
        Label progressLabel = new Label("0%");
        
        // 消息标签
        Label messageLabel = new Label("");
        messageLabel.setStyle("-fx-text-fill: blue;");
        
        // 开始按钮
        Button startBtn = new Button("开始任务");
        startBtn.setOnAction(e -> {
            startBtn.setDisable(true);
            
            // 创建Task
            Task<String> task = new Task<String>() {
                @Override
                protected String call() throws Exception {
                    int max = 100;
                    
                    for (int i = 0; i <= max; i++) {
                        // 检查是否被取消
                        if (isCancelled()) {
                            updateMessage("任务已取消");
                            break;
                        }
                        
                        // 更新进度
                        updateProgress(i, max);
                        updateMessage("处理中: " + i + "%");
                        
                        // 模拟耗时操作
                        Thread.sleep(50);
                    }
                    
                    return "任务完成！";
                }
            };
            
            // 绑定属性
            progressBar.progressProperty().bind(task.progressProperty());
            progressLabel.textProperty().bind(
                task.progressProperty().multiply(100).asString("%.0f%%")
            );
            messageLabel.textProperty().bind(task.messageProperty());
            
            // 任务成功
            task.setOnSucceeded(event -> {
                startBtn.setDisable(false);
                String result = task.getValue();
                showAlert(Alert.AlertType.INFORMATION, "完成", result);
            });
            
            // 任务失败
            task.setOnFailed(event -> {
                startBtn.setDisable(false);
                Throwable exception = task.getException();
                showAlert(Alert.AlertType.ERROR, "错误", 
                         exception.getMessage());
            });
            
            // 任务取消
            task.setOnCancelled(event -> {
                startBtn.setDisable(false);
                showAlert(Alert.AlertType.WARNING, "取消", "任务已取消");
            });
            
            // 启动任务
            new Thread(task).start();
        });
        
        root.getChildren().addAll(titleLabel, progressBar, progressLabel, 
                                 messageLabel, startBtn);
        
        Scene scene = new Scene(root, 450, 300);
        primaryStage.setTitle("Task 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    private void showAlert(Alert.AlertType type, String title, String content) {
        Alert alert = new Alert(type);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(content);
        alert.showAndWait();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 16.3 Service类

Service是可重用的Task包装器，可以多次启动。

```java
package com.example.javafx.concurrency;

import javafx.application.Application;
import javafx.concurrent.Service;
import javafx.concurrent.Task;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.HBox;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class ServiceDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setPadding(new Insets(30));
        root.setAlignment(Pos.CENTER);
        
        Label titleLabel = new Label("Service 示例");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        
        ProgressBar progressBar = new ProgressBar(0);
        progressBar.setPrefWidth(300);
        
        Label statusLabel = new Label("就绪");
        Label messageLabel = new Label("");
        
        // 创建Service
        Service<String> service = new Service<String>() {
            @Override
            protected Task<String> createTask() {
                return new Task<String>() {
                    @Override
                    protected String call() throws Exception {
                        updateMessage("开始处理...");
                        
                        for (int i = 0; i <= 100; i++) {
                            if (isCancelled()) {
                                updateMessage("已取消");
                                break;
                            }
                            
                            updateProgress(i, 100);
                            updateMessage("处理中: " + i + "%");
                            Thread.sleep(30);
                        }
                        
                        return "处理完成！";
                    }
                };
            }
        };
        
        // 绑定属性
        progressBar.progressProperty().bind(service.progressProperty());
        messageLabel.textProperty().bind(service.messageProperty());
        
        // 状态监听
        service.stateProperty().addListener((obs, oldState, newState) -> {
            statusLabel.setText("状态: " + newState);
        });
        
        // 成功处理
        service.setOnSucceeded(e -> {
            String result = service.getValue();
            System.out.println("成功: " + result);
        });
        
        // 失败处理
        service.setOnFailed(e -> {
            System.out.println("失败: " + service.getException().getMessage());
        });
        
        // 按钮
        HBox buttonBox = new HBox(10);
        buttonBox.setAlignment(Pos.CENTER);
        
        Button startBtn = new Button("启动");
        startBtn.setOnAction(e -> service.restart());
        startBtn.disableProperty().bind(service.runningProperty());
        
        Button cancelBtn = new Button("取消");
        cancelBtn.setOnAction(e -> service.cancel());
        cancelBtn.disableProperty().bind(service.runningProperty().not());
        
        Button resetBtn = new Button("重置");
        resetBtn.setOnAction(e -> service.reset());
        
        buttonBox.getChildren().addAll(startBtn, cancelBtn, resetBtn);
        
        root.getChildren().addAll(titleLabel, progressBar, messageLabel, 
                                 statusLabel, buttonBox);
        
        Scene scene = new Scene(root, 450, 300);
        primaryStage.setTitle("Service 示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 16.4 文件下载示例

```java
package com.example.javafx.concurrency;

import javafx.application.Application;
import javafx.concurrent.Task;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

import java.io.BufferedInputStream;
import java.io.FileOutputStream;
import java.io.IOException;
import java.net.URL;

public class DownloadDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        root.setAlignment(Pos.CENTER);
        
        Label titleLabel = new Label("文件下载器");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        
        TextField urlField = new TextField();
        urlField.setPromptText("输入文件URL");
        urlField.setPrefWidth(400);
        
        ProgressBar progressBar = new ProgressBar(0);
        progressBar.setPrefWidth(400);
        
        Label progressLabel = new Label("0%");
        Label speedLabel = new Label("");
        Label statusLabel = new Label("就绪");
        
        Button downloadBtn = new Button("开始下载");
        Button cancelBtn = new Button("取消");
        cancelBtn.setDisable(true);
        
        final Task<Void>[] currentTask = {null};
        
        downloadBtn.setOnAction(e -> {
            String url = urlField.getText();
            if (url.isEmpty()) {
                statusLabel.setText("请输入URL");
                return;
            }
            
            downloadBtn.setDisable(true);
            cancelBtn.setDisable(false);
            
            // 创建下载任务
            Task<Void> downloadTask = new Task<Void>() {
                @Override
                protected Void call() throws Exception {
                    String fileName = "download.tmp";
                    long totalBytes = 0;
                    long downloadedBytes = 0;
                    long startTime = System.currentTimeMillis();
                    
                    try (BufferedInputStream in = new BufferedInputStream(
                            new URL(url).openStream());
                         FileOutputStream fileOutputStream = new FileOutputStream(fileName)) {
                        
                        // 获取文件大小
                        totalBytes = in.available();
                        updateMessage("文件大小: " + formatSize(totalBytes));
                        
                        byte[] dataBuffer = new byte[1024];
                        int bytesRead;
                        
                        while ((bytesRead = in.read(dataBuffer, 0, 1024)) != -1) {
                            if (isCancelled()) {
                                updateMessage("下载已取消");
                                break;
                            }
                            
                            fileOutputStream.write(dataBuffer, 0, bytesRead);
                            downloadedBytes += bytesRead;
                            
                            // 更新进度
                            if (totalBytes > 0) {
                                updateProgress(downloadedBytes, totalBytes);
                            }
                            
                            // 计算速度
                            long elapsedTime = System.currentTimeMillis() - startTime;
                            if (elapsedTime > 0) {
                                long speed = downloadedBytes * 1000 / elapsedTime;
                                updateMessage("下载速度: " + formatSize(speed) + "/s");
                            }
                        }
                        
                        if (!isCancelled()) {
                            updateMessage("下载完成！");
                        }
                    }
                    
                    return null;
                }
                
                private String formatSize(long bytes) {
                    if (bytes < 1024) return bytes + " B";
                    if (bytes < 1024 * 1024) return String.format("%.2f KB", bytes / 1024.0);
                    if (bytes < 1024 * 1024 * 1024) return String.format("%.2f MB", bytes / (1024.0 * 1024));
                    return String.format("%.2f GB", bytes / (1024.0 * 1024 * 1024));
                }
            };
            
            currentTask[0] = downloadTask;
            
            // 绑定属性
            progressBar.progressProperty().bind(downloadTask.progressProperty());
            progressLabel.textProperty().bind(
                downloadTask.progressProperty().multiply(100).asString("%.0f%%")
            );
            statusLabel.textProperty().bind(downloadTask.messageProperty());
            
            // 完成处理
            downloadTask.setOnSucceeded(event -> {
                downloadBtn.setDisable(false);
                cancelBtn.setDisable(true);
            });
            
            downloadTask.setOnFailed(event -> {
                downloadBtn.setDisable(false);
                cancelBtn.setDisable(true);
                statusLabel.setText("下载失败: " + downloadTask.getException().getMessage());
            });
            
            downloadTask.setOnCancelled(event -> {
                downloadBtn.setDisable(false);
                cancelBtn.setDisable(true);
            });
            
            // 启动任务
            new Thread(downloadTask).start();
        });
        
        cancelBtn.setOnAction(e -> {
            if (currentTask[0] != null) {
                currentTask[0].cancel();
            }
        });
        
        root.getChildren().addAll(
            titleLabel, urlField, progressBar, progressLabel,
            statusLabel, new HBox(10, downloadBtn, cancelBtn)
        );
        
        Scene scene = new Scene(root, 500, 300);
        primaryStage.setTitle("文件下载器");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 16.5 最佳实践

### 16.5.1 使用ExecutorService

```java
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class TaskManager {
    private static final ExecutorService executor = 
        Executors.newFixedThreadPool(4);
    
    public static void executeTask(Task<?> task) {
        executor.submit(task);
    }
    
    public static void shutdown() {
        executor.shutdown();
    }
}
```

### 16.5.2 避免内存泄漏

```java
// 及时解除绑定
progressBar.progressProperty().unbind();

// 清理Task引用
task = null;

// Service重用
service.reset();  // 重置状态
service.restart(); // 重启服务
```

### 16.5.3 异常处理

```java
Task<String> task = new Task<String>() {
    @Override
    protected String call() throws Exception {
        try {
            // 任务逻辑
            return performOperation();
        } catch (Exception e) {
            updateMessage("错误: " + e.getMessage());
            throw e;  // 重新抛出，触发onFailed
        }
    }
};

task.setOnFailed(e -> {
    Throwable exception = task.getException();
    showErrorDialog("任务失败", exception.getMessage());
});
```
