# 08-事件处理

## 8.1 事件处理概述

JavaFX使用事件驱动模型来处理用户交互。事件（Event）表示发生的动作，如鼠标点击、键盘输入等。

### 8.1.1 事件处理流程

```
用户操作 → 生成事件 → 事件分发 → 事件处理器处理
```

## 8.2 事件类型

JavaFX提供了丰富的事件类型：

| 事件类别 | 事件类型 | 描述 |
|---------|---------|------|
| 鼠标事件 | MouseEvent | 点击、移动、拖拽等 |
| 键盘事件 | KeyEvent | 按键按下、释放 |
| 触摸事件 | TouchEvent | 触摸屏交互 |
| 窗口事件 | WindowEvent | 窗口显示、隐藏、关闭 |
| 动作事件 | ActionEvent | 按钮点击等 |

## 8.3 鼠标事件

### 8.3.1 基本鼠标事件

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.layout.StackPane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Rectangle;
import javafx.stage.Stage;

public class MouseEventDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        StackPane root = new StackPane();
        root.setPadding(new Insets(20));
        
        Rectangle rect = new Rectangle(200, 150);
        rect.setFill(Color.LIGHTBLUE);
        rect.setStroke(Color.BLUE);
        rect.setStrokeWidth(2);
        
        Label infoLabel = new Label("将鼠标移到矩形上");
        infoLabel.setStyle("-fx-font-size: 14px;");
        StackPane.setMargin(infoLabel, new Insets(0, 0, 200, 0));
        
        // 鼠标点击
        rect.setOnMouseClicked(e -> {
            String info = String.format("点击: 按钮=%s, 位置=(%.0f, %.0f), 点击次数=%d",
                e.getButton(), e.getX(), e.getY(), e.getClickCount());
            infoLabel.setText(info);
            System.out.println(info);
        });
        
        // 鼠标进入
        rect.setOnMouseEntered(e -> {
            rect.setFill(Color.LIGHTGREEN);
            infoLabel.setText("鼠标进入矩形");
        });
        
        // 鼠标离开
        rect.setOnMouseExited(e -> {
            rect.setFill(Color.LIGHTBLUE);
            infoLabel.setText("鼠标离开矩形");
        });
        
        // 鼠标移动
        rect.setOnMouseMoved(e -> {
            String info = String.format("鼠标位置: (%.0f, %.0f)", e.getX(), e.getY());
            infoLabel.setText(info);
        });
        
        // 鼠标按下
        rect.setOnMousePressed(e -> {
            rect.setFill(Color.YELLOW);
            System.out.println("鼠标按下");
        });
        
        // 鼠标释放
        rect.setOnMouseReleased(e -> {
            rect.setFill(Color.LIGHTGREEN);
            System.out.println("鼠标释放");
        });
        
        root.getChildren().addAll(rect, infoLabel);
        
        Scene scene = new Scene(root, 400, 350);
        primaryStage.setTitle("鼠标事件示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

### 8.3.2 鼠标拖拽

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.layout.Pane;
import javafx.scene.paint.Color;
import javafx.scene.shape.Circle;
import javafx.stage.Stage;

public class MouseDragDemo extends Application {

    private double orgSceneX, orgSceneY;
    private double orgTranslateX, orgTranslateY;

    @Override
    public void start(Stage primaryStage) {
        Pane root = new Pane();
        
        Label infoLabel = new Label("拖拽圆形移动它");
        infoLabel.setLayoutX(10);
        infoLabel.setLayoutY(10);
        infoLabel.setStyle("-fx-font-size: 14px; -fx-font-weight: bold;");
        
        Circle circle = new Circle(200, 200, 50);
        circle.setFill(Color.CORAL);
        circle.setStroke(Color.RED);
        circle.setStrokeWidth(2);
        
        // 鼠标按下 - 记录初始位置
        circle.setOnMousePressed(e -> {
            orgSceneX = e.getSceneX();
            orgSceneY = e.getSceneY();
            orgTranslateX = circle.getTranslateX();
            orgTranslateY = circle.getTranslateY();
            circle.setFill(Color.YELLOW);
        });
        
        // 鼠标拖拽 - 移动圆形
        circle.setOnMouseDragged(e -> {
            double offsetX = e.getSceneX() - orgSceneX;
            double offsetY = e.getSceneY() - orgSceneY;
            double newTranslateX = orgTranslateX + offsetX;
            double newTranslateY = orgTranslateY + offsetY;
            
            circle.setTranslateX(newTranslateX);
            circle.setTranslateY(newTranslateY);
            
            String info = String.format("位置: (%.0f, %.0f)", 
                circle.getCenterX() + newTranslateX,
                circle.getCenterY() + newTranslateY);
            infoLabel.setText(info);
        });
        
        // 鼠标释放
        circle.setOnMouseReleased(e -> {
            circle.setFill(Color.CORAL);
        });
        
        root.getChildren().addAll(circle, infoLabel);
        
        Scene scene = new Scene(root, 500, 400);
        primaryStage.setTitle("鼠标拖拽示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 8.4 键盘事件

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Label;
import javafx.scene.control.TextArea;
import javafx.scene.input.KeyCode;
import javafx.scene.input.KeyEvent;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class KeyEventDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        Label titleLabel = new Label("键盘事件演示");
        titleLabel.setStyle("-fx-font-size: 18px; -fx-font-weight: bold;");
        
        Label infoLabel = new Label("在文本框中输入，查看键盘事件信息");
        
        TextArea textArea = new TextArea();
        textArea.setPrefRowCount(5);
        textArea.setPromptText("在这里输入...");
        
        Label eventLabel = new Label();
        eventLabel.setStyle("-fx-font-size: 12px; -fx-text-fill: blue;");
        
        // 键盘按下事件
        textArea.setOnKeyPressed(e -> {
            String info = String.format("按下: %s (Code: %s) - Ctrl:%b Alt:%b Shift:%b",
                e.getText(), e.getCode(),
                e.isControlDown(), e.isAltDown(), e.isShiftDown());
            eventLabel.setText(info);
            System.out.println(info);
            
            // 特殊按键处理
            if (e.getCode() == KeyCode.ESCAPE) {
                System.out.println("按下了ESC键");
            } else if (e.isControlDown() && e.getCode() == KeyCode.S) {
                System.out.println("触发保存快捷键 Ctrl+S");
                e.consume();  // 阻止事件继续传播
            }
        });
        
        // 键盘释放事件
        textArea.setOnKeyReleased(e -> {
            System.out.println("释放: " + e.getCode());
        });
        
        // 键入字符事件（只有可打印字符）
        textArea.setOnKeyTyped(e -> {
            System.out.println("输入字符: " + e.getCharacter());
        });
        
        root.getChildren().addAll(titleLabel, infoLabel, textArea, eventLabel);
        
        Scene scene = new Scene(root, 500, 350);
        
        // Scene级别的键盘事件
        scene.setOnKeyPressed(e -> {
            if (e.getCode() == KeyCode.F1) {
                System.out.println("按下F1 - 显示帮助");
            }
        });
        
        primaryStage.setTitle("键盘事件示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

### 8.5.2 实现EventHandler接口

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.event.ActionEvent;
import javafx.event.EventHandler;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class EventHandlerDemo extends Application {

    private int count = 0;

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        root.setAlignment(Pos.CENTER);
        
        Label countLabel = new Label("计数: 0");
        countLabel.setStyle("-fx-font-size: 16px;");
        
        // 方式1: Lambda表达式
        Button lambdaBtn = new Button("Lambda方式");
        lambdaBtn.setOnAction(e -> {
            count++;
            countLabel.setText("计数: " + count);
        });
        
        // 方式2: 匿名内部类
        Button anonymousBtn = new Button("匿名内部类方式");
        anonymousBtn.setOnAction(new EventHandler<ActionEvent>() {
            @Override
            public void handle(ActionEvent event) {
                count++;
                countLabel.setText("计数: " + count);
            }
        });
        
        // 方式3: 独立的处理器类
        Button handlerBtn = new Button("独立处理器方式");
        handlerBtn.setOnAction(new MyEventHandler(countLabel));
        
        // 方式4: 方法引用
        Button methodRefBtn = new Button("方法引用方式");
        methodRefBtn.setOnAction(this::handleButtonClick);
        
        root.getChildren().addAll(countLabel, lambdaBtn, anonymousBtn, 
                                 handlerBtn, methodRefBtn);
        
        Scene scene = new Scene(root, 350, 300);
        primaryStage.setTitle("事件处理器示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    // 方法引用使用的方法
    private void handleButtonClick(ActionEvent event) {
        count++;
        System.out.println("方法引用处理，计数: " + count);
    }

    // 独立的事件处理器类
    static class MyEventHandler implements EventHandler<ActionEvent> {
        private Label label;
        private int count = 0;

        public MyEventHandler(Label label) {
            this.label = label;
        }

        @Override
        public void handle(ActionEvent event) {
            count++;
            label.setText("计数: " + count);
        }
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 8.6 事件传播机制

JavaFX的事件传播分为三个阶段：捕获阶段、目标阶段、冒泡阶段。

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.layout.HBox;
import javafx.scene.layout.StackPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class EventPropagationDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setPadding(new Insets(20));
        root.setStyle("-fx-border-color: red; -fx-border-width: 3;");
        
        HBox hbox = new HBox(15);
        hbox.setStyle("-fx-border-color: green; -fx-border-width: 3; -fx-padding: 15;");
        
        StackPane stack = new StackPane();
        stack.setStyle("-fx-border-color: blue; -fx-border-width: 3; -fx-padding: 15;");
        
        Button button = new Button("点击我");
        
        stack.getChildren().add(button);
        hbox.getChildren().add(stack);
        root.getChildren().add(hbox);
        
        // ===== 捕获阶段（从上到下） =====
        root.addEventFilter(javafx.scene.input.MouseEvent.MOUSE_CLICKED, e -> {
            System.out.println("1. 捕获阶段: VBox (红色)");
        });
        
        hbox.addEventFilter(javafx.scene.input.MouseEvent.MOUSE_CLICKED, e -> {
            System.out.println("2. 捕获阶段: HBox (绿色)");
        });
        
        stack.addEventFilter(javafx.scene.input.MouseEvent.MOUSE_CLICKED, e -> {
            System.out.println("3. 捕获阶段: StackPane (蓝色)");
        });
        
        // ===== 目标阶段 =====
        button.setOnMouseClicked(e -> {
            System.out.println("4. 目标阶段: Button");
        });
        
        // ===== 冒泡阶段（从下到上） =====
        stack.setOnMouseClicked(e -> {
            System.out.println("5. 冒泡阶段: StackPane (蓝色)");
        });
        
        hbox.setOnMouseClicked(e -> {
            System.out.println("6. 冒泡阶段: HBox (绿色)");
        });
        
        root.setOnMouseClicked(e -> {
            System.out.println("7. 冒泡阶段: VBox (红色)");
            System.out.println("---");
        });
        
        Scene scene = new Scene(root, 400, 250);
        primaryStage.setTitle("事件传播机制");
        primaryStage.setScene(scene);
        primaryStage.show();
        
        System.out.println("点击按钮查看事件传播顺序:");
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 8.7 自定义事件

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.event.Event;
import javafx.event.EventType;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class CustomEventDemo extends Application {

    // 定义自定义事件类型
    public static class ScoreEvent extends Event {
        public static final EventType<ScoreEvent> SCORE_CHANGED = 
            new EventType<>(Event.ANY, "SCORE_CHANGED");
        
        private int score;
        
        public ScoreEvent(int score) {
            super(SCORE_CHANGED);
            this.score = score;
        }
        
        public int getScore() {
            return score;
        }
    }

    private int score = 0;

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(20);
        root.setPadding(new Insets(30));
        root.setAlignment(Pos.CENTER);
        
        Label scoreLabel = new Label("分数: 0");
        scoreLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold;");
        
        Button addBtn = new Button("增加10分");
        Button subBtn = new Button("减少10分");
        
        // 注册自定义事件监听器
        root.addEventHandler(ScoreEvent.SCORE_CHANGED, e -> {
            scoreLabel.setText("分数: " + e.getScore());
            
            if (e.getScore() >= 100) {
                scoreLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold; -fx-text-fill: green;");
            } else if (e.getScore() < 0) {
                scoreLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold; -fx-text-fill: red;");
            } else {
                scoreLabel.setStyle("-fx-font-size: 24px; -fx-font-weight: bold;");
            }
        });
        
        addBtn.setOnAction(e -> {
            score += 10;
            root.fireEvent(new ScoreEvent(score));  // 触发自定义事件
        });
        
        subBtn.setOnAction(e -> {
            score -= 10;
            root.fireEvent(new ScoreEvent(score));
        });
        
        root.getChildren().addAll(scoreLabel, addBtn, subBtn);
        
        Scene scene = new Scene(root, 350, 250);
        primaryStage.setTitle("自定义事件示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```

## 8.8 事件过滤器

事件过滤器在捕获阶段处理事件，可以在事件到达目标之前拦截。

```java
package com.example.javafx.events;

import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.Label;
import javafx.scene.input.KeyCode;
import javafx.scene.input.KeyEvent;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class EventFilterDemo extends Application {

    @Override
    public void start(Stage primaryStage) {
        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        
        Label infoLabel = new Label("尝试按空格键或ESC键");
        infoLabel.setStyle("-fx-font-size: 14px;");
        
        Button button1 = new Button("按钮1");
        button1.setOnAction(e -> System.out.println("按钮1被点击"));
        
        Button button2 = new Button("按钮2");
        button2.setOnAction(e -> System.out.println("按钮2被点击"));
        
        root.getChildren().addAll(infoLabel, button1, button2);
        
        Scene scene = new Scene(root, 350, 200);
        
        // 添加事件过滤器 - 在事件到达按钮之前拦截
        scene.addEventFilter(KeyEvent.KEY_PRESSED, e -> {
            System.out.println("过滤器捕获按键: " + e.getCode());
            
            // 拦截空格键，不让它到达按钮
            if (e.getCode() == KeyCode.SPACE) {
                System.out.println("拦截空格键，阻止按钮被触发");
                e.consume();
            }
            
            // ESC键可以继续传播
            if (e.getCode() == KeyCode.ESCAPE) {
                System.out.println("ESC键通过过滤器");
            }
        });
        
        primaryStage.setTitle("事件过滤器示例");
        primaryStage.setScene(scene);
        primaryStage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
```
